1"""Shipped declarative-companion presets. 

2 

3A **preset** is a partial declarative-companion spec plus a bit of catalog 

4metadata (name/description i18n keys, icon, ``requires_integration`` gate). The 

5panel lists them under Settings → Companions → Declarative → *Add from preset*; 

6picking one seeds the Add dialog with the preset's ``default_spec``, the user 

7reviews, then Save persists a full spec (which the reconciler then materializes 

8into managed sensor tasks). 

9 

10Two presets ship in v1: 

11 

12* ``device_pulse`` — targets the standalone Device Pulse integration 

13 (studiobts/home-assistant-device-pulse) and watches its per-device 

14 ``sensor.*_total_failed_pings`` sensors via the existing ``threshold`` mode. 

15 Requires the Device Pulse integration to be installed. 

16* ``firmware_update_available`` — watches every ``update.*`` entity reporting 

17 ``on`` (HA's built-in firmware/software-update surface). Covers UniFi, ESPHome, 

18 HACS, Reolink, Bambu Lab firmware updates in one recipe. 

19 

20None of the shipped presets use the ``availability`` sensor mode — that mode is 

21added in this PR as a general capability for user-authored companions ("watch my 

22MQTT devices go offline"), not as a preset the panel installs. See 

23``declarative_companions.py`` for spec shape and ``docs/EVENTS.md`` / 

24``docs/INTEGRATING.md`` for the surface. 

25 

26Pure — no HA imports (the panel resolves ``name_key`` / ``description_key`` 

27through ``backend_i18n.resolve_string`` at request time). 

28"""

279def _pick(table: dict[str, str], lang: str | None) -> str: 

280 """*table*'s text for *lang*: exact, then the base language, then English.""" 

281 if lang: 

282 if lang in table: 

283 return table[lang] 

284 base = lang.split("-")[0].lower() 

285 for key, value in table.items(): 

286 if key.lower() == lang.lower() or key.lower() == base: 

287 return value 

288 return table[_DEFAULT_LANG]

291def localized_task_template(spec: dict[str, Any], lang: str | None) -> dict[str, Any]: 

292 """*spec*'s task template, with unchanged preset text put into *lang*. 

293 

294 A field is replaced only when the spec names a known preset and the field still 

295 reads as that preset's text in **some** language — so a recipe saved in English 

296 renders in German after the household switches, and back again. A field the user 

297 edited matches none of them and is returned as written. The input is not mutated. 

298 """ 

299 template = dict(spec.get("task_template") or {}) 

300 texts = PRESET_TASK_TEXT.get(str(spec.get("preset_id") or "")) 

301 if not texts: 

302 return template 

303 for field in _TEMPLATE_FIELDS: 

304 variants = texts[field] 

305 if template.get(field) in variants.values(): 

306 template[field] = _pick(variants, lang) 

307 return template

310def localized_default_spec( 

311 preset: PresetDefinition, lang: str | None, name: str 

312) -> dict[str, Any]: 

313 """A copy of *preset*'s ``default_spec`` in *lang*, named *name*. 

314 

315 This is what the panel seeds the Add dialog with, so a new recipe is saved in 

316 the household's language from the start. 

317 """ 

318 spec = dict(preset["default_spec"]) 

319 spec["name"] = name 

320 spec["task_template"] = localized_task_template(spec, lang) 

321 return spec
