1"""Shipped declarative-companion presets. 

2 

3A **preset** is a partial declarative-companion spec plus a bit of catalog 

4metadata (name/description i18n keys, icon, ``requires_integration`` gate). The 

5panel lists them under Settings → Companions → Declarative → *Add from preset*; 

6picking one seeds the Add dialog with the preset's ``default_spec``, the user 

7reviews, then Save persists a full spec (which the reconciler then materializes 

8into managed sensor tasks). 

9 

10Three general presets ship: 

11 

12* ``device_pulse`` — targets the standalone Device Pulse integration 

13 (studiobts/home-assistant-device-pulse) and watches its per-device 

14 ``sensor.*_total_failed_pings`` sensors via the existing ``threshold`` mode. 

15 Requires the Device Pulse integration to be installed. 

16* ``firmware_update_available`` — watches every ``update.*`` entity reporting 

17 ``on`` (HA's built-in firmware/software-update surface). Covers UniFi, ESPHome, 

18 HACS, Reolink, Bambu Lab firmware updates in one declarative companion. 

19* ``device_stopped_reporting`` — watches every ``sensor.*_last_seen`` entity through the 

20 ``template`` mode and opens a task once one is a day stale. Needs no upstream 

21 integration, and it is the worked example for what a template trigger is for. 

22 

23The **integration presets** follow them: one preset for each integration in 

24``declarative_presets_catalog.INTEGRATIONS`` and each way its readings become a task 

25(``SHAPES``). They select entities by their ``translation_key``, and each key gets 

26the task name of its duty from ``declarative_preset_text.DUTY_NAMES``. They are built 

27when this module loads, by :func:`_integration_presets`. 

28 

29No shipped preset uses the ``availability`` sensor mode. That mode is a general 

30capability for user-authored companions ("watch my MQTT devices go offline"), not a 

31preset the panel installs. See ``declarative_companions.py`` for spec shape and 

32``docs/EVENTS.md`` / ``docs/INTEGRATING.md`` for the surface. 

33 

34Pure — no HA imports (the panel resolves ``name_key`` / ``description_key`` 

35through ``backend_i18n.resolve_string`` at request time). 

36"""

373def _pick(table: dict[str, Any], lang: str | None) -> Any: 

374 """*table*'s text for *lang*: exact, then the base language, then English.""" 

375 if lang: 

376 if lang in table: 

377 return table[lang] 

378 base = lang.split("-")[0].lower() 

379 for key, value in table.items(): 

380 if key.lower() == lang.lower() or key.lower() == base: 

381 return value 

382 return table[_DEFAULT_LANG]

385def localized_task_template(spec: dict[str, Any], lang: str | None) -> dict[str, Any]: 

386 """*spec*'s task template, with unchanged preset text put into *lang*. 

387 

388 A field is replaced only when the spec names a known preset and the field still 

389 reads as that preset's text in **some** language — so a companion saved in English 

390 renders in German after the household switches, and back again. A field the user 

391 edited matches none of them and is returned as written. The input is not mutated. 

392 """ 

393 template = dict(spec.get("task_template") or {}) 

394 texts = PRESET_TASK_TEXT.get(str(spec.get("preset_id") or "")) 

395 if not texts: 

396 return template 

397 for field in _TEMPLATE_FIELDS: 

398 variants = texts.get(field) 

399 if variants and template.get(field) in variants.values(): 

400 template[field] = _pick(variants, lang) 

401 return template

419# 

420# One preset per integration and per *shape*: the way a kind of reading becomes a 

421# task. Each shape has one name string and one description string in 

422# ``backend_strings/`` with an ``{integration}`` placeholder, so 6 strings serve every 

423# integration. The English name here seeds ``default_spec["name"]``; the panel shows the 

424# localized one, and a test keeps the two equal. 

425SHAPES: dict[str, str] = { 

426 "percent_low": "{integration}: parts and supplies running low", 

427 "life_low": "{integration}: parts near the end of their life", 

428 "wear_high": "{integration}: wear counters", 

429 "reading_low": "{integration}: readings too low", 

430 "reading_high": "{integration}: readings too high", 

431 "alert": "{integration}: service alerts", 

432} 

433 

434# Hours in one of each time unit an entity can report. The time templates multiply the 

435# reading by this to compare it in hours, so the preset does not care whether the 

436# integration counts in seconds or days. A unit not in the table makes the ``life_low`` 

437# template fail to render, which decides nothing (a failed render neither opens nor 

438# closes a task), so a reading in an unknown unit is never compared as hours. 

439# 

440# Custom integrations often spell the unit out ("minutes", "days"), so the words are 

441# here too. A ``wear_high`` counter in a unit not in the table is compared as it is, so 

442# a time unit missing from it would be read as hours. 

443_TIME_FACTORS = ( 

444 "{'ms': 1 / 3600000, 's': 1 / 3600, 'sec': 1 / 3600, 'seconds': 1 / 3600, " 

445 "'min': 1 / 60, 'mins': 1 / 60, 'minutes': 1 / 60, 'h': 1, 'hr': 1, 'hrs': 1, " 

446 "'hours': 1, 'd': 24, 'day': 24, 'days': 24, 'w': 168, 'week': 168, " 

447 "'weeks': 168}" 

448)

451_PERCENT_FLOOR = 10

466def _limit(duties: list[dict[str, Any]]) -> tuple[float | None, str]: 

467 """The limit of *duties*: one number, or a per-key table for a template. 

468 

469 Returns ``(number, "")`` when every duty has the same limit, else ``(None, 

470 expression)`` where the expression looks the limit up by ``translation_key``. 

471 """ 

472 limits = {d["limit"] for d in duties} 

473 if len(limits) == 1: 

474 return next(iter(limits)), "" 

475 table = ", ".join(f"'{key}': {d['limit']}" for d in duties for key in d["keys"]) 

476 return None, "{" + table + "}[translation_key]"

479def _trigger(shape: str, duties: list[dict[str, Any]]) -> dict[str, Any]: 

480 """The trigger block for *duties*, which share *shape*. 

481 

482 Every integration preset closes its task itself when the reading recovers, so a 

483 reset on the device or a refill is what completes it. 

484 """ 

485 if shape == "alert": 

486 return {"mode": "state", "state": duties[0]["state"], "clear_on_recover": True} 

487 number, table = _limit(duties) 

488 limit = table or f"{number:g}" 

489 if shape == "life_low": 

490 # Some models of one integration report the life left as a percentage and 

491 # others as a time, under the same key. A percentage is compared with the 

492 # percentage floor, and a time in hours. 

493 template = ( 

494 f"{{{{ state | float < {_PERCENT_FLOOR} " 

495 "if attributes.unit_of_measurement == '%' " 

496 f"else state | float * {_TIME_FACTORS}[attributes.unit_of_measurement]" 

497 f" < {limit} }}}}" 

498 ) 

499 return {"mode": "template", "template": template, "clear_on_recover": True} 

500 if shape == "wear_high": 

501 # A counter in a unit that is not a time (washes, cycles) is compared as it is. 

502 template = ( 

503 f"{{{{ state | float * {_TIME_FACTORS}" 

504 f".get(attributes.get('unit_of_measurement'), 1) > {limit} }}}}" 

505 ) 

506 return {"mode": "template", "template": template, "clear_on_recover": True} 

507 comparison = "<" if shape in ("percent_low", "reading_low") else ">" 

508 if table: 

509 template = f"{{{{ state | float {comparison} {table} }}}}" 

510 return {"mode": "template", "template": template, "clear_on_recover": True} 

511 return { 

512 "mode": "threshold", 

513 "comparison": comparison, 

514 "value": number, 

515 "clear_on_recover": True, 

516 }

519def _integration_presets() -> tuple[ 

520 list[PresetDefinition], dict[str, dict[str, dict[str, Any]]] 

521]: 

522 """Build the integration presets and their task text from the catalog.""" 

523 built: list[PresetDefinition] = [] 

524 texts: dict[str, dict[str, dict[str, Any]]] = {} 

525 for entry in INTEGRATIONS: 

526 groups: dict[tuple[str, str, str], list[dict[str, Any]]] = {} 

527 for duty in entry["duties"]: 

528 platform = duty.get("platform", "sensor") 

529 group = (duty["shape"], platform, duty.get("state", ""))
