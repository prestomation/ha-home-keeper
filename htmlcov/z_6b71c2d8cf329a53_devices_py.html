1"""Registry-device provisioning for Home Keeper assets. 

2 

3This is the *virtual-device* half of the asset feature (see ``IDEAS.md`` / 

4``docs/DESIGN.md``): when an appliance has no Home Assistant device to attach 

5maintenance tasks to, Home Keeper registers a real device-registry entry for it so 

6tasks, future batteries, and asset metadata all converge on one device page. 

7 

8Devices we create are tied to our config entry, so Home Assistant removes them 

9automatically when the integration is removed — no residue. Reconciliation is 

10idempotent: it can run on every setup and after every asset mutation. 

11"""

120async def async_reconcile_assets( 

121 hass: HomeAssistant, entry: ConfigEntry, store: HomeKeeperStore 

122) -> None: 

123 """Make the device registry match the current set of assets. 

124 

125 * Virtual assets get a registry device created (idempotently) and kept in sync 

126 with their editable fields; the assigned ``device.id`` is written back to the 

127 asset so tasks/metadata entities can resolve it. 

128 * Existing-device assets get their identifiers/connections snapshotted for 

129 reconciliation, and a warning if the referenced device has gone away. 

130 * Orphan asset devices (ours, but no longer backed by an asset) are removed. 

131 """ 

132 registry = dr.async_get(hass) 

133 wanted_identifiers: set[tuple[str, str]] = set() 

134 dirty = False 

135 

136 # Provision parents before children so a subdevice's via_device parent already 

137 # has a resolved device id when we link it. 

138 for asset in sorted(store.list_assets(), key=lambda a: _ancestor_depth(store, a)): 

139 if asset.get("kind") == ASSET_KIND_VIRTUAL: 

140 # _reconcile_virtual persists its own device_id write-back. 

141 await _reconcile_virtual(hass, entry, store, registry, asset) 

142 wanted_identifiers.add(asset_model.asset_device_identifier(asset["id"])) 

143 elif _reconcile_existing(registry, asset): 

144 dirty = True 

145 

146 # In-place edits to existing-device assets (recovered device_id, refreshed 

147 # identifiers/connections snapshot) must be flushed to disk or they're lost on 

148 # restart and snapshot recovery can never work. 

149 if dirty: 

150 await store.async_persist() 

151 

152 # Prune asset devices we own that no longer correspond to an asset. Guard 

153 # against ever removing a per-task self-owned device — those key on the bare 

154 # task id — by excluding any device that matches a current task's identifier, 

155 # even in the (uuid-impossible) case a task id collides with the asset prefix. 

156 task_identifiers = {(DOMAIN, tid) for tid in store.get_tasks()} 

157 for device in dr.async_entries_for_config_entry(registry, entry.entry_id): 

158 if ( 

159 _is_asset_device(device) 

160 and not (device.identifiers & wanted_identifiers) 

161 and not (device.identifiers & task_identifiers) 

162 ): 

163 _LOGGER.debug("Removing orphaned asset device %s", device.id) 

164 registry.async_remove_device(device.id)

167def _split_successor( 

168 registry: dr.DeviceRegistry, 

169 device_id: str, 

170 entry_id: str, 

171 *, 

172 snapshot: dict[str, Any] | None = None, 

173) -> dr.DeviceEntry | None: 

174 """The live device a pre-2026.8 *composite* device id should now point at. 

175 

176 Home Assistant keeps the old id resolvable: ``async_get`` synthesizes a read-only 

177 composite for it, which is why a stale id looks perfectly healthy and why this asks 

178 ``async_get_devices_for_composite_device_id`` instead — that returns the real, live 

179 devices the id was split into, and an empty list for an ordinary id. 

180 

181 Of those, the one to adopt is the split that is **not ours**: our half is an 

182 artefact of the old identifier copy, and the task is about the other integration's 

183 device. Where several remain, the composite's former ``primary_config_entry`` names 

184 the successor Home Assistant itself treats as primary; the sorted fallback keeps 

185 repeated runs deterministic. 

186 

187 Once every half has been re-homed Home Assistant garbage-collects the composite, 

188 and ``async_get_devices_for_composite_device_id`` then answers the same empty list 

189 it gives an ordinary id — the composite is gone, so there is nothing left to ask. 

190 A caller that holds an identifiers/connections *snapshot* of the original device 

191 can still resolve it, so pass one and the lookup falls back to matching that 

192 snapshot against the live registry (``_resolve_by_snapshot``). Assets keep such a 

193 snapshot; tasks do not. 

194 

195 Returns ``None`` on Home Assistant versions with no composite concept (pre-2026.8), 

196 where there is nothing to heal. 

197 """ 

198 resolve = getattr(registry, "async_get_devices_for_composite_device_id", None) 

199 if resolve is None: 

200 return None 

201 splits = [d for d in resolve(device_id) if entry_id not in d.config_entries] 

202 if not splits: 

203 # An empty answer means one of two very different things: a collected 

204 # composite, or an ordinary id that was never split. Only the first is ours 

205 # to repair, and ``async_get`` separates them — a composite is synthesized 

206 # from the devices still pointing at it, so once they are all re-homed there 

207 # is nothing left to synthesize and this returns None, while a live device 

208 # answers for itself. Without the check a healthy asset whose snapshot has 

209 # drifted (identifiers moved to another device, and the heal runs before 

210 # ``_reconcile_existing`` refreshes them) would be silently repointed onto 

211 # whatever the stale snapshot happened to match. 

212 if snapshot is not None and registry.async_get(device_id) is None: 

213 return _resolve_by_snapshot(registry, snapshot, prefer_not_entry=entry_id) 

214 return None 

215 if len(splits) > 1: 

216 composite = registry.async_get(device_id) 

217 primary = getattr(composite, "primary_config_entry", None) 

218 if primary: 

219 preferred = [d for d in splits if primary in d.config_entries] 

220 if preferred: 220 ↛ 227line 220 didn't jump to line 227 because the condition on line 220 was always true

221 return preferred[0] 

222 # Three or more foreign splits with no primary named: nothing in the registry 

223 # says which one the task meant, so this is an arbitrary pick — sorted only so 

224 # that repeated setups keep choosing the same one instead of flip-flopping the 

225 # task between devices. A two-way split, which is what the merge this repairs 

226 # actually produces, never reaches here. 

227 return sorted(splits, key=lambda d: d.id)[0] 

228 return splits[0]

231async def async_heal_split_device_ids( 

232 hass: HomeAssistant, entry: ConfigEntry, store: HomeKeeperStore 

233) -> None: 

234 """Re-point tasks whose device Home Assistant 2026.8 split apart (#183). 

235 

236 An install that upgraded Home Assistant before Home Keeper has, per attached thing, 

237 a device id that is now a *composite*: Home Assistant split the merged device into 

238 one per config entry and handed us our own half. Left alone that shows a raw 

239 identifier where a device name belongs, splits a device's tasks across two entries, 

240 and makes companion integrations create duplicate tasks. 

241 

242 It does **not** present as a missing device. Home Assistant answers ``async_get`` 

243 for the old id with a synthesized composite, so a naive "does this still resolve?" 

244 check finds nothing wrong — it only refuses to *link an entity* to it, which is the 

245 visible symptom and the reason this repair exists. 

246 

247 Assets of kind ``existing`` carry the same kind of reference and are healed in the 

248 same pass, against the same mapping. Sharing one mapping is what keeps a device's 

249 task and its asset together: only the asset keeps an identifiers/connections 

250 snapshot, so a composite Home Assistant has already garbage-collected is resolvable 

251 *only* from the asset — and a task on that same device has to inherit the answer or 

252 it stays pointing at the dead id while the asset moves on without it. 

253 

254 Runs before the platforms set up, so entities are created against the healed id 

255 straight away. Our old half then holds none of our entities and 

256 ``async_prune_orphaned_devices`` removes it at the end of setup. Idempotent: once 

257 healed nothing resolves to a composite and this does no writes. 

258 """ 

259 registry = dr.async_get(hass) 

260 # One mapping (dead id -> live id) for tasks and assets alike. 

261 mapping: dict[str, str] = {} 

262 # Ids the snapshot-less pass could not resolve, so a hundred tasks on one dead 

263 # device cost one lookup rather than a hundred. Assets still retry these: a 

264 # snapshot is exactly the extra information that can resolve them. 

265 unresolved: set[str] = set()

267 def _record(device_id: str | None, snapshot: dict[str, Any] | None = None) -> None: 

268 """Resolve *device_id*'s successor into the shared mapping, if it has one.""" 

269 if not device_id or device_id in mapping: 

270 return 

271 if snapshot is None and device_id in unresolved: 

272 return 

273 successor = _split_successor( 

274 registry, device_id, entry.entry_id, snapshot=snapshot 

275 ) 

276 # A successor equal to the id we started from is not a repair; skipping it 

277 # keeps the pass a genuine no-op once everything is healed. 

278 if successor is not None and successor.id != device_id: 

279 mapping[device_id] = successor.id 

280 elif snapshot is None: 

281 unresolved.add(device_id) 

282 

283 for task in store.get_tasks().values(): 

284 # No snapshot: a task stores only the id. A GC'd composite is unresolvable 

285 # from here, and gets picked up below if an asset shares the device. 

286 _record(task.get("device_id")) 

287 for asset in store.list_assets(): 

288 if asset.get("kind") == ASSET_KIND_EXISTING: 

289 # The asset dict doubles as the snapshot — ``_reconcile_existing`` keeps 

290 # its identifiers/connections refreshed from the live device. 

291 _record(asset.get("device_id"), asset) 

292 

293 if mapping: 

294 changed_tasks = await store.async_repoint_device_ids(mapping) 

295 changed_assets = await store.async_repoint_asset_device_ids(mapping) 

296 _LOGGER.info( 

297 "Repaired %s task and %s asset device reference(s) across %s device(s) " 

298 "that Home Assistant 2026.8 split into one device per config entry", 

299 changed_tasks, 

300 changed_assets, 

301 len(mapping), 

302 ) 

303 

304 # Healing the ids stops contributors duplicating from here on, but a duplicate 

305 # already created while the ids were broken is still sitting there, and the user 

306 # can't remove it — contributed tasks are deletion-protected. Merging needs to know 

307 # which live devices came from the same original, or the two copies look unrelated: 

308 # they point at different halves of the same split. 

309 canonical: dict[str, str] = {} 

310 for device in registry.devices.values(): 

311 # Explicit loop rather than a comprehension: composite_device_id is 

312 # `str | None`, and a truthiness guard inside a comprehension doesn't narrow 

313 # the value expression for the type checker. 

314 composite_id = getattr(device, "composite_device_id", None) 

315 if composite_id: 

316 canonical[device.id] = composite_id 

317 merged = await store.async_merge_split_duplicates(canonical) 

318 if merged: 318 ↛ 319line 318 didn't jump to line 319 because the condition on line 318 was never true

319 _LOGGER.info("Merged %s duplicate contributed task(s) after the split", merged)

322async def async_detach_legacy_merged_devices( 

323 hass: HomeAssistant, entry: ConfigEntry 

324) -> None: 

325 """Drop our config entry from devices an older Home Keeper merged onto. 

326 

327 Before HA 2026.8, attaching a task copied the target device's identifiers into a 

328 ``DeviceInfo``, which merged our entities onto that device **and** added our config 

329 entry to it. This version links entities instead (see 

330 ``coordinator.device_link_for_task``) and never needs to be on a device it doesn't 

331 own, but the old association survives in the registry on its own. 

332 

333 That association is what makes Home Assistant split the device on upgrade, so 

334 clearing it is what lets someone still on a pre-2026.8 Home Assistant update Home 

335 Keeper first and then upgrade cleanly. Measured in 

336 ``tests/upgrade/test_upgrade_order.py``: without this, every upgrade order is 

337 damaged identically, and there is no advice worth giving. 

338 

339 Deliberately limited to devices that have **another** config entry besides ours. 

340 Removing the last entry from a device deletes it, which would strand the entities 

341 sitting on it — that case is the already-split leftover, and repairing it needs the 

342 entities re-pointed first (still open; see ``docs/DEVICE_REGISTRY_2026_8_PLAN.md``). 

343 

344 That check and the update below are not atomic: nothing stops another integration 

345 removing its own entry in between, which would make ours the last one after all. 

346 Home Assistant's registry offers no remove-if-not-last, and the window is a few 

347 synchronous statements during our setup, so this is the best the API allows. 

348 Our own devices are skipped: they carry a ``home_keeper`` identifier, and we are 

349 supposed to own those. 

350 """ 

351 dev_reg = dr.async_get(hass) 

352 for device in list(dr.async_entries_for_config_entry(dev_reg, entry.entry_id)): 

353 if any(domain == DOMAIN for domain, _ in device.identifiers): 

354 continue # one of ours (virtual asset or self-owned task device) 

355 others = set(device.config_entries) - {entry.entry_id} 

356 if not others: 

357 continue # sole owner: removing us would delete the device and its entities 

358 _LOGGER.debug( 

359 "Detaching Home Keeper from %s: linked, not owned (legacy merge)", device.id 

360 ) 

361 dev_reg.async_update_device(device.id, remove_config_entry_id=entry.entry_id)

364async def async_prune_orphaned_devices(hass: HomeAssistant, entry: ConfigEntry) -> None: 

365 """Drop Home Keeper from devices that no longer carry any of our entities. 

366 

367 Per-task entities (``sensor.*_next_due``, ``binary_sensor.*_overdue``) attach to 

368 the task's device — an existing/shared device, or a self-owned device we create 

369 when the task has none. When the task goes away (Problem Sensor Sync disabled, an 

370 entity/device/area/label excluded, or an ordinary delete) the platforms remove 

371 those entities, but Home Assistant keeps the device/association — so the device 

372 lingers under **Settings → Devices & Services → Home Keeper** with zero entities. 

373 

374 Call this *after* ``async_forward_entry_setups`` so the entity registry already 

375 reflects the current set, then drop our config-entry link from any non-asset 

376 device we no longer have an entity on. ``async_update_device`` removes a 

377 self-owned device outright (we were its last config entry) and merely detaches us 

378 from a shared device (its real owner keeps it). Virtual asset devices may be 

379 legitimately entity-less and are reconciled by ``async_reconcile_assets``, so 

380 they are skipped here. 

381 """ 

382 dev_reg = dr.async_get(hass) 

383 ent_reg = er.async_get(hass) 

384 for device in list(dr.async_entries_for_config_entry(dev_reg, entry.entry_id)): 

385 if _is_asset_device(device): 

386 continue 

387 has_our_entity = any( 

388 entity.config_entry_id == entry.entry_id 

389 for entity in er.async_entries_for_device( 

390 ent_reg, device.id, include_disabled_entities=True 

391 ) 

392 ) 

393 if not has_our_entity: 

394 _LOGGER.debug("Pruning Home Keeper from orphaned device %s", device.id) 

395 dev_reg.async_update_device( 

396 device.id, remove_config_entry_id=entry.entry_id 

397 )

400async def _reconcile_virtual( 

401 hass: HomeAssistant, 

402 entry: ConfigEntry, 

403 store: HomeKeeperStore, 

404 registry: dr.DeviceRegistry, 

405 asset: dict[str, Any], 

406) -> None: 

407 identifier = asset_model.asset_device_identifier(asset["id"]) 

408 

409 # Resolve the native via_device parent (only our own virtual subdevices). 

410 parent_asset_id = asset.get("parent_asset_id") or None 

411 via_device = ( 

412 asset_model.asset_device_identifier(parent_asset_id) 

413 if parent_asset_id 

414 else None 

415 ) 

416 parent = store.get_asset(parent_asset_id) if parent_asset_id else None 

417 parent_device_id = parent.get("device_id") if parent else None 

418 

419 configuration_url = _asset_configuration_url(asset["id"]) 

420 create_kwargs: dict[str, Any] = { 

421 "config_entry_id": entry.entry_id, 

422 "identifiers": {identifier}, 

423 "name": asset["name"], 

424 "manufacturer": asset.get("manufacturer") or None, 

425 "model": asset.get("model") or None, 

426 "configuration_url": configuration_url, 

427 } 

428 # serial_number reached DeviceInfo/async_get_or_create later than the others; only 

429 # seed it on create when this HA version accepts it (the update loop below is 

430 # likewise guarded), so an older core still provisions the device cleanly. 

431 if _supports_kwarg(registry.async_get_or_create, "serial_number"): 

432 create_kwargs["serial_number"] = asset.get("serial_number") or None 

433 if via_device is not None: 

434 create_kwargs["via_device"] = via_device 

435 device = registry.async_get_or_create(**create_kwargs) 

436 

437 # Keep the registry in sync with subsequent edits. 

438 updates: dict[str, Any] = {} 

439 if device.name != asset["name"]: 

440 updates["name"] = asset["name"] 

441 for field in ("manufacturer", "model", "serial_number"): 

442 desired = asset.get(field) or None 

443 if getattr(device, field, None) != desired and _supports_kwarg( 

444 registry.async_update_device, field 

445 ): 

446 updates[field] = desired 

447 # Note the absence of a truthy guard on area_id: passing None propagates an 

448 # area *clear* to the device (a truthy-only check would strand the old area). 

449 # Guard against a dangling area_id (the area was deleted in HA after it was 

450 # assigned): async_update_device rejects an unknown area, so self-heal to None. 

451 area_id = asset.get("area_id") or None 

452 if area_id and not area_exists(hass, area_id): 

453 area_id = None 

454 if area_id != device.area_id: 

455 updates["area_id"] = area_id 

456 if device.configuration_url != configuration_url: 

457 updates["configuration_url"] = configuration_url 

458 # Re-parent / un-parent after creation (via_device on create only applies the 

459 # first time). via_device_id is the parent's *device id*, which parents-first 

460 # ordering has already resolved. 

461 if _supports_kwarg(registry.async_update_device, "via_device_id"): 

462 if device.via_device_id != parent_device_id: 

463 updates["via_device_id"] = parent_device_id 

464 elif parent_asset_id and device.via_device_id is None: 

465 _LOGGER.warning( 

466 "This Home Assistant version can't update a device's parent after " 

467 "creation; subdevice %s may not nest under its parent", 

468 asset["id"], 

469 ) 

470 if updates: 

471 updated = registry.async_update_device(device.id, **updates) 

472 if updated is not None: 

473 device = updated 

474 

475 await store.set_asset_device_id(asset["id"], device.id)

478def _reconcile_existing(registry: dr.DeviceRegistry, asset: dict[str, Any]) -> bool: 

479 """Resolve an existing-device asset and refresh its snapshot in place. 

480 

481 Returns ``True`` if the asset dict was mutated (so the caller persists it). 

482 """ 

483 device_id = asset.get("device_id") 

484 device = registry.async_get(device_id) if device_id else None 

485 changed = False 

486 if device is None: 

487 # The referenced device may have been recreated under a new id by its 

488 # owning integration; try to recover it from the stored snapshot. 

489 device = _resolve_by_snapshot(registry, asset) 

490 if device is None: 

491 _LOGGER.warning( 

492 "Home Keeper asset %s references missing device %s; metadata " 

493 "entities will not appear until the device returns", 

494 asset["id"], 

495 device_id, 

496 ) 

497 return False 

498 asset["device_id"] = device.id 

499 changed = True 

500 # Refresh the reconciliation snapshot from the live device (only marking the 

501 # asset dirty when it actually changed, to avoid needless writes). 

502 identifiers = [list(i) for i in device.identifiers] 

503 connections = [list(c) for c in device.connections] 

504 if asset.get("identifiers") != identifiers: 

505 asset["identifiers"] = identifiers 

506 changed = True 

507 if asset.get("connections") != connections: 

508 asset["connections"] = connections 

509 changed = True 

510 return changed

513def _resolve_by_snapshot( 

514 registry: dr.DeviceRegistry, 

515 snapshot: dict[str, Any], 

516 *, 

517 prefer_not_entry: str | None = None, 

518) -> dr.DeviceEntry | None: 

519 """Find a live device matching a snapshot's stored identifiers/connections. 

520 

521 *snapshot* is an asset dict (or any mapping with ``identifiers`` and 

522 ``connections`` keys). Serves two callers: 

523 

524 * ``_reconcile_existing`` recovering an asset whose device its owning integration 

525 re-created under a new id — no preference, first match wins. 

526 * ``_split_successor`` resolving a composite Home Assistant has already 

527 garbage-collected. There, ``prefer_not_entry`` names *our* config entry, and a 

528 matching device that isn't ours wins: after a 2026.8 split the identifiers were 

529 copied onto our half too, and the one worth adopting is the other integration's 

530 real device, not the artefact. The sorted tie-break only keeps repeated setups 

531 choosing the same device rather than flip-flopping between them. 

532 """ 

533 candidates: list[dr.DeviceEntry] = [] 

534 seen: set[str] = set() 

535 

536 def _add(device: dr.DeviceEntry | None) -> None: 

537 if device is not None and device.id not in seen: 

538 seen.add(device.id) 

539 candidates.append(device) 

540 

541 for ident in snapshot.get("identifiers", []): 

542 _add(registry.async_get_device(identifiers={tuple(ident)})) 

543 # With no preference there is nothing a second candidate could change, so 

544 # stop at the first hit rather than finishing the sweep. 

545 if prefer_not_entry is None and candidates: 

546 return candidates[0] 

547 connections = {tuple(c) for c in snapshot.get("connections", [])} 

548 if connections: 

549 _add(registry.async_get_device(connections=connections)) 

550 

551 if not candidates: 

552 return None 

553 if prefer_not_entry is None: 

554 return candidates[0] 

555 foreign = [d for d in candidates if prefer_not_entry not in d.config_entries] 

556 return sorted(foreign or candidates, key=lambda d: d.id)[0] 

557 

558 

559async def async_remove_asset_device( 

560 hass: HomeAssistant, asset: dict[str, Any] 

561) -> str | None: 

562 """Remove the virtual device backing *asset*; returns its device id if removed. 

563 

564 No-op for existing-device assets (we never owned that device). Tasks attached 

565 to a removed virtual device should be detached by the caller. 

566 """ 

567 if asset.get("kind") != ASSET_KIND_VIRTUAL: 

568 return None 

569 registry = dr.async_get(hass) 

570 identifier = asset_model.asset_device_identifier(asset["id"]) 

571 device = registry.async_get_device(identifiers={identifier}) 

572 if device is not None: 

573 registry.async_remove_device(device.id) 

574 return device.id 

575 return asset.get("device_id")
