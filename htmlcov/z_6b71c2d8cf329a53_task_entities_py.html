1"""Names and identity of a task's per-task device-page entities. Pure, no HA imports. 

2 

3A task attached to an existing device gets three entities on that device's page 

4(mark-done button, next-due sensor, overdue binary sensor). Several tasks can share 

5one device page, so each entity name starts with a short label for its task. Home 

6Assistant then puts the device name in front of the whole name, and makes the 

7``entity_id`` from the result the first time the entity registers. 

8 

9:func:`entity_name_prefix` picks that label. :func:`entity_set_key` says when the 

10entities must be made again (an entry reload), because Home Assistant caches an 

11entity's name. 

12""" 

13 

14from __future__ import annotations 

15 

16import re 

17from typing import Any 

18 

19from .declarative_companions import declarative_source 

20from .notifications import is_completion_blocked

22# Separators left over when the device name is cut off the start or end of a task 

23# name: "Kitchen Sensor: Check battery" -> ": Check battery" -> "Check battery". 

24_TRIM = " \t:-\u2013\u2014," 

25 

26 

27def recipe_name(task: dict[str, Any]) -> str | None: 

28 """The name of the recipe that owns *task*, or ``None`` for any other task. 

29 

30 ``build_managed_by`` stamps the recipe name as ``managed_by.display_name``. A 

31 problem-sensor task stamps a display name too, so the recipe source is what 

32 decides, not the display name alone. 

33 """ 

34 if declarative_source(task) is None: 

35 return None 

36 managed_by = task.get("managed_by") 

37 if not isinstance(managed_by, dict): 

38 return None 

39 name = managed_by.get("display_name") 

40 if isinstance(name, str) and name.strip(): 

41 return name.strip() 

42 return None

45def _strip_device_name(name: str, device_name: str) -> str: 

46 """*name* with *device_name* cut off its start or end, as a whole word.""" 

47 escaped = re.escape(device_name.strip()) 

48 for pattern in (rf"^{escaped}(?!\w)", rf"(?<!\w){escaped}$"): 

49 name = re.sub(pattern, "", name, count=1, flags=re.IGNORECASE) 

50 return name.strip(_TRIM)

53def entity_name_prefix(task: dict[str, Any], device_name: str | None) -> str: 

54 """The label for *task* in front of its entity names on *device_name*'s page. 

55 

56 A recipe task uses the recipe name: it is short, the user chose it, and it says 

57 which rule opened the task. The rendered task name usually holds the device name, 

58 which Home Assistant adds again, so it made long, repeated names and entity_ids 

59 (#377). Any other task uses its name without the device name at its start or end, 

60 or its whole name when nothing else is left. The caller adds the ``": "``. 

61 """ 

62 recipe = recipe_name(task) 

63 if recipe is not None: 

64 return recipe 

65 name = str(task.get("name") or "").strip() 

66 if not name or not device_name or not device_name.strip(): 

67 return name 

68 return _strip_device_name(name, device_name) or name

71def entity_set_key(task: dict[str, Any] | None) -> tuple[Any, ...]: 

72 """Identity of a task's per-task entity set. 

73 

74 Per-task entities (button/sensor/binary_sensor) exist only for an enabled, 

75 device-attached task. When this key changes between an update's before and after, 

76 the entry must be reloaded so entities are created, removed or renamed; otherwise 

77 a plain coordinator refresh is enough. 

78 

79 * ``name`` and the recipe name are in the key because Home Assistant caches an 

80 entity's computed ``name``. Making the entity again on reload is how a rename 

81 reaches the device page (and how a self-owned task device gets its new name). 

82 * ``completion_blocked`` is in the key because it decides whether the task has a 

83 mark-done button at all. 

84 """ 

85 if not task: 

86 return (None, False, None, None, False) 

87 return ( 

88 task.get("device_id"), 

89 bool(task.get("enabled", True)), 

90 task.get("name"), 

91 recipe_name(task), 

92 is_completion_blocked(task), 

93 )
