1"""Names and identity of a task's per-task device-page entities. Pure, no HA imports. 

2 

3A task attached to an existing device gets three entities on that device's page 

4(mark-done button, next-due sensor, overdue binary sensor). Several tasks can share 

5one device page, so each entity name starts with a short label for its task. Home 

6Assistant then puts the device name in front of the whole name, and makes the 

7``entity_id`` from the result the first time the entity registers. 

8 

9:func:`entity_name_prefix` picks that label. :func:`entity_set_key` says when the 

10entities must be made again (an entry reload), because Home Assistant caches an 

11entity's name. 

12""" 

13

25# Separators left over when the device name is cut off the start or end of a task 

26# name: "Kitchen Sensor: Check battery" -> ": Check battery" -> "Check battery". 

27_TRIM = " \t:-\u2013\u2014,"

30def companion_name(task: dict[str, Any]) -> str | None: 

31 """The name of the declarative companion that owns *task*, else ``None``. 

32 

33 ``build_managed_by`` stamps the companion name as ``managed_by.display_name``. A 

34 problem-sensor task stamps a display name too, so the companion source is what 

35 decides, not the display name alone. 

36 """ 

37 if declarative_source(task) is None: 

38 return None 

39 managed_by = task.get("managed_by") 

40 if not isinstance(managed_by, dict): 

41 return None 

42 name = managed_by.get("display_name") 

43 if isinstance(name, str) and name.strip(): 

44 return name.strip() 

45 return None

48def _strip_device_name(name: str, device_name: str) -> str: 

49 """*name* with *device_name* cut off its start or end, as a whole word.""" 

50 escaped = re.escape(device_name.strip()) 

51 for pattern in (rf"^{escaped}(?!\w)", rf"(?<!\w){escaped}$"): 

52 name = re.sub(pattern, "", name, count=1, flags=re.IGNORECASE) 

53 return name.strip(_TRIM)

56def _shares_companion_on_device( 

57 task: dict[str, Any], tasks: Iterable[dict[str, Any]] 

58) -> bool: 

59 """Whether another enabled task of *task*'s companion is on the same device.""" 

60 source = declarative_source(task) or {} 

61 for other in tasks: 

62 if other.get("id") == task.get("id") or not other.get("enabled", True): 

63 continue 

64 if other.get("device_id") != task.get("device_id"): 

65 continue 

66 if (declarative_source(other) or {}).get("spec_id") == source.get("spec_id"): 

67 return True 

68 return False

71def entity_name_prefix( 

72 task: dict[str, Any], 

73 device_name: str | None, 

74 tasks: Iterable[dict[str, Any]] = (), 

75) -> str: 

76 """The label for *task* in front of its entity names on *device_name*'s page. 

77 

78 A declarative companion task uses the companion name: it is short, the user chose 

79 it, and it says which rule opened the task. The rendered task name usually holds the 

80 device name, which Home Assistant adds again, so it made long, repeated names and 

81 entity_ids (#377). Any other task uses its name without the device name at its start 

82 or end, or its whole name when nothing else is left. The caller adds the ``": "``. 

83 

84 *tasks* is every task. When one companion has more than one enabled task on the 

85 device, as a printer preset has one per ink, the companion name is the same on 

86 each, so each task uses its own name as any other task does (B15-1). 

87 """ 

88 companion = companion_name(task) 

89 if companion is not None and not _shares_companion_on_device(task, tasks): 

90 return companion 

91 name = str(task.get("name") or "").strip() 

92 if not name or not device_name or not device_name.strip(): 

93 return name 

94 return _strip_device_name(name, device_name) or name

97def has_mark_done_button(task: dict[str, Any]) -> bool: 

98 """Whether a device-attached *task* gets a mark-done button. 

99 

100 The button completes the task with no origin. A task that refuses that 

101 completion gets no button, because each press can only fail (B15-2): 

102 

103 * a problem-sensor task: the integration that owns the sensor clears it. 

104 * a declarative companion task that clears itself (``completion_blocked``). 

105 * a task with ``require_tag_scan``: only a scan of its tag completes it. 

106 """ 

107 return ( 

108 problem_source(task) is None 

109 and not is_completion_blocked(task) 

110 and completion_allowed(task, None) 

111 )

114def entity_set_key(task: dict[str, Any] | None) -> tuple[Any, ...]: 

115 """Identity of a task's per-task entity set. 

116 

117 Per-task entities (button/sensor/binary_sensor) exist only for an enabled, 

118 device-attached task. When this key changes between an update's before and after, 

119 the entry must be reloaded so entities are created, removed or renamed; otherwise 

120 a plain coordinator refresh is enough. 

121 

122 * ``name`` and the companion name are in the key because Home Assistant caches an 

123 entity's computed ``name``. Making the entity again on reload is how a rename 

124 reaches the device page (and how a self-owned task device gets its new name). 

125 * ``completion_blocked`` and ``require_tag_scan`` are in the key because they 

126 decide whether the task has a mark-done button at all 

127 (:func:`has_mark_done_button`). 

128 """ 

129 if not task: 

130 return (None, False, None, None, False, False) 

131 return ( 

132 task.get("device_id"), 

133 bool(task.get("enabled", True)), 

134 task.get("name"), 

135 companion_name(task), 

136 is_completion_blocked(task), 

137 bool(task.get("require_tag_scan")), 

138 )
