1"""Pure edge-detection for Home Keeper's time-based task events. 

2 

3The coordinator wakes periodically (and after every mutation) and holds the current 

4task map, but a due date passing is not itself a mutation — nothing *calls* anything 

5when ``now`` crosses a task's ``next_due``. This module turns "what the tasks look 

6like now" plus "what we already announced" into the set of `home_keeper_task_overdue` 

7/ `home_keeper_task_due_soon` events to fire, **edge-triggered** so each is announced 

8at most once per ``next_due`` value. 

9 

10It imports nothing from Home Assistant (only the pure ``recurrence``/``events`` core 

11and ``const``), so the crossing logic — fire-once, reset-on-reschedule, skip 

12dormant/disabled — is unit-testable with an injected ``now``. The coordinator owns the 

13small state map and does the actual ``hass.bus`` firing; see 

14``coordinator._async_update_data``. 

15""" 

16 

17from __future__ import annotations 

18 

19from datetime import timedelta 

20from typing import Any 

21 

22from . import events, recurrence 

23from .const import EVENT_TASK_DUE_SOON, EVENT_TASK_OVERDUE 

24 

25# How far ahead of ``next_due`` a task counts as "due soon". Matches the binary 

26# sensor's ``due_soon`` attribute window so the event and the entity agree. 

27DUE_SOON_WINDOW = timedelta(days=3) 

28 

29# Per-task edge state carried between coordinator refreshes. Keyed by task id: 

30# {"next_due": <iso|None>, "due_soon_fired": bool, "overdue_fired": bool} 

31# The two flags are independent because a task crosses *due-soon first, then overdue* 

32# against the **same** ``next_due`` — a bare next_due key couldn't fire both. The flags 

33# reset whenever ``next_due`` changes (a completion/reschedule/re-arm), which naturally 

34# re-arms the next announcement. 

35TaskEdgeState = dict[str, Any] 

36StateMap = dict[str, TaskEdgeState] 

37 

38 

39def detect_transitions( 

40 prev: StateMap, 

41 tasks: dict[str, dict[str, Any]], 

42 *, 

43 now: Any, 

44 window: timedelta = DUE_SOON_WINDOW, 

45) -> tuple[list[tuple[str, dict[str, Any]]], StateMap]: 

46 """Return ``(fired, next_state)`` for the current *tasks* against *prev* state. 

47 

48 ``fired`` is a list of ``(event_name, payload)`` ready for the caller to put on the 

49 bus; ``next_state`` is the new per-task edge state to carry forward. A task fires 

50 ``due_soon`` once when it enters the window and ``overdue`` once when it reaches 

51 ``next_due`` — never again for the same ``next_due``. Dormant (``next_due is None``) 

52 and disabled tasks never fire (they're still tracked, so re-arming/re-enabling 

53 starts fresh). Tasks absent from *tasks* drop out of the state (deleted tasks can't 

54 leak stale flags). 

55 """ 

56 fired: list[tuple[str, dict[str, Any]]] = [] 

57 next_state: StateMap = {} 

58 

59 for tid, task in tasks.items(): 

60 next_due = task.get("next_due") 

61 prior = prev.get(tid) 

62 if prior is not None and prior.get("next_due") == next_due: 

63 due_soon_fired = bool(prior.get("due_soon_fired")) 

64 overdue_fired = bool(prior.get("overdue_fired")) 

65 else: 

66 # New task, or its schedule moved — re-arm both announcements. 

67 due_soon_fired = False 

68 overdue_fired = False 

69 

70 if task.get("enabled", True) and next_due is not None: 

71 parsed = recurrence._parse(next_due) 

72 if not due_soon_fired and recurrence.is_due_soon(task, window, now=now): 

73 due_in_hours = round((parsed - now).total_seconds() / 3600, 1) 

74 fired.append( 

75 ( 

76 EVENT_TASK_DUE_SOON, 

77 events.task_event_data( 

78 task, extra={"due_in_hours": due_in_hours} 

79 ), 

80 ) 

81 ) 

82 due_soon_fired = True 

83 if not overdue_fired and recurrence.is_overdue(task, now=now): 

84 days_overdue = (now - parsed).days 

85 fired.append( 

86 ( 

87 EVENT_TASK_OVERDUE, 

88 events.task_event_data( 

89 task, extra={"days_overdue": days_overdue} 

90 ), 

91 ) 

92 ) 

93 overdue_fired = True 

94 

95 next_state[tid] = { 

96 "next_due": next_due, 

97 "due_soon_fired": due_soon_fired, 

98 "overdue_fired": overdue_fired, 

99 } 

100 

101 return fired, next_state 

102 

103 

104#: The ``auto`` trigger kind of a notification that each fired event matches. 

105AUTO_KINDS = {EVENT_TASK_OVERDUE: "overdue", EVENT_TASK_DUE_SOON: "due_soon"} 

106 

107 

108def auto_crossings( 

109 prev: StateMap, fired: list[tuple[str, dict[str, Any]]] 

110) -> list[tuple[str, str]]: 

111 """The ``(kind, task_id)`` pairs that can send an automatic notification. 

112 

113 *prev* is the edge state before the pass that returned *fired*. Every fired event 

114 still goes on the bus. But a ``due_soon`` event for a task that had already 

115 announced its old ``next_due`` does not send a notification (B16-6). A user who 

116 snoozes or completes a due task moves ``next_due``, and the new date is often 

117 inside the window. A push at that moment answers the action that the user just 

118 did. The ``overdue`` event of the new date still sends. 

119 """ 

120 crossed: list[tuple[str, str]] = [] 

121 for name, payload in fired: 

122 kind = AUTO_KINDS.get(name) 

123 task_id = payload.get("task_id") 

124 if kind is None or not task_id: 

125 continue 

126 # A due_soon event fires only for a next_due that has not announced it, so a 

127 # prior flag that is set belongs to the old next_due. 

128 prior = prev.get(str(task_id)) or {} 

129 announced = prior.get("due_soon_fired") or prior.get("overdue_fired") 

130 if kind == "due_soon" and announced: 

131 continue 

132 crossed.append((kind, str(task_id))) 

133 return crossed
