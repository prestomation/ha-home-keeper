1"""Pure logic for the wear-part → maintenance-task reconciliation. 

2 

3A wear part with a ``replace_interval`` yields a floating maintenance task attached 

4to the asset's device (so it reuses the existing to-do/calendar/per-task entities). 

5The task is keyed by ``source = {"part": {asset_id, part_id}}`` so the reconciler 

6exclusively owns it. 

7 

8This module imports nothing from Home Assistant: it is a pure transformation over 

9plain dicts (assets, tasks) that :class:`store.HomeKeeperStore` wraps with 

10persistence. Keeping it pure lets the create/anchor/heal/orphan branches be 

11unit-tested directly (see ``tests/unit/test_reconcile.py``). 

12"""

57def part_source(task: dict[str, Any]) -> dict[str, Any] | None: 

58 """Return a task's ``{asset_id, part_id}`` part provenance, or None. 

59 

60 Covers both reconciler-derived tasks (auto-generated from a wear part's 

61 ``replace_interval``) and tasks a user *manually linked* to a consumable — both 

62 carry ``source = {"part": {"asset_id", "part_id", ...}}`` so completion consumes 

63 one spare from the part's stock. A manual link additionally carries 

64 ``"manual": True`` inside the part dict; use :func:`is_manual_part_link` to tell 

65 them apart where ownership matters (the reconciler must not own manual links). 

66 """ 

67 source = task.get("source") 

68 if isinstance(source, dict) and isinstance(source.get(TASK_SOURCE_PART), dict): 

69 part = source[TASK_SOURCE_PART] 

70 # Require the identifying keys before treating this as a part source. A 

71 # malformed reserved shape (e.g. a caller passing ``source={"part": {...}}`` 

72 # without asset_id/part_id) must not reach the bracket accesses in the 

73 # reconciler — otherwise it raises KeyError inside async_setup_entry and the 

74 # entry fails to set up on every restart until storage is hand-edited. 

75 if "asset_id" in part and "part_id" in part: 

76 return part 

77 return None

80def is_manual_part_link(task: dict[str, Any]) -> bool: 

81 """True when a task's part link was set by the user, not the reconciler. 

82 

83 The reconciler exclusively owns the wear-part tasks it generates (it creates, 

84 updates, and *deletes* them as parts change). A manual link reuses the same 

85 ``source.part`` shape so it consumes stock on completion, but must be invisible 

86 to the reconciler — otherwise the next reconcile pass would delete it as an 

87 "orphan" (its part has no ``replace_interval``). The ``manual`` flag is the 

88 discriminator that keeps the two apart with no storage migration: existing 

89 derived tasks lack it and stay reconciler-owned. 

90 """ 

91 src = part_source(task) 

92 return bool(src and src.get("manual"))

111def part_role(task: dict[str, Any]) -> str: 

112 """Which half of a wear part a derived task is: ``use`` or ``replace``. 

113 

114 An **absent** role reads as ``replace``. That is the whole migration story for 

115 counted wear items: every derived task written before they existed lacks the key 

116 and keeps its identity, so the reconciler recognizes it, updates it, and never 

117 orphans it as a stranger. 

118 """ 

119 src = part_source(task) 

120 if src is None: 

121 return PART_ROLE_REPLACE 

122 role = src.get("role") 

123 return PART_ROLE_USE if role == PART_ROLE_USE else PART_ROLE_REPLACE

169def cycle_start(replace_task: dict[str, Any]) -> str | None: 

170 """When the current count cycle started, or ``None`` before it has ever started. 

171 

172 The later of 2 instants: the replacement task's ``last_completed``, and its most 

173 recent skip. A completion is the obvious one — the part was replaced, so the next 

174 count starts from zero. 

175 

176 A skip has to count for the same reason. ``recurrence.skip_occurrence`` leaves 

177 ``last_completed`` alone by contract and sets ``next_due`` to None, which is the 

178 exact state :func:`settle_use_tasks` arms from — so without the skip here the 

179 task re-arms on the next coordinator tick and fires a fresh 

180 ``home_keeper_task_triggered``. Skip on a counted wear item was a no-op that 

181 bounced straight back. It is the same failure ``store.skip_task`` resets a usage 

182 meter for (#268), and the same answer: a skip means "not this cycle", so it 

183 starts the next one. 

184 

185 An entry that cannot be parsed is ignored rather than treated as the epoch, so 

186 one bad row in a user-edited history does not restart the count. 

187 """ 

188 best: str | None = None 

189 best_key = (0, 0.0) 

190 candidates = [replace_task.get("last_completed")] 

191 candidates += [entry.get("ts") for entry in replace_task.get("skips") or []] 

192 for value in candidates: 

193 if not value: 

194 continue 

195 key = _sortable(value) 

196 if key[0] and key > best_key: 

197 best, best_key = str(value), key 

198 return best

201def uses_since_replacement( 

202 use_task: dict[str, Any], replace_task: dict[str, Any] 

203) -> int: 

204 """How many uses have been recorded since the current cycle started. 

205 

206 The count *is* the use task's completion log, filtered to the entries later than 

207 :func:`cycle_start`. There is no stored counter, and so nothing to keep in sync: 

208 completing or skipping the replacement task moves the marker forward and the next 

209 count starts from zero by construction. 

210 

211 A replacement task that has never been completed or skipped counts every entry, 

212 which is right for a part in its first cycle. 

213 """ 

214 completions = use_task.get("completions") or [] 

215 since = cycle_start(replace_task) 

216 if since is None: 

217 return len(completions) 

218 marker = _sortable(since) 

219 return sum(1 for entry in completions if _sortable(entry.get("ts")) > marker)

222def replacement_backstop_due( 

223 part: dict[str, Any], 

224 replace_task: dict[str, Any], 

225 *, 

226 tz: tzinfo | None, 

227) -> datetime | None: 

228 """When a counted wear item's time backstop comes due, or ``None`` if it has none. 

229 

230 "Every 25 wears, **or** every 12 months, whichever comes first." The backstop 

231 measures from the same instant the count does — :func:`cycle_start` — so one 

232 completion or skip resets both halves and they can never disagree about which 

233 cycle they are in. While the cycle has never started it falls back to the part's 

234 recorded ``last_replaced`` and then to the task's ``created``, which is the ladder 

235 ``sensor_tasks.backstop_due`` already walks for a usage task. 

236 """ 

237 backstop = part_replace_backstop(part) 

238 if backstop is None: 

239 return None 

240 raw = ( 

241 cycle_start(replace_task) 

242 or qualify_iso(part.get("last_replaced"), tz) 

243 or replace_task.get("created") 

244 ) 

245 anchor = _parse_ts(raw) 

246 if anchor is None: 246 ↛ 247line 246 didn't jump to line 247 because the condition on line 246 was never true

247 return None 

248 if anchor.tzinfo is None: 248 ↛ 249line 248 didn't jump to line 249 because the condition on line 248 was never true

249 anchor = anchor.replace(tzinfo=tz) if tz is not None else anchor.astimezone() 

250 return recurrence.add_interval( 

251 anchor, int(backstop["interval"]), str(backstop["unit"])

255def replacement_is_due( 

256 part: dict[str, Any], 

257 use_task: dict[str, Any], 

258 replace_task: dict[str, Any], 

259 *, 

260 now: datetime, 

261) -> bool: 

262 """True when a counted wear item has earned its replacement task. 

263 

264 Whichever comes first: the count reaching the target, or the optional time 

265 backstop elapsing. There is deliberately no "both must be met" combinator — a 

266 usage task offers one because a service interval has a real "no earlier than" 

267 floor, and a wear item does not. 

268 """ 

269 target = part_use_target(part) 

270 if target and uses_since_replacement(use_task, replace_task) >= target: 

271 return True 

272 due = replacement_backstop_due(part, replace_task, tz=now.tzinfo) 

273 return due is not None and due <= now

276def trim_use_completions( 

277 use_task: dict[str, Any], replace_task: dict[str, Any], *, cap: int 

278) -> bool: 

279 """Trim *use_task*'s completion log to *cap*, protecting the live count. 

280 

281 Returns ``True`` when entries were dropped. The window the reminder depends on is 

282 "uses since the last replacement", so an entry inside it is **never** trimmed even 

283 when it puts the log over the cap: losing one silently lowers the count and the 

284 replacement task then never comes due, with nothing on any surface to say why. 

285 Every other entry is history, kept newest-first for the cadence report. 

286 """ 

287 completions = list(use_task.get("completions") or []) 

288 protected = uses_since_replacement(use_task, replace_task) 

289 keep = max(cap, protected) 

290 if len(completions) <= keep: 

291 return False 

292 completions.sort(key=lambda entry: _sortable(entry.get("ts"))) 

293 use_task["completions"] = completions[-keep:] 

294 return True

301

313 for asset in assets.values(): 

314 for part in asset.get("parts", []):

316 continue 

317 asset_id = asset.get("id") 

318 part_id = part.get("id")

331def settle_use_tasks( 

332 assets: dict[str, dict[str, Any]], 

333 tasks: dict[str, dict[str, Any]], 

334 *, 

335 now: datetime, 

336) -> tuple[list[str], bool]: 

337 """Decide which replacement tasks to arm, and trim the use logs. 

338 

339 Returns ``(task_ids_to_arm, trimmed)``. Pure: it arms nothing itself, because 

340 arming fires ``home_keeper_task_triggered`` and that belongs at the store's 

341 chokepoint. The trim *is* applied here, in place, since it is ordinary data 

342 hygiene with no event of its own — the caller persists when ``trimmed`` is true. 

343 

344 A replacement task that is already armed is left alone, so a household that lets 

345 the count run past the target does not get a fresh event on every tick. A 

346 **disabled** one is left alone too, matching ``sensor_watcher``, its twin for 

347 usage tasks: arming it would fire ``home_keeper_task_triggered`` at a device 

348 trigger nobody asked for, and leave it overdue the moment it is re-enabled. 

349 """ 

350 to_arm: list[str] = [] 

351 trimmed = False 

352 for _asset, part, use_task, replace_task in counted_part_pairs(assets, tasks): 

353 if ( 

354 replace_task.get("next_due") is None 

355 and replace_task.get("enabled", True) 

356 and replacement_is_due(part, use_task, replace_task, now=now) 

357 ): 

358 to_arm.append(replace_task["id"]) 

359 if trim_use_completions(use_task, replace_task, cap=use_retention_cap(part)): 

360 trimmed = True 

361 return to_arm, trimmed

398def reconcile_part_tasks( 

399 assets: dict[str, dict[str, Any]], 

400 tasks: dict[str, dict[str, Any]], 

401 *, 

402 name_template: str = _DEFAULT_NAME_TEMPLATE, 

403 appliance_fallback: str = _DEFAULT_APPLIANCE_FALLBACK, 

404 use_name_template: str = _DEFAULT_USE_NAME_TEMPLATE, 

405 language: str | None = None, 

406 now: datetime, 

407) -> tuple[dict[str, dict[str, Any]], bool]: 

408 """Compute the task map derived from wear parts. 

409 

410 Returns ``(new_tasks, changed)``. ``new_tasks`` is a fresh dict (the input is 

411 not mutated); ``changed`` is ``True`` when it differs from ``tasks``. Tasks 

412 without a part source are carried through untouched. 

413 

414 ``name_template`` (with ``{part}``/``{asset}`` placeholders) and 

415 ``appliance_fallback`` (substituted for an unnamed appliance) are the localized 

416 strings for the generated task name; the caller resolves them from 

417 ``hass.config.language`` (see ``const.resolve_wear_task_naming``). Defaulting to 

418 English keeps this function pure and independently testable. Because the name is 

419 recomputed here on every pass, a language change is picked up as ordinary name 

420 drift — the existing ``before.get("name") != name`` branch rewrites it. 

421 

422 ``name_template`` covers the ``replace`` action only, which is what every wear part 

423 used before actions existed. The other 6 actions resolve from *language* through 

424 ``const.resolve_action_task_naming``, and ``use_name_template`` names the use task 

425 a counted wear item generates. Both stay parameters rather than a language lookup 

426 inside, so this function keeps its "pass me the strings" contract. 

427 

428 **A counted wear item owns 2 tasks**, so the index below is keyed by 

429 ``(asset_id, part_id, role)`` rather than by the part alone. Keyed by the part, the 

430 2 tasks collide, the second overwrites the first, and the orphan sweep deletes 

431 whichever one lost. 

432 """ 

433 result = dict(tasks) 

434 

435 desired: dict[tuple[str, str, str], tuple[dict, dict]] = {} 

436 for asset in assets.values(): 

437 for part in asset.get("parts", []): 

438 if part.get("type") == PART_WEAR and part.get("replace_interval"): 

439 desired[(asset["id"], part["id"], PART_ROLE_REPLACE)] = (asset, part) 

440 if part_counts_uses(part): 

441 desired[(asset["id"], part["id"], PART_ROLE_USE)] = (asset, part) 

442 

443 existing_by_key: dict[tuple[str, str, str], str] = {} 

444 for tid, task in result.items(): 

445 src = part_source(task) 

446 # Skip manually-linked tasks: they reuse the part-source shape (to consume 

447 # stock on completion) but are user-owned, so the reconciler must never 

448 # update or orphan-delete them. 

449 if src and not src.get("manual"): 

450 existing_by_key[(src["asset_id"], src["part_id"], part_role(task))] = tid 

451 

452 changed = False 

453 

454 # Remove orphaned part-tasks (the part or its wear cadence went away). 

455 for key, tid in list(existing_by_key.items()): 

456 if key not in desired: 

457 del result[tid] 

458 existing_by_key.pop(key, None) 

459 changed = True 

460 

461 # Create or update the rest. 

462 for key, (asset, part) in desired.items(): 

463 role = key[2] 

464 asset_name = asset.get("name") or appliance_fallback 

465 counted = part_counts_uses(part) 

466 if role == PART_ROLE_USE: 

467 # A part may name its use task outright ("Wear rain jacket"); otherwise it 

468 # gets the localized "Use {asset}". The part is not named either way — the 

469 # household taps this to record using the *thing*, not the component that 

470 # wears out. 

471 name = part.get("use_task_name") or use_name_template.format( 

472 asset=asset_name 

473 ) 

474 else: 

475 name = _replace_name_template(part, name_template, language).format( 

476 part=part["name"], asset=asset_name 

477 ) 

478 # The replacement half of a counted wear item is ``triggered``: dormant until 

479 # the settle step arms it, which is the existing shape for "something else 

480 # decides when this is due". A time-measured part keeps its floating cadence. 

481 rec_type = ( 

482 REC_USE 

483 if role == PART_ROLE_USE 

484 else (REC_TRIGGERED if counted else REC_FLOATING) 

485 ) 

486 # A part's last_replaced is a date-only string; qualify it to HA's tz so the 

487 # derived next_due is timezone-aware. A naive next_due otherwise crashes the 

488 # next-due sensor, overdue binary_sensor, and the calendar (which compare it 

489 # against an aware "now"). 

490 anchored = qualify_iso(part.get("last_replaced"), now.tzinfo) 

491 existing_tid = existing_by_key.get(key) 

492 if existing_tid is None: 

493 # ``role`` is written only for a use task. An absent role already reads 

494 # as ``replace`` (see part_role), so stamping it on the replacement half 

495 # would put a second shape in storage for a value that never changes 

496 # meaning — and would rewrite the source of every wear part in the wild 

497 # for no gain. 

498 part_link: dict[str, Any] = { 

499 "asset_id": asset["id"], 

500 "part_id": part["id"], 

501 } 

502 if role == PART_ROLE_USE: 

503 part_link["role"] = PART_ROLE_USE 

504 payload: dict[str, Any] = { 

505 "name": name, 

506 "recurrence_type": rec_type, 

507 "device_id": asset.get("device_id"), 

508 "area_id": asset.get("area_id"), 

509 "source": {"part": part_link}, 

510 } 

511 if rec_type == REC_FLOATING: 

512 payload["interval"] = part["replace_interval"] 

513 payload["unit"] = part["replace_unit"] 

514 task = models.build_task(payload, now=now) 

515 if rec_type == REC_TRIGGERED: 

516 # ``build_task`` creates a triggered task **armed**: compute_next_due 

517 # returns ``now`` for it (the re-arm contract). A counted wear item 

518 # has earned nothing yet, so left alone every one of them would ship 

519 # overdue the moment the part is saved. Start it dormant, exactly as 

520 # build_task itself does for a sensor task. 

521 task["next_due"] = None 

522 elif rec_type == REC_FLOATING and anchored: 

523 # Anchor the floating clock to the recorded replacement. With no 

524 # recorded replacement we leave last_completed unset (build_task's 

525 # default), so the part reads as due now rather than "assumed fresh" a 

526 # full interval out: an unknown replacement history is better surfaced 

527 # now — the user can backdate the replacement or mark it done — than 

528 # silently hidden for a cycle. 

529 task["last_completed"] = anchored 

530 task["next_due"] = recurrence.compute_next_due( 

531 task, now=now 

532 ).isoformat() 

533 result[task["id"]] = task 

534 changed = True 

535 else: 

536 # Only pass fields that actually changed. Passing interval/unit 

537 # unconditionally would re-trigger a next_due recompute on every

613def reconcile_buy_tasks( 

614 assets: dict[str, dict[str, Any]], 

615 tasks: dict[str, dict[str, Any]], 

616 *, 

617 name_template: str = _DEFAULT_BUY_NAME_TEMPLATE, 

618 now: datetime, 

619) -> tuple[dict[str, dict[str, Any]], bool]: 

620 """Compute the task map with auto-created "buy" tasks synced to low parts. 

621 

622 Returns ``(new_tasks, changed)``. A buy task is **desired** for every ``(asset, 

623 part)`` where the part opts in (``part_wants_buy_task``) *and* is currently low 

624 (``part_is_low``). The task is a one-off ``"Buy {part}"`` reminder attached to the 

625 asset's device, keyed by ``source = {"buy": {asset_id, part_id}}`` so the 

626 reconciler exclusively owns it. 

627 

628 Idempotency is **per low episode**: the existing index counts a buy task for a 

629 part whether it is open *or already completed*, so completing the reminder does 

630 not respawn it while the part stays low. The buy task is orphan-removed the moment 

631 its key leaves the desired set — the part restocked above the threshold, opted 

632 out, or was deleted — which also ends the episode and re-arms the next one. 

633 

634 Pure: the input maps are not mutated, and the name is localized by the caller 

635 (``store.reconcile_buy_tasks`` resolves ``hass.config.language``). Buy tasks are 

636 never *updated* here (a one-off has no cadence to re-derive); only created and 

637 removed, so a rename after a language change is picked up as a delete + recreate 

638 on the next reconcile — acceptable for a transient reminder. 

639 """ 

640 result = dict(tasks) 

641 

642 desired: dict[tuple[str, str], tuple[dict, dict]] = {} 

643 for asset in assets.values(): 

644 for part in asset.get("parts", []): 

645 if part_wants_buy_task(part) and part_is_low(part): 

646 desired[(asset["id"], part["id"])] = (asset, part) 

647 

648 # Index existing buy tasks by their part key — open OR completed — so a completed 

649 # reminder still "occupies" the episode and blocks a duplicate. 

650 existing_by_key: dict[tuple[str, str], str] = {} 

651 for tid, task in result.items(): 

652 src = buy_source(task) 

653 if src: 653 ↛ 651line 653 didn't jump to line 651 because the condition on line 653 was always true

654 existing_by_key[(src["asset_id"], src["part_id"])] = tid 

655 

656 changed = False 

657 

658 # Remove orphaned buy tasks (part restocked / opted out / part or asset gone). 

659 for key, tid in list(existing_by_key.items()): 

660 if key not in desired: 

661 del result[tid] 

662 existing_by_key.pop(key, None) 

663 changed = True 

664 

665 # Create a buy task for each desired key that doesn't already have one. 

666 for key, (asset, part) in desired.items(): 

667 if key in existing_by_key:

674 "area_id": asset.get("area_id"), 

675 "source": {"buy": {"asset_id": asset["id"], "part_id": part["id"]}}, 

676 }, 

677 now=now, 

678 ) 

679 result[task["id"]] = task 

680 changed = True 

681
