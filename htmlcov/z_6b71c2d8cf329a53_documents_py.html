1"""Pure (HA-free) helpers for uploaded asset documents. 

2 

3The on-disk storage and HTTP view live in ``manuals.py`` (which needs Home 

4Assistant); the security-critical, side-effect-free pieces — magic-byte sniffing, 

5filename sanitization, the upload allowlist/size guard, and the path-traversal guard — 

6live here so they stay unit-testable without an HA runtime (see 

7``tests/unit/test_documents.py``). This module imports nothing from Home Assistant. 

8"""

23# content-type -> canonical extension. The key set is the upload allowlist. 

24TYPE_EXTENSIONS = { 

25 "application/pdf": ".pdf", 

26 "image/png": ".png", 

27 "image/jpeg": ".jpg", 

28 "image/webp": ".webp", 

29 "image/gif": ".gif", 

30} 

31 

32_MAX_FILENAME_LEN = 120 

33# The same limit as a document name in ``assets``. 

34_MAX_DISPLAY_NAME_LEN = 200 

35 

36# How many leading bytes ``sniff_content_type`` needs. The longest signature check 

37# reads ``data[8:12]`` (WebP), so 16 is comfortably enough — uploads are streamed to 

38# disk, and this is all that's kept in memory to identify the type. 

39SNIFF_BYTES = 16

42def sniff_content_type(data: bytes) -> str | None: 

43 """Return the allowlisted content type *data* matches by magic bytes, or None.""" 

44 if data.startswith(b"%PDF-"): 

45 return "application/pdf" 

46 if data.startswith(b"\x89PNG\r\n\x1a\n"): 

47 return "image/png" 

48 if data.startswith(b"\xff\xd8\xff"): 

49 return "image/jpeg" 

50 if data[:6] in (b"GIF87a", b"GIF89a"): 

51 return "image/gif" 

52 if data[:4] == b"RIFF" and data[8:12] == b"WEBP": 

53 return "image/webp" 

54 return None

57def safe_filename(name: str, content_type: str) -> str: 

58 """Sanitize an uploaded filename to a plain, extension-correct basename. 

59 

60 Strips any path component, collapses unusual characters, guarantees a non-empty 

61 stem, and forces the canonical extension for *content_type* so the served file is 

62 typed consistently regardless of what the client claimed. 

63 """ 

64 base = PurePath(str(name)).name 

65 base = re.sub(r"[^A-Za-z0-9._-]", "_", base).lstrip(".") 

66 stem = PurePath(base).stem or "document" 

67 stem = stem[:_MAX_FILENAME_LEN] 

68 return f"{stem}{TYPE_EXTENSIONS[content_type]}"

71def display_filename(name: str) -> str: 

72 """Return the name of an uploaded file as a user reads it (B06-6). 

73 

74 :func:`safe_filename` keeps only ASCII for the key on disk, so a name such as 

75 ``Инструкция.pdf`` becomes ``__________.pdf``. This keeps the real name for the 

76 display: the path is removed, the text is NFC-normalized, control characters 

77 are removed, and the length is capped. An empty result gives ``""``. 

78 """ 

79 base = PurePath(str(name).replace("\\", "/")).name 

80 text = unicodedata.normalize("NFC", base) 

81 text = "".join(ch for ch in text if unicodedata.category(ch)[0] != "C") 

82 return text.strip()[:_MAX_DISPLAY_NAME_LEN].strip()

85def content_disposition(filename: str, display_name: str = "") -> str: 

86 """Return the ``Content-Disposition`` value to serve a stored file with. 

87 

88 *filename* is the ASCII key on disk. When *display_name* is set, an RFC 6266 

89 ``filename*`` gives the browser the real name in UTF-8 (B06-6). The display 

90 name gets the extension of *filename* when it does not have it, so the saved 

91 file still opens. 

92 """ 

93 value = f'inline; filename="{filename}"' 

94 display = display_filename(display_name) 

95 if not display: 

96 return value 

97 suffix = PurePath(filename).suffix 

98 if suffix and not display.lower().endswith(suffix.lower()): 

99 display = f"{display}{suffix}" 

100 # Equivalent mutants: the default ``safe`` is "/", and display_filename keeps 

101 # only the last path part, so the name never holds a "/". 

102 return f"{value}; filename*=UTF-8''{quote(display, safe='')}" # pragma: no mutate

105def validate_upload_stream(filename: str, header: bytes, size: int) -> tuple[str, str]: 

106 """Validate a streamed upload from its first bytes and total size. 

107 

108 The counterpart to :func:`validate_upload` for uploads that are written straight 

109 to disk and never held in memory (see ``manuals._parse_upload``): everything the 

110 checks need is the leading ``header`` (at least :data:`SNIFF_BYTES`) and the byte 

111 count. Returns ``(content_type, safe_filename)``. 

112 

113 Raises :class:`AssetValidationError` when the file is empty, over the size ceiling, 

114 or not a recognized allowlisted type (sniffed by magic bytes — any client-declared 

115 MIME is ignored entirely). The returned content type is the sniffed one, so the 

116 stored metadata can't be spoofed by a misleading client header. 

117 """ 

118 if size <= 0: 

119 raise AssetValidationError("uploaded file is empty") 

120 if size > MAX_DOCUMENT_BYTES: 

121 raise AssetValidationError( 

122 f"file exceeds the {MAX_DOCUMENT_BYTES // (1024 * 1024)} MB limit" 

123 ) 

124 content_type = sniff_content_type(header) 

125 if content_type is None: 

126 raise AssetValidationError( 

127 "unsupported file type (allowed: PDF, PNG, JPEG, WebP, GIF)" 

128 ) 

129 return content_type, safe_filename(filename, content_type)

132def validate_upload(filename: str, data: bytes) -> tuple[str, str]: 

133 """Validate an in-memory uploaded blob; return ``(content_type, safe_filename)``. 

134 

135 Thin wrapper over :func:`validate_upload_stream` for callers that already hold the 

136 whole file. The HTTP upload path does *not* — it streams to disk — so prefer the 

137 stream variant for anything that could be large. 

138 """ 

139 return validate_upload_stream(filename, data[:SNIFF_BYTES], len(data))

142def upload_document_id(requested: str, taken: Iterable[str]) -> str: 

143 """Return the id to store a new uploaded document under (B06-2). 

144 

145 The client sends the id, and the file goes on disk under it before the 

146 metadata is saved. So an id that another document already has, or that is not 

147 a uuid in hex digits and hyphens only, gets a new uuid here, before any file is 

148 written. An id in another shape can put 2 records on one path (``part_<id>`` is 

149 a part's file key, and ``__`` splits the id from the file name). 

150 """ 

151 if _is_plain_uuid(requested) and requested not in set(taken): 

152 return requested 

153 return str(uuid.uuid4())

156def _is_plain_uuid(value: str) -> bool: 

157 """Whether *value* is a uuid written with hex digits and hyphens only.""" 

158 if not isinstance(value, str) or not _HEX_ID.fullmatch(value): 

159 return False 

160 try: 

161 uuid.UUID(value) 

162 except ValueError: 

163 return False 

164 return True

167_HEX_ID = re.compile(r"[0-9a-fA-F-]+")

170def purge_stale_temps(tmp_root: Path, max_age_s: float) -> None: 

171 """Delete temp uploads old enough that no request can still be writing them. 

172 

173 Deliberately age-based rather than a blanket wipe of *tmp_root*: the caller runs 

174 this at setup, which re-runs on every config entry *reload* while the upload view 

175 stays registered — clearing the directory outright could delete the temp file an 

176 in-flight upload is still writing, turning it into a 500 mid-transfer. 

177 """ 

178 if not tmp_root.is_dir(): 

179 return 

180 cutoff = time.time() - max_age_s 

181 for path in tmp_root.glob("upload-*"): 

182 try: 

183 if path.stat().st_mtime < cutoff: 

184 path.unlink(missing_ok=True) 

185 except OSError: 

186 # Raced with the upload that owns it, or it vanished — try again next time. 

187 continue

190def safe_segment(value: str) -> str: 

191 """Reduce an id/filename to a single safe path segment (defense in depth).""" 

192 seg = re.sub(r"[^A-Za-z0-9._-]", "_", PurePath(str(value)).name).lstrip(".") 

193 if not seg or seg in (".", ".."): 

194 raise AssetValidationError("invalid document path") 

195 return seg

198def resolve_under_root(root: Path, *parts: str) -> Path: 

199 """Join *parts* under *root* and assert the result stays inside it. 

200 

201 Each part is first reduced to a safe single segment, then the fully-resolved path 

202 is checked to be relative to the resolved root — a belt-and-braces guard against a 

203 crafted asset/document id or filename escaping the storage tree (``..``, absolute 

204 paths, separators). 

205 """ 

206 root = root.resolve() 

207 candidate = root.joinpath(*(safe_segment(p) for p in parts)).resolve() 

208 if not candidate.is_relative_to(root): 208 ↛ 209line 208 didn't jump to line 209 because the condition on line 208 was never true

209 raise AssetValidationError("document path escapes the storage root") 

210 return candidate

214 """The on-disk path for an uploaded document, guarded against traversal."""
