1"""Recurrence engine for Home Keeper. 

2 

3These are deliberately *pure* functions: they take and return timezone-aware 

4``datetime`` objects and plain task dicts, and import nothing from Home Assistant. 

5That keeps the product's core logic trivially unit-testable in isolation (the 

6caller is responsible for passing an aware ``now`` from ``homeassistant.util.dt``). 

7 

8The task has 6 recurrence types (``const.RECURRENCE_TYPES``). The 2 clock-based ones 

9are below; ``docs/design/recurrence.md`` covers all 6. 

10 

11* **floating** — the next due date is measured from the last completion: 

12 ``next_due = last_completed + interval·unit``. A task that has *never* been 

13 completed has no clock to measure from, so it is due immediately (``now``) rather 

14 than a full interval into the future — a brand-new chore you haven't done yet is 

15 due now, not "in N days". Completing the task (or seeding an initial completion at 

16 creation) starts the clock. A missed task simply stays overdue; the due date does 

17 not march forward on its own. 

18 

19* **fixed** — the next due date follows a calendar schedule anchored at a fixed 

20 datetime (``FREQ=DAILY|WEEKLY|MONTHLY`` every ``interval`` steps). Completing an 

21 occurrence records history but the schedule advances independently of when the 

22 task was actually completed. 

23"""

65def add_months(dt: datetime, months: int) -> datetime: 

66 """Return *dt* shifted by *months*, clamping the day to the month's length. 

67 

68 e.g. ``add_months(Jan 31, 1) -> Feb 28`` (or Feb 29 in a leap year). The 

69 time-of-day and tzinfo are preserved. 

70 """ 

71 if months == 0: 

72 return dt 

73 # Convert to a zero-based month index for clean arithmetic. Python's floor 

74 # division and non-negative modulo make this correct for negative months too 

75 # (e.g. Jan + (-2) months -> month_index -2 -> year-1, month 11 = November), 

76 # though in practice this integration only ever advances by positive months. 

77 month_index = dt.month - 1 + months 

78 year = dt.year + month_index // 12 

79 month = month_index % 12 + 1 

80 last_day = _calendar.monthrange(year, month)[1] 

81 day = min(dt.day, last_day) 

82 return dt.replace(year=year, month=month, day=day)

85def add_interval(dt: datetime, interval: int, unit: str) -> datetime: 

86 """Return *dt* advanced by ``interval`` of ``unit`` (days/weeks/months).""" 

87 if interval < 1: 

88 raise ValueError(f"interval must be >= 1, got {interval}") 

89 if unit == UNIT_DAYS: 

90 return dt + timedelta(days=interval) 

91 if unit == UNIT_WEEKS: 

92 return dt + timedelta(weeks=interval) 

93 if unit == UNIT_MONTHS: 

94 return add_months(dt, interval) 

95 raise ValueError(f"unknown unit: {unit!r}")

111def in_season(dt_val: datetime, season: dict | list) -> bool: 

112 """True when *dt_val*'s month-day falls inside any active-season window. 

113 

114 Non-wrapping (e.g. Apr-Sep): ``start <= md <= end``. 

115 Wrapping (e.g. Nov-Mar): ``md >= start or md <= end``. 

116 """ 

117 md = (dt_val.month, dt_val.day) 

118 for window in _normalize_season(season): 

119 s_m, s_d = _parse_mmdd(window["start"]) 

120 e_m, e_d = _parse_mmdd(window["end"]) 

121 start = (s_m, s_d) 

122 end = (e_m, e_d) 

123 if start <= end: 

124 if start <= md <= end: 

125 return True 

126 elif md >= start or md <= end: 

127 return True 

128 return False

131def _next_season_start(dt_val: datetime, season: dict | list) -> datetime: 

132 """Earliest datetime at any window's start on or after *dt_val*. 

133 

134 Clamps the day via ``calendar.monthrange`` so a season start like ``"01-31"`` 

135 landing in February becomes Feb 28/29. 

136 """ 

137 candidates: list[datetime] = [] 

138 for window in _normalize_season(season): 

139 s_m, s_d = _parse_mmdd(window["start"]) 

140 year = dt_val.year 

141 last_day = _calendar.monthrange(year, s_m)[1] 

142 day = min(s_d, last_day) 

143 candidate = dt_val.replace( 

144 month=s_m, day=day, hour=0, minute=0, second=0, microsecond=0 

145 ) 

146 if candidate < dt_val: 

147 year += 1 

148 last_day = _calendar.monthrange(year, s_m)[1] 

149 day = min(s_d, last_day) 

150 candidate = dt_val.replace( 

151 year=year, 

152 month=s_m, 

153 day=day, 

154 hour=0, 

155 minute=0, 

156 second=0, 

157 microsecond=0, 

158 ) 

159 candidates.append(candidate) 

160 return min(candidates)

163def _clamp_season(next_due: datetime, task: dict) -> datetime: 

164 """Push *next_due* to the next season start if it falls outside the season. 

165 

166 For fixed tasks the result stays grid-aligned: the first fixed occurrence 

167 on or after the season start is returned instead of the bare start date. 

168 """ 

169 season = task.get("active_season") 

170 if not season: 

171 return next_due 

172 if in_season(next_due, season): 

173 return next_due 

174 season_start = _next_season_start(next_due, season) 

175 rec_type = task.get("recurrence_type", REC_FLOATING) 

176 if rec_type == REC_FIXED: 

177 anchor = _parse(task["anchor"]) 

178 assert anchor is not None 

179 after = season_start - timedelta(seconds=1) 

180 freq, interval = task["freq"], int(task["interval"]) 

181 # The first grid occurrence after the season start can fall after the season 

182 # end too, for a sparse grid and a short season. Walk on to the first one in 

183 # season, the same walk as the calendar (B07-9). A grid that never meets the 

184 # season keeps the first occurrence after the season start. 

185 found = next_in_season_occurrence(anchor, freq, interval, season, after=after) 

186 if found is not None: 

187 return found 

188 return next_fixed_occurrence(anchor, freq, interval, after=after) 

189 return season_start

192def next_in_season_occurrence( 

193 anchor: datetime, 

194 freq: str, 

195 interval: int, 

196 season: dict | list, 

197 *, 

198 after: datetime, 

199) -> datetime | None: 

200 """First fixed occurrence after *after* that is inside *season*, or None. 

201 

202 Bounded by ``MAX_EXPAND_ITERATIONS`` grid steps. ``None`` means that the grid 

203 does not meet the season inside the bound, for example every 12 months from 

204 January with a March season. ``_clamp_season`` and the calendar both use this 

205 walk, so ``next_due`` and the calendar show the same date. 

206 """ 

207 occ = next_fixed_occurrence(anchor, freq, interval, after=after) 

208 for _ in range(MAX_EXPAND_ITERATIONS): 

209 if in_season(occ, season): 

210 return occ 

211 # Step from the last occurrence (B11-8). A new search from the anchor on 

212 # each step costs a full grid walk for a monthly anchor on day 29-31. 

213 occ = step_fixed(occ, freq, interval) 

214 return None

258def _fast_forward( 

259 anchor: datetime, freq: str, interval: int, target: datetime 

260) -> datetime: 

261 """An occurrence at or just before *target*, jumped to in O(1) where possible. 

262 

263 A long-dormant fixed schedule can be thousands of steps past its anchor; 

264 stepping one occurrence at a time would be slow (and historically raised once 

265 it blew an iteration cap). We deliberately *under*-shoot (``- 1`` step) so the 

266 caller's short loop finishes on the exact occurrence using wall-clock stepping 

267 — correct across DST and month-length clamping. Day/week deltas are exact, so 

268 the bulk jump is too. 

269 

270 MONTHLY is trickier: progressive day-clamping makes the grid path-dependent, so 

271 we can't jump directly from the anchor for a day > 28 (``add_months(Jan 31, 2)`` 

272 is Mar 31, but stepping is Jan 31 -> Feb 28 -> Mar 28). We exploit the fact that 

273 the clamped day is monotonically non-increasing toward 28 — once it bottoms out 

274 at 28 (the global floor: every month has >= 28 days) it never changes again, so 

275 from that occurrence we *can* jump in O(1). We therefore step until the day hits 

276 28 (typically a handful of steps — interval-1 monthly reaches a non-leap 

277 February within a few years) and then bulk-jump the remainder. Schedules whose 

278 reachable months never include a short-enough month (e.g. an even interval that 

279 skips February) keep a day > 28 forever; for those we simply step to the target, 

280 bounded by the (finite) span between anchor and target — slower, but never the 

281 old iteration-cap crash. 

282 """ 

283 elapsed = (target - anchor).total_seconds() 

284 if elapsed <= 0: 

285 return anchor 

286 if freq == FREQ_DAILY: 

287 steps = max(0, int(elapsed // (86_400 * interval)) - 1) 

288 return anchor + timedelta(days=interval * steps) 

289 if freq == FREQ_WEEKLY: 

290 steps = max(0, int(elapsed // (604_800 * interval)) - 1) 

291 return anchor + timedelta(weeks=interval * steps) 

292 # MONTHLY 

293 occ = anchor 

294 while occ.day > 28: 

295 nxt = add_months(occ, interval) 

296 if nxt > target: 

297 # Reached the target before the day stabilized: hand the (exact) grid 

298 # occurrence to the caller's loop. Bounded by the anchor→target span. 

299 return occ 

300 occ = nxt 

301 # The day has bottomed out at 28 and is now stable for every further step, so 

302 # the remaining whole steps can be jumped at once (under-shooting by one). 

303 remaining = (target.year - occ.year) * 12 + (target.month - occ.month) 

304 jump = max(0, remaining // interval - 1) 

305 return add_months(occ, jump * interval)

308def _regrid(anchor: datetime, probe: datetime) -> datetime: 

309 """*anchor* re-homed into *probe*'s timezone, so stepping keeps local wall time. 

310 

311 A schedule promises a wall clock: "every day at 10:00" means 10:00 on the kitchen 

312 clock, on both sides of a daylight-saving transition. :func:`_step` delivers that 

313 — but only while the anchor carries a real zone, because ``dt + timedelta`` holds 

314 the *tzinfo it is given*. A fixed UTC offset is a tzinfo too, and holding one of 

315 those across a transition is what makes a 10:00 task read 09:00 every autumn. 

316 

317 Storage is what strips the zone. An ISO string carries an offset and not a zone 

318 identity, so ``ZoneInfo("America/Los_Angeles")`` is written as ``-07:00`` and 

319 reloads as ``timezone(timedelta(hours=-7))``. Every anchor therefore comes back 

320 from the store offset-only, no matter how it was written — which is why writing it 

321 differently cannot fix this and re-homing on the way in has to. 

322 

323 *probe* is the caller's clock (``after`` here, ``start`` in 

324 :func:`expand_fixed_occurrences`), which in this integration is always derived 

325 from ``dt_util.now()`` and so carries Home Assistant's configured zone. Re-homing 

326 keeps the instant and swaps the tzinfo, so an anchor stored as ``17:00+00:00``, 

327 ``10:00-07:00`` or naive-turned-``10:00-07:00`` all resolve to the same 10:00 

328 local and all hold it through the transition. 

329 """ 

330 return anchor.astimezone(probe.tzinfo)

366def expand_fixed_occurrences( 

367 anchor: datetime, 

368 freq: str, 

369 interval: int, 

370 start: datetime, 

371 end: datetime, 

372) -> list[datetime]: 

373 """All fixed occurrences within the half-open range ``[start, end)``. 

374 

375 Occurrences read at the anchor's time-of-day in *start*'s timezone, the same rule

388 while occ < start:

394 occurrences.append(occ)

396 iterations += 1

426def latest_completion(completions: Iterable[dict]) -> dict | None: 

427 """The completion with the latest ``ts``, compared as instants, else ``None``. 

428 

429 The text of two ``ts`` values with different UTC offsets does not sort in time

448def compute_next_due(task: dict, *, now: datetime) -> datetime: 

449 """Compute next_due for *task* from its current state (no mutation).""" 

450 rec_type = task.get("recurrence_type", REC_FLOATING) 

451 if rec_type == REC_FLOATING: 

452 return _clamp_season( 

453 compute_floating_next_due( 

454 _local(_parse(task.get("last_completed")), now), 

455 int(task["interval"]), 

456 task["unit"], 

457 now=now, 

458 ), 

459 task, 

460 ) 

461 if rec_type == REC_FIXED: 

462 anchor = _parse(task["anchor"]) 

463 assert anchor is not None 

464 return _clamp_season( 

465 next_fixed_occurrence( 

466 anchor, task["freq"], int(task["interval"]), after=now 

467 ), 

468 task, 

469 ) 

470 if rec_type in (REC_TRIGGERED, REC_SENSOR): 

471 # A condition/sensor-driven task has no schedule: computing a due date means 

472 # *arming* it (the condition is true), so it reads as due-now. Going 

473 # dormant is the asymmetric job of ``apply_completion`` (next_due -> None); 

474 # this function is only ever called to (re)arm. A sensor task is armed by the 

475 # watcher (``store.trigger_task``); ``build_task`` starts it dormant. 

476 return now 

477 if rec_type == REC_ONE_OFF: 

478 # A do-once task is due at its stored ``due`` date. Going dormant on 

479 # completion is ``apply_completion``'s job (next_due -> None); this function 

480 # only ever (re)arms it back to ``due`` (e.g. when a completion is undone). 

481 due = _parse(task["due"]) 

482 assert due is not None 

483 return due 

484 if rec_type == REC_USE: 484 ↛ 493line 484 didn't jump to line 493 because the condition on line 484 was always true

485 # Every other type answers "when is this due?" with a date. A use task answers 

486 # it with "never": nothing arms it, so there is no correct datetime to return 

487 # and returning ``now`` (the triggered/sensor answer) would strand a counting 

488 # task as permanently overdue on every time surface. Each of the 5 callers 

489 # below skips a use task before reaching here, so this is a regression guard 

490 # rather than a live path — loud on purpose, because the silent alternative 

491 # ships a wrong due date. 

492 raise ValueError("a use task is never due; nothing arms it") 

493 raise ValueError(f"unknown recurrence_type: {rec_type!r}")

496def _record_entry(history: Iterable[dict], entry: dict) -> list[dict]: 

497 """Return *history* with *entry* recorded, deduped on ``ts`` and capped. 

498 

499 Shared by the completion log and the skip log: both are ``ts``-keyed lists of 

500 dated entries with the same identity rule, so they get the same insert. A second 

501 entry at the exact same instant (a double-tapped notification action, a duplicated 

502 automation) would otherwise create an ambiguous twin that undo/edit can't tell 

503 apart, so it *replaces* the entry already at that ``ts`` rather than appending. 

504 """ 

505 entries = list(history) 

506 ts_iso = entry["ts"] 

507 dup = next((i for i, e in enumerate(entries) if e.get("ts") == ts_iso), None) 

508 if dup is not None: 

509 entries[dup] = entry 

510 else: 

511 entries.append(entry) 

512 # `>=` here would be equivalent, not a bug: at exactly the cap the slice returns 

513 # the same entries, so only the strict form is observable. 

514 if len(entries) > MAX_COMPLETION_HISTORY: # pragma: no mutate 

515 entries = entries[-MAX_COMPLETION_HISTORY:] 

516 return entries

520# a daylight-saving shift and less than the smallest grid step (1 day). 

521_OCCURRENCE_PROBE = timedelta(hours=3)

524def _is_occurrence(task: dict, anchor: datetime, moment: datetime) -> bool: 

525 """Whether *moment* is one of the schedule's own occurrences. 

526 

527 Asked for exactly one instant per call, so it is answered by asking the grid for 

528 the occurrence after the moment before *moment*: a date on the grid is its own 

529 answer. An off-grid date belongs to a snooze or a due-today rather than to the 

530 schedule, and :func:`_advance_fixed_schedule` must not move the schedule past it. 

531 

532 A season is not consulted. ``_clamp_season`` only ever returns a grid occurrence, 

533 so a clamped ``next_due`` passes here, and a raw grid date that the season would 

534 reject is still a date the schedule owns. 

535 """ 

536 # The probe starts some hours before *moment*, not 1 microsecond before. The 

537 # grid steps by wall clock, and an occurrence in the hour that a spring-forward 

538 # skips has the wall clock 02:30 on the grid but reads back from storage as 

539 # 03:30 EDT. A probe at 03:29 already passed it on the grid. Occurrences are at 

540 # least 1 day apart, so the probe window finds only the occurrence at *moment* 

541 # (B07-8). 

542 probe = moment - _OCCURRENCE_PROBE 

543 found = next_fixed_occurrence( 

544 anchor, task["freq"], int(task["interval"]), after=probe 

545 ) 

546 # Compare instants, not wall clocks. With one tzinfo on both sides, ``==`` 

547 # compares the wall clock, and 02:30 and 03:30 EDT are one instant here. 

548 return found.astimezone(UTC) == moment.astimezone(UTC)

551def _advance_fixed_schedule(task: dict, *, now: datetime) -> str: 

552 """The ``next_due`` a fixed task moves to once its occurrence is dealt with. 

553 

554 Shared by :func:`apply_completion` and :func:`skip_occurrence`, because completing 

555 an occurrence and skipping it move the schedule in exactly the same way — only the 

556 log each one writes is different. 

557 

558 The schedule advances past ``max(now, next_due)``, and both halves earn their keep: 

559 

560 * **Past ``next_due``**, because the occurrence the task is *showing* is the one 

561 the user just dealt with. Advancing past ``now`` alone hands that occurrence 

562 straight back whenever the task is completed before its time of day: a task 

563 anchored at 10:00 and marked done at 09:00 stayed due at 10:00 today, so the 

564 completion read as though it had done nothing (#331). An anchor late in the 

565 evening was unusable that way for nearly the whole day. 

566 * **Past ``now``**, because an overdue task must not crawl the grid one missed 

567 occurrence per press. A daily task last due a fortnight ago lands on the next 

568 occurrence after today, collapsing the fortnight into a single step. 

569 

570 A task with no ``next_due`` has no occurrence on the board, so *now* is the whole 

571 answer. ``models.build_task`` reaches this state on purpose: a ``last_completed`` 

572 seed is recorded as a completion *before* the schedule is ever computed. 

573 

574 This is deliberately not what :func:`compute_next_due` does. That function derives 

575 a due date from a task's own state with no memory of where the schedule already 

576 stood, which is what makes it the right tool for *rewinding* one (see 

577 :func:`remove_completion`) and the wrong tool for advancing it. 

578 """ 

579 anchor = _parse(task["anchor"]) 

580 assert anchor is not None 

581 current = _parse(task.get("next_due")) 

582 # ``current`` comes off a stored ISO string, so it carries a bare offset; re-home 

583 # it before ``max`` so the winner always hands ``next_fixed_occurrence`` a probe in 

584 # Home Assistant's zone (see ``_regrid``). Comparison itself is instant-based and 

585 # would be correct either way — it is the *tzinfo of the winner* that matters. 

586 if current is not None: 

587 current = current.astimezone(now.tzinfo) 

588 # ...and only when it really is an occurrence. ``next_due`` holds a *deferred* 

589 # instant after a snooze or a due-today, and neither of those is a date the 

590 # user dealt with — advancing past a snooze target threw away every occurrence 

591 # between the task's own one and the target, so snoozing a Monday task a week 

592 # and then doing it anyway lost the Monday in between. 

593 # 

594 # The occurrence that the deferral moved is in ``deferred_from`` (see 

595 # :func:`defer`). That is the occurrence the user deals with, so the 

596 # schedule moves past it. Without it, a due-today and then Done gave back the 

597 # occurrence the user just did (B07-5). When there is no such record, or the 

598 # record is not on the grid, a deferred date reads as "no occurrence on the 

599 # board", which is the ``now``-only branch below. 

600 if not _is_occurrence(task, anchor, current): 

601 current = _local(_parse(task.get(DEFERRED_FROM)), now) 

602 if current is not None and not _is_occurrence(task, anchor, current): 

603 current = None 

604 after = max(now, current) if current is not None else now 

605 return _clamp_season( 

606 next_fixed_occurrence(anchor, task["freq"], int(task["interval"]), after=after), 

607 task, 

608 ).isoformat()

611def apply_completion( 

612 task: dict, 

613 completed_at: datetime, 

614 *, 

615 now: datetime, 

616 metadata: dict | None = None, 

617) -> dict: 

618 """Return *task* mutated to reflect a completion at *completed_at*. 

619 

620 *metadata* is an optional, pre-cleaned mapping of per-completion context 

621 (``note``/``cost``/``photo``/``who`` — see 

622 ``models.normalize_completion_metadata``). Its keys are merged into the new 

623 history entry alongside the mandatory ``ts``; an empty/None mapping records just 

624 the timestamp (the historical behaviour). This function stays agnostic about 

625 *which* keys are valid — cleaning/validation is the caller's job — so the pure 

626 recurrence math is unaffected by the metadata feature. 

627 

628 Records the completion in history (capped) and recomputes ``next_due``: 

629 

630 * floating -> measured from ``completed_at`` (the clock resets) 

631 * fixed -> the next scheduled occurrence after the one the task was showing 

632 (schedule-driven; the completion marks that occurrence done rather than 

633 restarting a clock — see :func:`_advance_fixed_schedule`) 

634 * triggered -> ``next_due = None`` (dormant): completing a condition-driven 

635 task *clears* the condition rather than rescheduling it, so it leaves every 

636 time surface until the owner re-arms it. History is still recorded, so the 

637 replacement cadence accumulates on the task. 

638 """ 

639 # A caller-supplied ``completed_at`` may be naive (HA's ``cv.datetime`` parses 

640 # offset-less strings naively). Qualify it with *now*'s zone before it is 

641 # persisted anywhere: a naive ``last_completed``/``ts``/``next_due`` poisons the 

642 # store — every later aware-vs-naive comparison (``is_overdue``, history ``max``) 

643 # raises ``TypeError`` until the storage file is hand-edited. 

644 if completed_at.tzinfo is None: 

645 completed_at = completed_at.replace(tzinfo=now.tzinfo) 

646 ts_iso = completed_at.isoformat() 

647 rec_type = task.get("recurrence_type", REC_FLOATING) 

648 latest = _latest_ts(task.get("completions", [])) 

649 # A completion older than the latest one fills in the log. It does not tell us 

650 # when the task was last done, so it does not move ``last_completed`` or the 

651 # schedule (B07-3). A later completion already did that. 

652 backfill = latest is not None and completed_at < latest 

653 entry: dict = {"ts": ts_iso} 

654 if metadata: 

655 entry.update(metadata) 

656 if not backfill and _keeps_prior_due(task, rec_type, now=now): 

657 # Keep the due date this completion replaces, so that an undo can put it 

658 # back (B07-1). 

659 entry[PRIOR_DUE] = task.get("next_due") 

660 task["completions"] = _record_entry(task.get("completions", []), entry) 

661 if backfill and rec_type in (REC_FLOATING, REC_FIXED): 

662 return task 

663 if not backfill: 

664 task["last_completed"] = ts_iso 

665 

666 if rec_type == REC_FLOATING: 

667 task["next_due"] = _clamp_season( 

668 # The interval counts on Home Assistant's wall clock (see ``_local``). 

669 add_interval( 

670 completed_at.astimezone(now.tzinfo), 

671 int(task["interval"]), 

672 task["unit"], 

673 ), 

674 task, 

675 ).isoformat() 

676 elif rec_type == REC_FIXED: 

677 task["next_due"] = _advance_fixed_schedule(task, now=now) 

678 elif rec_type in (REC_TRIGGERED, REC_ONE_OFF, REC_SENSOR, REC_USE): 678 ↛ 694line 678 didn't jump to line 694 because the condition on line 678 was always true

679 # A one-off is permanently complete; a triggered task clears its condition; a 

680 # sensor task's crossing has been actioned (and its meter reset by the store, 

681 # which has the live reading). All go dormant (every time surface drops them). 

682 # Undoing the completion re-arms a one-off (see ``remove_completion``); a 

683 # triggered task is re-armed only by its owning integration, and a sensor task 

684 # only by the watcher on a fresh crossing. 

685 # 

686 # A use task is the odd one here: it was already dormant and stays dormant, so 

687 # this assignment changes nothing. Completing it is not "done with the thing" — 

688 # it is the *record of 1 use*, and the entry this call just appended to 

689 # ``completions`` is the whole point. The replacement task beside it is what 

690 # comes due, armed by ``coordinator.async_settle_use_tasks`` once enough 

691 # entries accumulate. 

692 task["next_due"] = None 

693 else: 

694 raise ValueError(f"unknown recurrence_type: {rec_type!r}") 

695 return task

698def skip_occurrence(task: dict, *, now: datetime, metadata: dict | None = None) -> dict: 

699 """Return *task* advanced past its current occurrence with **no** completion. 

700 

701 "Skip this one" — move the task forward off every time surface without recording 

702 that it was *done*. The maintenance log (``completions``) and ``last_completed`` 

703 stay untouched, so a floating task's clock is not reset and every completion count, 

704 cadence average and "last done" reading is unaffected: a skip is the record of 

705 deliberately *not* doing the thing. 

706 

707 The skip is still logged, in its own ``skips`` list — same ``ts``-keyed entry shape 

708 as a completion, carrying the same optional *metadata* (``note``/``who``, plus 

709 ``reading`` for a meter task) — so the panel can show it in history and the user 

710 can amend or undo it. Keeping the two lists apart is what makes "a skip is not a 

711 completion" true by construction rather than by filtering at eighteen call sites. 

712 

713 ``next_due`` moves; because it does, the coordinator's edge detection re-arms the 

714 overdue/due-soon announcements for the new date. 

715 

716 * floating -> ``now + interval·unit`` (a fresh interval from now; the next 

717 completion still measures from *its* ``completed_at``). 

718 * fixed -> the next scheduled occurrence strictly after ``max(now, next_due)``, 

719 exactly as a completion advances it (:func:`_advance_fixed_schedule`). An 

720 upcoming task advances exactly one occurrence; an overdue task jumps to the next 

721 occurrence after *now*, collapsing any already-missed occurrences rather than 

722 taking one skip per missed period. 

723 * one-off / triggered / sensor -> dormant (``next_due = None``): there is no "next" 

724 occurrence to advance to, so skipping clears it from every time surface (a 

725 triggered/sensor task is re-armed only by its owner/the watcher; a one-off stays 

726 done unless its completion is undone). 

727 """ 

728 entry: dict = {"ts": now.isoformat()} 

729 if metadata: 

730 entry.update(metadata) 

731 task["skips"] = _record_entry(task.get("skips", []), entry) 

732 

733 rec_type = task.get("recurrence_type", REC_FLOATING) 

734 if rec_type == REC_FLOATING: 

735 task["next_due"] = _clamp_season( 

736 add_interval(now, int(task["interval"]), task["unit"]), task 

737 ).isoformat() 

738 elif rec_type == REC_FIXED: 

739 task["next_due"] = _advance_fixed_schedule(task, now=now) 

740 elif rec_type in (REC_TRIGGERED, REC_ONE_OFF, REC_SENSOR, REC_USE): 

741 # A use task is already dormant, so this is a no-op on ``next_due``. The skip 

742 # is still logged: "I deliberately did not use it today" is a legitimate thing 

743 # to record, and it stays out of ``completions``, so it never counts as a use. 

744 task["next_due"] = None 

745 else: 

746 raise ValueError(f"unknown recurrence_type: {rec_type!r}") 

747 return task

750def snooze_from(task: dict, now: datetime) -> datetime: 

751 """Return the instant that a snooze length counts from: *now* or the due date. 

752 

753 Snooze moves the due date later. A length counted from *now* moved a task that 

754 is due in 30 days to 7 days from now, which is earlier (F10-2). So the length 

755 counts from the due date when that is later than *now*. For a task that is due 

756 or overdue, it counts from *now*, so the task is not due again at once. 

757 """ 

758 due = _parse(task.get("next_due")) 

759 return now if due is None else max(due, now)

762def defer(task: dict, until: datetime, *, now: datetime) -> dict: 

763 """Return *task* with ``next_due`` moved to *until* by a snooze or a due-today. 

764 

765 No completion or skip is recorded, and the schedule does not change. On a fixed 

766 task the occurrence that the task showed goes in ``deferred_from``, so that a 

767 completion or a skip on the deferred date moves the schedule past that 

768 occurrence (B07-5). A second deferral keeps the first record, because the first 

769 one moved the occurrence off the grid. 

770 """ 

771 if task.get("recurrence_type") == REC_FIXED: 

772 current = _parse(task.get("next_due")) 

773 anchor = _parse(task["anchor"]) 

774 assert anchor is not None 

775 if current is not None and _is_occurrence( 

776 task, anchor, current.astimezone(now.tzinfo) 

777 ): 

778 task[DEFERRED_FROM] = task["next_due"] 

779 task["next_due"] = until.isoformat() 

780 return task

807def _rewinds(task: dict, previous: str | None, moment: datetime) -> bool: 

808 """Whether a log edit at *moment* must calculate ``next_due`` again. 

809 

810 Only an edit that changes ``last_completed`` can change the due date that the 

811 log gives. An edit to an older row must not calculate it again, because the due 

812 date can hold a later skip, snooze, due-today or early completion that the log 

813 does not show (B07-2). A skip that is later than the edit keeps its due date 

814 for the same reason. 

815 """ 

816 if task.get("last_completed") == previous: 

817 return False 

818 skipped = _latest_ts(task.get("skips", [])) 

819 return skipped is None or skipped < moment

822def _keeps_prior_due(task: dict, rec_type: str, *, now: datetime) -> bool: 

823 """Whether a completion must record the due date it replaces. 

824 

825 A fixed schedule cannot calculate that date again from the log, so it always 

826 records it. A floating task can, except after a snooze, a due today or a moved 

827 date: the log does not show those. So a floating task records the date only 

828 when it is not the one that the log gives. 

829 """ 

830 due = _parse(task.get("next_due")) 

831 if due is None: 

832 return False 

833 if rec_type == REC_FIXED: 

834 return True 

835 if rec_type != REC_FLOATING: 

836 return False 

837 return compute_next_due(task, now=now) != due

840def remove_completion(task: dict, ts: str, *, now: datetime) -> dict: 

841 """Return *task* with the completion at ISO timestamp *ts* removed. 

842 

843 Undoes an accidental completion: drops the first matching history entry, 

844 re-derives ``last_completed`` from the remaining history (the latest, or None), 

845 and recomputes ``next_due`` from that state. A fixed or floating completion that 

846 recorded the due date it replaced puts that date back. Else a floating task 

847 rewinds the clock to the prior completion. For a triggered task ``next_due`` is 

848 left untouched (its armed/dormant state is condition-driven, not 

849 history-driven — editing the replacement log must not arm a dormant task). A 

850 no-op when *ts* is not present. 

851 """ 

852 history = list(task.get("completions", [])) 

853 removed: dict | None = None 

854 for index, entry in enumerate(history): 

855 if entry.get("ts") == ts: 

856 removed = history.pop(index) 

857 break 

858 task["completions"] = history 

859 previous = task.get("last_completed") 

860 if history: 

861 latest = max(history, key=lambda entry: datetime.fromisoformat(entry["ts"])) 

862 task["last_completed"] = latest["ts"] 

863 else: 

864 task["last_completed"] = None 

865 rec_type = task.get("recurrence_type") 

866 if removed is None or not _rewinds( 

867 task, previous, datetime.fromisoformat(removed["ts"]) 

868 ): 

869 # The log changed below the latest decision, so the due date that the 

870 # latest completion, skip, snooze or due-today set stays (B07-2). 

871 return task 

872 prior_due = removed.get(PRIOR_DUE) 

873 if rec_type in (REC_FIXED, REC_FLOATING) and prior_due: 

874 # Put back the due date this completion replaced. An overdue occurrence 

875 # comes back as overdue (B07-1), and a snoozed or moved date comes back as 

876 # it was. 

877 task["next_due"] = prior_due 

878 elif rec_type == REC_ONE_OFF: 

879 # Undoing the (final) completion of a do-once task re-arms it to its ``due`` 

880 # date so it returns to every time surface; if any completion remains it 

881 # stays dormant. Unlike a triggered task, a one-off's armed/dormant state is 

882 # history-driven, so editing the log *does* re-arm it. 

883 task["next_due"] = ( 

884 compute_next_due(task, now=now).isoformat() if not history else None 

885 ) 

886 elif rec_type not in (REC_TRIGGERED, REC_SENSOR, REC_USE): 

887 task["next_due"] = compute_next_due(task, now=now).isoformat() 

888 return task

891def update_completion( 

892 task: dict, ts: str, metadata: dict, *, fields: tuple[str, ...] 

893) -> tuple[dict, str | None]: 

894 """Edit the metadata of the completion at ISO timestamp *ts* in place. 

895 

896 Used to amend a past completion (fix a note, add a forgotten cost/photo). For 

897 each key in *fields* (the recognised metadata keys), a non-empty value in 

898 *metadata* is set on the entry and an absent/empty value clears it — so an edit 

899 that blanks the note removes the key rather than storing ``""``. ``ts``, 

900 schedule, and ``last_completed``/``next_due`` are never touched: amending a log 

901 entry must not rewind or re-arm a task. Raises :class:`KeyError`-free — instead a 

902 ``ValueError`` is raised when no entry matches *ts* so the caller can surface a 

903 clear error. 

904 

905 Returns ``(task, replaced_photo)`` where *replaced_photo* is the previous 

906 ``photo`` id when the edit changed or cleared it (so the HA layer can delete the 

907 now-orphaned image), else ``None``. 

908 """ 

909 history = list(task.get("completions", [])) 

910 target: dict | None = None 

911 for entry in history: 

912 if entry.get("ts") == ts: 

913 target = entry 

914 break 

915 if target is None: 

916 raise ValueError(f"no completion at {ts!r}") 

917 old_photo = target.get("photo") 

918 for key in fields: 

919 value = metadata.get(key) 

920 if value in (None, ""): 

921 target.pop(key, None) 

922 else: 

923 target[key] = value 

924 task["completions"] = history 

925 new_photo = target.get("photo") 

926 replaced_photo = old_photo if old_photo and old_photo != new_photo else None 

927 return task, replaced_photo

930def move_completion(task: dict, old_ts: str, new_ts: str, *, now: datetime) -> dict: 

931 """Return *task* with the completion at *old_ts* re-timestamped to *new_ts*. 

932 

933 Back-dates (or corrects) an already-recorded completion — distinct from 

934 ``update_completion`` (which edits metadata but never ``ts``) and from deleting 

935 and re-adding (which would lose the entry's metadata and re-stamp it at "now"). 

936 A caller-supplied *new_ts* may be naive (the ``move_completion`` service's 

937 ``new_completed_at`` field is ``cv.datetime``, which — like ``complete_task``'s 

938 ``completed_at`` — accepts an offset-less value); it is qualified with *now*'s 

939 zone the same way ``apply_completion`` does. 

940 

941 Removes the entry at *old_ts* (raising ``ValueError`` if none matches — mirrors 

942 ``update_completion``'s missing-``ts`` behaviour) and re-inserts it at *new_ts*, 

943 preserving its metadata. Colliding with an existing entry already at *new_ts* 

944 replaces it (the same same-instant dedup ``apply_completion`` does); the moved 

945 entry's metadata wins and the entry it collided with is discarded. 

946 

947 Triggered/sensor tasks' ``next_due`` is left untouched, and so is a **fixed** 

948 task's: its due date is schedule state that completing or skipping already moved 

949 on (:func:`_advance_fixed_schedule`), not a value derived from the log, so 

950 recomputing it from the anchor here would rewind the task onto an occurrence it 

951 has already dealt with. Only :func:`remove_completion` rewinds, because an undo is 

952 meant to. 

953 

954 Both the removal and the re-insertion happen *before* ``last_completed``/ 

955 ``next_due`` are re-derived — re-deriving in between (as if this were 

956 ``remove_completion`` followed by a separate insert) would misfire for a one-off 

957 task: removing its only completion looks like "history now empty" and re-arms 

958 ``next_due`` to ``due``, which would be wrong once the moved entry lands back in 

959 history. Deriving once from the final, post-move history avoids that: a one-off 

960 with its only completion moved stays dormant (``next_due = None``), exactly as it 

961 was before the move. Triggered/sensor tasks' ``next_due`` is left untouched, same 

962 as ``remove_completion`` — their armed/dormant state is condition-driven, not 

963 history-driven, and this function does not validate the new timestamp against it 

964 (a manual history edit is not schedule-validated, same as ``update_completion``/ 

965 ``remove_completion`` today). 

966 """ 

967 history = list(task.get("completions", [])) 

968 index = next((i for i, e in enumerate(history) if e.get("ts") == old_ts), None) 

969 if index is None: 

970 raise ValueError(f"no completion at {old_ts!r}") 

971 entry = dict(history[index]) 

972 del history[index] 

973 

974 new_dt = _parse(new_ts) 

975 assert new_dt is not None 

976 if new_dt.tzinfo is None: 

977 new_dt = new_dt.replace(tzinfo=now.tzinfo) 

978 new_ts_iso = new_dt.isoformat() 

979 entry["ts"] = new_ts_iso 

980 dup = next((i for i, e in enumerate(history) if e.get("ts") == new_ts_iso), None) 

981 if dup is not None: 

982 history[dup] = entry 

983 else: 

984 history.append(entry) 

985 task["completions"] = history 

986 

987 # Unlike remove_completion, history can never be empty here: removing old_ts 

988 # always re-inserts (or collapses) exactly one entry, so there's always a 

989 # latest to derive last_completed from. 

990 previous = task.get("last_completed") 

991 latest = max(history, key=lambda e: datetime.fromisoformat(e["ts"])) 

992 task["last_completed"] = latest["ts"] 

993 

994 rec_type = task.get("recurrence_type") 

995 # *old_ts* matched a stored entry, and a stored ``ts`` is always aware. 

996 old_dt = datetime.fromisoformat(old_ts) 

997 if not _rewinds(task, previous, max(old_dt, new_dt)): 

998 # The moved entry was not the latest and did not become the latest, so the 

999 # due date stays where the latest decision put it (B07-2). 

1000 return task 

1001 if rec_type == REC_ONE_OFF: 

1002 # A moved completion is still a completion: history is never empty here 

1003 # (see above), so a one-off always stays dormant post-move — it only 

1004 # re-arms via remove_completion, which can genuinely empty history. 

1005 task["next_due"] = None 

1006 elif rec_type not in (REC_TRIGGERED, REC_SENSOR, REC_FIXED, REC_USE): 

1007 task["next_due"] = compute_next_due(task, now=now).isoformat() 

1008 return task

1041
