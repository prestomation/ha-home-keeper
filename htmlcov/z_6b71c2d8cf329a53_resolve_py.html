1"""Resolve a service call's ``*_id`` field from either an id or a name. 

2 

3Home Keeper's ids are ``uuid4`` strings (``models.build_task``, 

4``assets.build_asset``). They are exactly right for an integration holding a 

5reference across restarts and exactly wrong for a person writing YAML: nothing in 

6the panel shows them, so ``complete_task`` used to mean "go dig a uuid out of a 

7``list_tasks`` dump first". 

8 

9Home Assistant's own answer to this is one field that takes either form, with the 

10id winning — ``todo.update_item``'s ``item`` is labelled "Item name or UID" and 

11resolved by ``_find_by_uid_or_summary``. This module is that resolution, kept 

12free of Home Assistant imports so ``tests/unit`` can drive it directly. The 

13callers in ``__init__.py`` translate the two exceptions below into a localized 

14``ServiceValidationError``. 

15 

16One deliberate departure from core: core takes the *first* name match and never 

17mentions the others. Home Keeper raises :class:`AmbiguousName` instead, because 

18names here are not unique by design (``docs/INTEGRATING.md`` tells contributors to 

19expect collisions) and the services reached this way include ``delete_task`` and 

20``delete_asset``. Silently deleting one of two identically named tasks is a worse 

21outcome than an error naming both ids. 

22""" 

23 

24from __future__ import annotations 

25 

26from collections.abc import Iterable, Mapping, Sequence 

27from typing import Any

30class ResolveError(Exception): 

31 """Base for a reference that named no object, or more than one.""" 

32 

33 def __init__(self, key: str) -> None: 

34 super().__init__(key) 

35 self.key = key 

36 

37 

38class NotFound(ResolveError): 

39 """No object carries this id or name.""" 

40 

41 

42class AmbiguousName(ResolveError): 

43 """Several objects share this name, so the caller must use an id.""" 

44 

45 def __init__(self, key: str, ids: Iterable[str]) -> None: 

46 super().__init__(key) 

47 self.ids = sorted(ids)

50def _norm(value: str) -> str: 

51 """Fold a name for the last-chance match: trimmed and case-insensitive.""" 

52 return value.strip().casefold() 

53 

54 

55def _folds_to(value: Any, wanted: str) -> bool: 

56 """Whether a stored name folds to *wanted*, tolerating bad stored data. 

57 

58 The type check lives here rather than inside :func:`_norm` returning a 

59 sentinel for a non-string. A sentinel is only unmatchable while every *key* 

60 is a string too, and one non-string key would make it match every object 

61 whose name is also non-string — a trap not worth leaving for the next caller. 

62 """ 

63 return isinstance(value, str) and _norm(value) == wanted 

64 

65 

66def _by_name(candidates: Sequence[tuple[str, Any]], key: str, field: str) -> str | None: 

67 """The single id whose *field* equals *key*, exactly then case-folded. 

68 

69 Takes a sequence, not an iterator: it makes two passes over the candidates. 

70 Returns ``None`` when nothing matched at either strength, so the caller can 

71 distinguish "no such name" from "too many". Ambiguity is judged per pass: two 

72 tasks named "Filter" and "filter" are told apart by the exact pass and only 

73 collide if the caller types neither exactly. 

74 """ 

75 if not isinstance(key, str) or not key.strip(): 

76 return None 

77 wanted = _norm(key) 

78 for match_exact in (True, False): 

79 hits = [ 

80 obj_id 

81 for obj_id, obj in candidates 

82 if ( 

83 (obj.get(field) == key) 

84 if match_exact 

85 else _folds_to(obj.get(field), wanted) 

86 ) 

87 ] 

88 if hits: 

89 if len(hits) == 1: 

90 return hits[0] 

91 raise AmbiguousName(key, hits) 

92 return None 

93

95def _resolve(objects: Iterable[tuple[str, Any]], key: str, field: str) -> str: 

96 """The id of the one object *key* names, by id first and then by *field*.""" 

97 pairs = list(objects) 

98 for obj_id, _obj in pairs: 

99 if obj_id == key: 

100 return obj_id 

101 if (found := _by_name(pairs, key, field)) is not None: 

102 return found 

103 raise NotFound(key)

106def resolve_task_id(tasks: Mapping[str, Any], key: str) -> str: 

107 """The id of the task *key* refers to, by id or by name.""" 

108 return _resolve(tasks.items(), key, "name")

111def resolve_asset_id(assets: Mapping[str, Any], key: str) -> str: 

112 """The id of the appliance *key* refers to, by id or by name.""" 

113 return _resolve(assets.items(), key, "name")

116def _entries(asset: Any, collection: str) -> list[tuple[str, Any]]: 

117 """The ``(id, entry)`` pairs of one of an asset's nested collections.""" 

118 raw = asset.get(collection) if isinstance(asset, Mapping) else None 

119 if not isinstance(raw, (list, tuple)): 

120 return [] 

121 return [ 

122 (str(entry.get("id")), entry) 

123 for entry in raw 

124 if isinstance(entry, Mapping) and entry.get("id") 

125 ]

128def resolve_part_id(asset: Any, key: str) -> str: 

129 """The id of a part *key* refers to, scoped to one appliance. 

130 

131 Scoping is what makes the name form workable here: two appliances may each 

132 have a "Filter", but one appliance rarely has two. 

133 """ 

134 return _resolve(_entries(asset, "parts"), key, "name") 

135

137def resolve_document_id(asset: Any, key: str) -> str: 

138 """The id of a document *key* refers to, scoped to one appliance. 

139 

140 A document's name is never blank — ``assets._normalize_document_entry`` falls 

141 back to the link's host or the uploaded filename — so every document is 

142 reachable by name. 

143 """ 

144 return _resolve(_entries(asset, "documents"), key, "name")
