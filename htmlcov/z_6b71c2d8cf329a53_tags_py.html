1"""Pure routing and authorization for NFC/RFID tag completion. 

2 

3A task can carry a ``tag_id`` (a Home Assistant tag) and, optionally, 

4``require_tag_scan``. This module answers the two questions that follow from that 

5pairing: which tasks a scanned tag completes, and whether a given completion is 

6allowed to go through at all. Both are plain dict-in/answer-out functions with no 

7Home Assistant imports, so the routing table and the authorization rule are 

8unit-testable without a bus; ``tag_listener.py`` is the thin Home Assistant boundary 

9that feeds them real scans and ``store.complete_task`` enforces the rule. 

10""" 

11

15from typing import Any

19 ORIGIN_SENSOR_RECOVER, 

20 ORIGIN_TAG_SCAN, 

21)

23# The ``origin`` markers that satisfy ``require_tag_scan``. A real scan 

24# (``ORIGIN_TAG_SCAN``) is the point of the feature; the other two are Home Keeper 

25# completing the task *itself* — the problem-sensor sync clearing its mirror and a 

26# ``clear_on_recover`` sensor task clearing when its entity returns to normal. Those 

27# are not a person clicking Done, so blocking them would strand the task armed 

28# forever with nobody able to scan it clear. Every human-facing surface (to-do, 

29# button, notification tap, panel, bare ``complete_task`` service) sends no origin or 

30# its own, and is rejected. 

31SCAN_ALLOWED_ORIGINS: frozenset[str] = frozenset( 

32 {ORIGIN_TAG_SCAN, ORIGIN_SENSOR_RECOVER, ORIGIN_PROBLEM_SENSOR_SYNC} 

33)

36def tasks_for_tag(tasks: Iterable[dict[str, Any]], tag_id: str) -> list[dict[str, Any]]: 

37 """Return the enabled tasks bound to *tag_id*. 

38 

39 A tag is many-to-one on purpose: sticking one tag on the coffee machine can 

40 complete "descale" and "refill beans" in a single scan. Disabled tasks are 

41 skipped — a disabled task is invisible on every other surface, so a scan must not 

42 be a back door that completes it. 

43 """ 

44 return [ 

45 task 

46 for task in tasks 

47 if task.get("enabled", True) and task.get("tag_id") == tag_id 

48 ] 

49 

50 

51def completion_allowed(task: dict[str, Any], origin: str | None) -> bool: 

52 """Whether a completion of *task* carrying *origin* may proceed. 

53 

54 ``require_tag_scan`` turns completion into a physical presence check: the task is 

55 only completable by scanning its tag, so a completion is allowed only when it 

56 carries one of :data:`SCAN_ALLOWED_ORIGINS`. A task without the flag is 

57 completable from anywhere, whatever the origin. 

58 """ 

59 if not task.get("require_tag_scan"): 

60 return True 

61 return origin in SCAN_ALLOWED_ORIGINS
