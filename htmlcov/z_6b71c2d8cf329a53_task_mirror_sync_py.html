1"""Home-Assistant-aware half of the task mirror. 

2 

3Reads every configured to-do list over ``todo.get_items``, asks the pure planner in 

4``task_mirror.py`` what each mirror wants done about it, applies the answer with 

5``todo.add_item`` / ``todo.update_item`` / ``todo.remove_item``, and feeds the 

6household's side of the loop — a chore ticked off on somebody's phone — back into 

7the store as a completion. Same split as ``shopping.py`` / ``shopping_sync.py``, 

8and this module inherits that sibling's three properties: 

9 

10* **It never breaks its caller.** A pass runs off the back of a task event, a 

11 list's own state change, or the coordinator's tick; a to-do list that is 

12 unavailable, unloaded, or refuses new items must cost nothing more than a log 

13 line. Every outbound call is best-effort and ``async_sync`` swallows whatever 

14 reaches it. 

15* **A failed call is retried, never compensated.** The planner hands back the 

16 bookkeeping as it will read once every operation has succeeded; anything that 

17 raised puts its entry back, so the next pass tries again. Dropping an entry for 

18 a remove that never happened would orphan that line on the list forever. 

19* **It reads a to-do list only when there is a reason to.** ``needs_pass`` 

20 answers "has Home Keeper drifted from what it mirrored" from memory alone, so 

21 the great majority of task events — which concern tasks no mirror wants — 

22 cost no service calls. The surfaces that watch for the household's side (each 

23 list's own state changes, and the periodic sweep) ask for a full pass 

24 regardless, since that side is invisible from here. 

25 

26Where the two mirrors diverge is what a *vanished* item means: the shopping mirror 

27leaves a deleted line deleted, while a task mirror may read it as "done" so that 

28providers which drop completed items (Todoist) can still tick a chore off. That 

29divergence lives entirely in the planner — see ``task_mirror.py`` — and this 

30driver only supplies the inputs it needs. 

31"""

75# One pass can beget another: an inbound tick completes a task, which reschedules 

76# it, which the lists then have to be told about. Two passes cover that; the extra 

77# headroom absorbs our own writes echoing back through each list's state. The cap 

78# is what guarantees the loop ends whatever the lists do. 

79_MAX_PASSES = 4

81# Every store mutation that can change which tasks a profile surfaces, or what a

84# wants costs nothing, and our own inbound completion echoing back through here

174 @callback 

175 def async_schedule_sweep(self) -> None: 

176 """Ask for a full pass off the coordinator's periodic tick. 

177 

178 Two things only this notices. A list whose outstanding count happens to 

179 land back where it started — one item ticked off while another was added — 

180 produces no state change at all, so the listener alone can miss a tick-off. 

181 And a task *falling due* is not a store mutation, so no task event fires 

182 for it: the periodic pass is what puts a newly-overdue chore on the list. 

183 Costs nothing until something is configured or mirrored. 

184 """ 

185 if self._stopped: 

186 return 

187 configured = profiles.synced_profiles(self._configured_mirrors()) 

188 if not configured and not self._coordinator.store.get_task_mirror_items(): 

189 return 

190 self._hass.async_create_task(self.async_sync(force=True))

192 # ── the pass ───────────────────────────────────────────────────────────── 

193 async def async_sync(self, *, force: bool = False) -> None: 

194 """Bring the mirrors in step. Never raises. 

195 

196 *force* reads the lists even when Home Keeper's own state looks settled, 

197 which is the only way to notice the household's side of the loop. 

198 """ 

199 if self._running: 

200 # Poked mid-pass — our own writes echoing back, or the completion this 

201 # pass just made. Fold it into the run already in flight instead of 

202 # nesting. 

203 self._pending = True 

204 return 

205 self._running = True 

206 self._pending = False 

207 try: 

208 for attempt in range(_MAX_PASSES): 

209 again = await self._sync_once(force=force or attempt > 0) 

210 if not (again or self._pending): 

211 break 

212 self._pending = False 

213 else: 

214 # The budget exists so the loop always ends, not because ending 

215 # this way is fine: a converging mirror settles in two passes. 

216 # Burning the whole budget means something is failing and 

217 # retrying — say so, rather than exiting quietly and leaving the 

218 # lists stale with no trace of why. 

219 _LOGGER.warning( 

220 "Home Keeper's to-do list sync did not settle in %d passes; " 

221 "the mirrored lists may be out of step until the next change", 

222 _MAX_PASSES, 

223 ) 

224 except Exception: # a broken list must never break a task mutation 

225 _LOGGER.exception( 

226 "Home Keeper to-do list sync failed; it will retry on the next change" 

227 ) 

228 finally: 

229 self._running = False 

230 self._pending = False

288 async def _complete_tasks(self, plan: task_mirror.TaskMirrorPlan) -> bool: 

289 """Complete the tasks whose items the household ticked off. 

290 

291 Nothing is restored to the bookkeeping here, and both failure paths mean 

292 it. A task deleted between plan and apply is gone, and the item somebody 

293 ticked off is their record of it now. A task that *refuses* remote 

294 completion — ``require_tag_scan``, by design — must not be quietly re-bound 

295 to a line already ticked off either: leaving the entry dropped puts a fresh 

296 open item back on the list next pass, which is honest feedback that the 

297 tick did not take. 

298 """ 

299 store = self._coordinator.store 

300 completed = False 

301 for op in plan.complete: 

302 try: 

303 await store.complete_task(op.task_id, origin=ORIGIN_TODO_MIRROR) 

304 except KeyError: 

305 # Deleted while the pass was in flight; its item is somebody else's 

306 # record now and there is nothing left to mirror. 

307 _LOGGER.debug( 

308 "Task %s was gone before the to-do list tick reached it", 

309 op.task_id, 

310 ) 

311 except TaskValidationError as err: 

312 self._warn_once( 

313 f"complete:{op.task_id}", 

314 "Home Keeper could not complete task %s from a to-do list (%s), " 

315 "so the tick did not take and the task will reappear on the list", 

316 op.task_id, 

317 err, 

318 ) 

319 else: 

320 completed = True 

321 return completed

335 def _resolve_mirrors(self) -> list[dict[str, Any]]: 

336 """The configured profiles, with any pointed at our own list switched off. 

337 

338 Mirroring Home Keeper's tasks onto Home Keeper's own to-do list is a 

339 feedback loop, and that list declares only ``UPDATE_TODO_ITEM`` so it could 

340 not accept the items anyway. Blanking the target rather than dropping the 

341 profile keeps the planner's own rule intact: a mirror the user turned off 

342 clears what it wrote, so anything already on that list still comes back off 

343 it. 

344 """ 

345 ours = set(own_todo_entity_ids(self._hass)) 

346 resolved: list[dict[str, Any]] = [] 

347 for mirror in self._configured_mirrors(): 

348 entity_id = str(mirror["sync"]["entity_id"]) 

349 if entity_id and entity_id in ours: 

350 self._warn_once( 

351 f"own:{entity_id}", 

352 "Home Keeper cannot mirror its tasks onto its own to-do list " 

353 "(%s); pick a different list in Settings", 

354 entity_id, 

355 ) 

356 mirror = {**mirror, "sync": {**mirror["sync"], "entity_id": ""}} 

357 resolved.append(mirror) 

358 return resolved

413 """Which optional item fields *entity_id* can actually hold. 

414 

415 The planner neither writes nor diffs a field outside this set, so a list 

416 without due dates is never told one — otherwise every pass would "fix" the
