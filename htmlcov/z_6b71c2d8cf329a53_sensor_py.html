1"""Home Keeper sensors. 

2 

3Three kinds, all living on a device page: 

4 

5* ``HomeKeeperNextDueSensor`` — per-task "next due" timestamp, for tasks attached 

6 to a device. 

7* ``HomeKeeperTaskCountSensor`` — an aggregate count, one per profile plus one for 

8 every task Home Keeper keeps, on the integration's own service device. This is the 

9 number a dashboard badge shows. 

10* ``HomeKeeperAssetDateSensor`` — a tracked ``date`` metadata entry on an asset, so 

11 temporal attributes are automatable natively (e.g. "warranty expiring in 30 days 

12 -> notify") and show in state history. Only metadata dates the user opts into 

13 tracking (``track``) get a sensor; the rest are display-only. 

14"""

49# Unique id of the count sensor that spans every task, and the prefix/suffix pair the 

50# per-profile count sensors are keyed by. Anchored to the profile *id* rather than its 

51# name, so renaming a profile keeps the entity id it is already on a dashboard under. 

52_ALL_COUNT_UID = f"{DOMAIN}_tasks" 

53_PROFILE_COUNT_PREFIX = f"{DOMAIN}_profile_" 

54_PROFILE_COUNT_SUFFIX = "_tasks"

63 return [

70async def async_setup_entry( 

71 hass: HomeAssistant, 

72 entry: ConfigEntry, 

73 async_add_entities: AddEntitiesCallback, 

74) -> None: 

75 """Create the next-due, count and asset-date sensors.""" 

76 coordinator: HomeKeeperCoordinator = entry.runtime_data 

77 

78 count_entities, live_count_uids = _count_entities(coordinator, entry) 

79 

80 task_ids = coordinator.device_attached_task_ids() 

81 live_task_ids = set(task_ids) 

82 

83 # Build the set of live asset-date sensor unique-ids (only for assets that have 

84 # a resolvable device and tracked date entries with a value set). 

85 live_asset_meta_uids: set[str] = set() 

86 asset_entities: list[SensorEntity] = [] 

87 for asset in coordinator.store.list_assets(): 

88 device = coordinator.device_entry_for_device_id(asset.get("device_id")) 

89 if device is None: 

90 continue 

91 for meta in _tracked_dates(asset): 

92 uid = f"{DOMAIN}_asset_{asset['id']}_meta_{meta['id']}" 

93 live_asset_meta_uids.add(uid) 

94 asset_entities.append( 

95 HomeKeeperAssetDateSensor(coordinator, asset["id"], meta, device) 

96 ) 

97 

98 # Remove entity-registry entries for sensors whose source no longer exists: 

99 # • per-task next-due sensors for deleted/detached tasks 

100 # • asset date sensors for removed or un-tracked metadata entries 

101 task_prefix = f"{DOMAIN}_" 

102 task_suffix = "_next_due" 

103 asset_prefix = f"{DOMAIN}_asset_" 

104 asset_infix = "_meta_" 

105 

106 def keep(uid: str) -> bool | None: 

107 # The count branches come first so the intent reads in order. They cannot 

108 # collide with the task branch below, which requires the ``_next_due`` suffix. 

109 if uid == _ALL_COUNT_UID: 

110 return True 

111 if uid.startswith(_PROFILE_COUNT_PREFIX) and uid.endswith( 

112 _PROFILE_COUNT_SUFFIX 

113 ): 

114 return uid in live_count_uids 

115 if uid.startswith(task_prefix) and uid.endswith(task_suffix): 

116 return uid[len(task_prefix) : -len(task_suffix)] in live_task_ids 

117 if uid.startswith(asset_prefix) and asset_infix in uid: 

118 return uid in live_asset_meta_uids 

119 return None 

120 

121 prune_registry_entries(hass, entry, "sensor", keep) 

122 

123 task_entities: list[SensorEntity] = [ 

124 HomeKeeperNextDueSensor(coordinator, task_id) for task_id in task_ids 

125 ] 

126 async_add_entities(task_entities + asset_entities + count_entities)

129def _count_entities( 

130 coordinator: HomeKeeperCoordinator, entry: ConfigEntry 

131) -> tuple[list[SensorEntity], set[str]]: 

132 """The count sensors to add, and the unique ids that are still live. 

133 

134 The all-tasks sensor is built **unconditionally** — with no profiles, no assets 

135 and no tasks at all. That is what keeps at least one entity on the service device, 

136 which is what stops ``devices.async_prune_orphaned_devices`` removing it. 

137 """ 

138 # No profile spans everything, so synthesize the filter the same way a stored one 

139 # is built: every include and exclude list empty, so the scope is every active 

140 # scheduled task. The state is then the overdue count, which is the badge case. 

141 all_filter = profiles.normalize_filter({"status": profiles.STATUS_OVERDUE}) 

142 entities: list[SensorEntity] = [ 

143 HomeKeeperTaskCountSensor( 

144 coordinator, 

145 unique_id=_ALL_COUNT_UID, 

146 translation_key="all_tasks", 

147 placeholders={}, 

148 task_filter=all_filter, 

149 ) 

150 ] 

151 live: set[str] = {_ALL_COUNT_UID} 

152 stored = profiles.normalize_profiles(current_options(entry).get(OPTION_PROFILES)) 

153 for profile in stored: 

154 uid = _profile_count_uid(profile["id"]) 

155 live.add(uid) 

156 entities.append( 

157 HomeKeeperTaskCountSensor( 

158 coordinator, 

159 unique_id=uid, 

160 translation_key="profile_tasks", 

161 placeholders={"profile": profile["name"]}, 

162 task_filter=profile["filter"], 

163 ) 

164 ) 

165 return entities, live

178class HomeKeeperNextDueSensor(HomeKeeperTaskEntity, SensorEntity): 

179 """Timestamp sensor reporting when a task is next due.""" 

180 

181 _attr_translation_key = "next_due" 

182 _attr_icon = "mdi:calendar-clock" 

183 _attr_device_class = SensorDeviceClass.TIMESTAMP 

184 

185 def __init__(self, coordinator: HomeKeeperCoordinator, task_id: str) -> None: 

186 super().__init__(coordinator, task_id) 

187 self._attr_unique_id = f"{DOMAIN}_{task_id}_next_due" 

188 

189 @property 

190 def native_value(self) -> datetime | None: 

191 due = self._task.get("next_due") 

192 return dt_util.parse_datetime(due) if due else None 

193 

194 @property 

195 def extra_state_attributes(self) -> dict: 

196 task = self._task 

197 completions = task.get("completions", []) 

198 attrs: dict[str, Any] = { 

199 "task_id": self._task_id, 

200 "task_name": task.get("name"), 

201 "recurrence_type": task.get("recurrence_type"), 

202 "last_completed": task.get("last_completed"), 

203 "completions_count": len(completions), 

204 } 

205 # Surface the most recent completion's fields so automations/dashboards can 

206 # read "who did it / what did it cost / the note / the photo / where the meter 

207 # stood" without parsing the history array. Driven off the shared field list 

208 # rather than a local literal, so a new completion field lands here too — this 

209 # loop silently skipped ``reading`` for exactly that reason. Keys are only 

210 # present when that completion recorded them. 

211 if completions: 

212 # Match ``last_completed`` (the chronologically latest), not merely the 

213 # last appended, so an out-of-order seed can't shadow a real completion. 

214 latest = max(completions, key=lambda c: c.get("ts") or "") 

215 for key in COMPLETION_ENTRY_FIELDS: 

216 if key in latest: 

217 attrs[f"last_completion_{key}"] = latest[key] 

218 attrs.update(self._usage_attributes(task)) 

219 return attrs 

220 

221 def _usage_attributes(self, task: dict[str, Any]) -> dict[str, Any]: 

222 """Meter progress for a usage sensor task, so automations can read it. 

223 

224 A usage task's whole state is "how far through the interval am I" — the same 

225 figures a dedicated maintenance helper would expose as separate entities. They 

226 ride on this task's existing next-due sensor instead, which keeps one entity per 

227 task (Home Keeper's model) while still making the numbers automatable and 

228 templatable. Absent entirely for non-usage tasks. 

229 """ 

230 cfg = sensor_tasks.sensor_config(task) 

231 if cfg is None or cfg.get("mode") != SENSOR_MODE_USAGE: 

232 return {} 

233 target = float(cfg["target"]) 

234 attrs: dict[str, Any] = {"usage_target": target} 

235 if unit := cfg.get("unit"): 

236 attrs["usage_unit"] = unit 

237 reading = read_sensor_value(self.hass, cfg) 

238 baseline = cfg.get("baseline") 

239 if baseline is not None: 

240 # The anchor the consumed figure is measured from. Worth exposing in its 

241 # own right now that it is user-settable at creation and correctable from 

242 # the history: a dashboard showing "3,000 of 10,000 used" can say what the 

243 # odometer read when the oil was last changed. 

244 attrs["usage_baseline"] = float(baseline) 

245 if reading is not None and baseline is not None: 

246 consumed = max(0.0, reading - float(baseline)) 

247 attrs["usage_consumed"] = round(consumed, 3) 

248 attrs["usage_remaining"] = round(target - consumed, 3) 

249 attrs["usage_percent"] = round(min(100.0, consumed / target * 100), 1) 

250 due_at = sensor_tasks.backstop_due(task, cfg) 

251 if due_at is not None: 

252 attrs["backstop_due"] = due_at.isoformat() 

253 # How much use each past service interval actually ran, summarized. The 

254 # readings are already in the history; a template that wants "is this one 

255 # running short?" should not have to fetch the whole log and subtract. 

256 # Absent until two completions carry a reading. The attribute names are 

257 # spelled out rather than derived from the stats keys, because they are the 

258 # public contract a template writes against. 

259 intervals = sensor_tasks.usage_interval_stats(task) 

260 for stat, attr in _USAGE_INTERVAL_ATTRS.items(): 

261 if stat in intervals: 

262 attrs[attr] = round(intervals[stat], 3) 

263 return attrs

266class HomeKeeperTaskCountSensor(CoordinatorEntity[HomeKeeperCoordinator], SensorEntity): 

267 """How many tasks a profile surfaces right now, plus the next one up. 

268 

269 The state is the count **that profile's own status tier selects** — an overdue 

270 profile counts overdue tasks, a due-soon profile counts overdue and due-soon 

271 together, an every-scheduled-task profile counts its whole scope. That is the 

272 number a dashboard badge shows, and it is computed by 

273 ``profiles.matches_filter``, so it can never disagree with the panel, the 

274 dashboard card, a notification or a to-do list sync about the same profile. 

275 

276 The attributes are taken over the profile's **scope** (the same filter with the 

277 status widened to "all"), so ``total`` keeps its meaning however the tier is set 

278 and the next-up pointers name a task even when the state is 0. 

279 

280 Counts refresh on the coordinator's cadence, the same as 

281 ``binary_sensor.*_overdue``, so an overdue count can trail reality by a few 

282 minutes. 

283 """ 

284 

285 _attr_has_entity_name = True 

286 _attr_icon = "mdi:clipboard-text-clock" 

287 # A plain count: MEASUREMENT makes Home Assistant keep long-term statistics, so a 

288 # user can graph "is the backlog growing". Deliberately no unit — a literal 

289 # "tasks" would not translate, and adding a unit to a recorded statistic later is 

290 # a migration rather than a free change. 

291 _attr_state_class = SensorStateClass.MEASUREMENT 

292

309 def _counts(self) -> dict[str, Any]: 

310 """The whole count result for this profile, as of now. 

311 

312 Tasks go through ``notifier.effective_filter_tasks`` first: a task inherits 

313 labels and an area from the device it is attached to, and the pure filter only 

314 sees a task's own fields. Without the enrichment a profile filtering on an 

315 inherited label would count a different set here than the panel and the card 

316 show for that same profile. 

317 

318 ``dt_util.now()`` — local time, never ``utcnow()``, because ``due_today`` is a 

319 calendar date in the user's own zone. 

320 """ 

321 tasks = notifier.effective_filter_tasks( 

322 self.hass, list(self.coordinator.data.values()) 

323 ) 

324 return task_counts.count_tasks(tasks, self._filter, now=dt_util.now())

327 def native_value(self) -> int: 

328 return int(self._counts()["state"])

330 @property 

331 def extra_state_attributes(self) -> dict[str, Any]: 

332 return {key: value for key, value in self._counts().items() if key != "state"}

335class HomeKeeperAssetDateSensor(CoordinatorEntity[HomeKeeperCoordinator], SensorEntity): 

336 """A single tracked ``date`` metadata entry for an asset, on its device page. 

337 

338 The value lives in the asset's free-form ``metadata`` list (not the task map); 

339 asset edits reload the config entry, which recreates these entities, so a plain 

340 coordinator read is enough to stay current. The entity is named from the entry's 

341 user-supplied ``label`` (not a translation, since labels are free-form). 

342 """ 

343 

344 _attr_has_entity_name = True 

345 _attr_device_class = SensorDeviceClass.DATE 

346 

347 def __init__( 

348 self, 

349 coordinator: HomeKeeperCoordinator, 

350 asset_id: str, 

351 entry: dict[str, Any], 

352 device: dr.DeviceEntry, 

353 ) -> None: 

354 super().__init__(coordinator) 

355 self._asset_id = asset_id 

356 self._entry_id = entry["id"] 

357 # Name is the user's free-form label (no localization for custom fields). 

358 self._attr_name = entry.get("label") 

359 # A user-chosen appliance icon overrides the default. 

360 asset = coordinator.store.get_asset(asset_id) or {} 

361 self._attr_icon = asset.get("icon") or _DATE_ICON 

362 self._attr_unique_id = f"{DOMAIN}_asset_{asset_id}_meta_{entry['id']}" 

363 # Linked, not owned: the appliance device belongs to whoever created it. 

364 self.device_entry = device 

365 

366 def _entry(self) -> dict[str, Any] | None: 

367 asset = self.coordinator.store.get_asset(self._asset_id) or {} 

368 for entry in asset.get("metadata") or []: 

369 if entry.get("id") == self._entry_id: 

370 return entry 

371 return None 

372 

373 @property 

374 def native_value(self) -> date | None: 

375 entry = self._entry() 

376 value = entry.get("value") if entry else None 

377 if not value: 

378 return None 

379 try: 

380 return date.fromisoformat(str(value)[:10]) 

381 except ValueError: 

382 return None
