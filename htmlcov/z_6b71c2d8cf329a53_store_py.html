119def _changed_fields(before: dict[str, Any], after: dict[str, Any]) -> list[str]: 

120 """Top-level keys whose value differs between *before* and *after*. 

121 

122 Drives the ``changed_fields`` payload on the update events (and the decision to 

123 suppress a no-op update event entirely). Backend-managed bookkeeping keys that 

124 aren't user-meaningful are ignored so a reschedule doesn't spam ``next_due``/ 

125 ``completions`` churn as "changes". 

126 """ 

127 ignore = {"completions", "last_completed", "next_due", "created"} 

128 keys = (set(before) | set(after)) - ignore 

129 return sorted(k for k in keys if before.get(k) != after.get(k))

293 async def add_task(self, data: dict[str, Any]) -> dict[str, Any]: 

294 """Create and persist a new task; returns the created task dict. 

295

297 ``problem_sensor`` and ``buy`` namespaces are *reserved* — the reconcilers own

299 doesn't match a live part/sensor). Reject them here so an external ``add_task`` 

300 can't mint a task that masquerades as reconciler-owned and then vanishes on the 

301 next reconcile pass. Home Keeper's own reconcilers build such tasks via 

302 ``models.build_task`` directly, not through this service path. 

303 """

314 "own task reconcilers and cannot be set via add_task"

353 async def set_task_consumable( 

354 self, task_id: str, asset_id: str | None, part_id: str | None 

355 ) -> dict[str, Any]: 

356 """Link a task to an asset consumable/part, or clear the link. 

357 

358 A manual link sets the task's ``source`` to a part reference (flagged 

359 ``manual`` so the part-task reconciler leaves it alone — see 

360 ``reconcile.is_manual_part_link``) so that *completing* the task draws the 

361 part's per-use amount off its ``stock`` (one whole spare unless the part sets 

362 its own ``consume_quantity``), firing the edge-triggered low/out-of-stock 

363 events when the reorder threshold is crossed — exactly like a wear-part 

364 replacement. This is how a user wires an arbitrary task (e.g. a sensor task 

365 armed by a fridge's filter-life entity) to the consumable it depletes. 

366 

367 Pass both ``asset_id`` and ``part_id`` to link; pass both as ``None`` to clear. 

368 Raises ``KeyError`` for an unknown task; ``TaskValidationError`` for an unknown 

369 asset/part, a half-specified link, or a task already owned by another source (a

374 raise KeyError(task_id) 

375 _reject_synced_problem(existing, None)

377 # hand would just be undone on the next reconcile. Only a user-owned task (no 

378 # part source) or an existing manual link may be (re)linked or cleared. 

379 src = _part_source(existing) 

380 if src is not None and not src.get("manual"): 

381 raise models.TaskValidationError( 

382 "This task is auto-generated from an appliance wear part and is " 

383 "already linked to it — manage its part in the appliance editor." 

384 ) 

385 

386 if asset_id is None and part_id is None: 

387 if existing.get("source") is None: 

388 return existing # already unlinked — no-op, no event 

389 existing["source"] = None 

390 else: 

391 if not asset_id or not part_id: 

392 raise models.TaskValidationError( 

393 "linking a consumable needs both asset_id and part_id" 

394 )

396 if asset is None: 

397 raise models.TaskValidationError(f"unknown asset: {asset_id!r}")

403 TASK_SOURCE_PART: { 

404 "asset_id": asset_id, 

405 "part_id": part_id, 

406 "manual": True, 

407 } 

408 }

410 return existing # already linked to this part — no-op, no event

412 await self._save()

423 self, task_id: str, *, origin: str | None = None 

424 ) -> dict[str, Any]: 

425 """Arm a condition-driven (``triggered``) or sensor-based task so it's due-now. 

426 

427 Sets ``next_due`` to now, moving the task from dormant to active. This is the 

428 owner-facing counterpart to ``complete_task`` (which clears it back to

447 self._hass.bus.async_fire(

449 )

452 async def snooze_task( 

453 self, task_id: str, until: Any, *, origin: str | None = None 

454 ) -> dict[str, Any]: 

455 """Push a task's ``next_due`` to *until* without recording a completion. 

456 

457 "Remind me later": unlike :meth:`complete_task` this advances *only* 

458 ``next_due`` — recurrence, ``last_completed`` and the completion history are 

459 untouched — so the schedule isn't advanced, just deferred. Because 

460 ``next_due`` changes, the coordinator re-arms the edge-triggered 

461 overdue/due-soon events for the new date (a fresh reminder fires when the 

462 snooze lapses). *until* is a timezone-aware datetime (the caller computes it 

463 from the requested duration). Rejects a **dormant** task 

464 (``next_due is None``) — there's no due date to defer. 

465 Fires ``home_keeper_task_snoozed``. 

466 

467 Unlike ``complete_task``/``skip_task`` this **accepts a synced problem-sensor 

468 task**. Those reject the other two because both assert the problem is dealt 

469 with, which only the originating integration can decide. Snooze asserts 

470 nothing of the sort: it defers the reminder and leaves the problem standing. 

471 It also survives the sync — ``problem_tasks.reconcile_problem_tasks`` reads 

472 armed as ``next_due is not None``, so a snoozed mirror stays armed while its 

473 sensor is bad and still auto-clears when the sensor returns to OK. Without it 

474 a walk notification had no button that could move such a task on, so it was 

475 left out of walks entirely (#248). 

476 """ 

477 existing = self._tasks.get(task_id) 

478 if existing is None: 

479 raise KeyError(task_id) 

480 if existing.get("next_due") is None: 

481 # A dormant task (a completed one-off, or a condition/sensor task not yet 

482 # armed) has no due date to defer; snoozing it would silently re-arm 

483 # something that was intentionally off every time surface. 

484 raise models.TaskValidationError( 

485 "This task is dormant (no due date) — snooze only defers a task that " 

486 "is currently scheduled. Re-arm it instead (undo a completion, or wait " 

487 "for its condition/sensor)." 

488 ) 

489 existing["next_due"] = until.isoformat() 

490 await self._save() 

491 _LOGGER.debug("Snoozed task %s until %s", task_id, existing["next_due"]) 

492 self._hass.bus.async_fire( 

493 EVENT_TASK_SNOOZED, 

494 events.task_event_data( 

495 existing, extra={"snoozed_until": existing["next_due"]} 

496 ), 

497 ) 

498 return existing

890 async def async_merge_split_duplicates(self, canonical: dict[str, str]) -> int: 

891 """Merge contributed tasks the device split duplicated; return how many went. 

892 

893 When Home Assistant 2026.8 renumbered a device, a contributor looking for its 

894 task by its own stored device id found nothing and created a second one. The 

895 user can remove neither by hand: contributed tasks are deletion-protected and 

896 their ``device_id`` is locked. So this has to do it. 

897 

898 *canonical* maps every split device id back to the composite it came from, which 

899 is what makes the two recognisable as one thing — they point at different live 

900 devices (each half of the same original) and so would otherwise look unrelated. 

901 Two tasks are the same only when the contributor's whole ``source`` payload 

902 matches once device ids are canonicalized, which for every known contributor 

903 means the same device *and* item (bambu-lab distinguishes firmware from each 

904 maintenance item there, Pawsistant keys on a schedule id). 

905 

906 The survivor keeps the **history** (oldest wins ties) but adopts the **newest** 

907 task's ``device_id``: the newer one was created by the contributor *after* the 

908 split, so its id is that contributor's own current answer for where the task 

909 belongs. Adopting anything else just gets the task duplicated again on the next 

910 reconcile. 

911 

912 Never removes a task carrying completions, so no recorded completion can be 

913 lost. Deletions go through ``delete_task`` so ``home_keeper_task_deleted`` still 

914 fires and contributors stay in step, with ``force=True`` because contributed 

915 tasks are deletion-protected — that protection exists to stop a *user* deleting 

916 something their integration will recreate, and here the integration created the 

917 duplicate and has no way to clean it up itself. 

918 """ 

919 

920 def key_for(namespace: str, payload: dict[str, Any]) -> str: 

921 normalized = dict(payload) 

922 device_id = normalized.get("device_id") 

923 if isinstance(device_id, str): 

924 normalized["device_id"] = canonical.get(device_id, device_id) 

925 return f"{namespace}|{json.dumps(normalized, sort_keys=True, default=str)}" 

926 

927 groups: dict[str, list[dict[str, Any]]] = {} 

928 for task in self._tasks.values(): 

929 for namespace, payload in (task.get("source") or {}).items(): 

930 if isinstance(payload, dict): 

931 groups.setdefault(key_for(namespace, payload), []).append(task) 

932 

933 removed = 0 

934 for group in groups.values(): 

935 if len(group) < 2: 

936 continue 

937 by_age = sorted(group, key=lambda t: t.get("created") or "") 

938 # Rank by position rather than `by_age.index(task)`: `list.index` compares 

939 # with `==`, so it would resolve two value-equal tasks to the same position. 

940 # Distinct ids make that unreachable today, which is exactly the kind of 

941 # invariant that quietly stops holding. 

942 _, survivor = max( 

943 enumerate(by_age), 

944 key=lambda pair: (len(pair[1].get("completions") or []), -pair[0]), 

945 ) 

946 adopted = by_age[-1].get("device_id") 

947 

948 if adopted and survivor.get("device_id") != adopted: 

949 survivor["device_id"] = adopted 

950 for payload in (survivor.get("source") or {}).values(): 

951 if isinstance(payload, dict) and payload.get("device_id"): 

952 payload["device_id"] = adopted 

953 

954 for duplicate in group: 

955 # Skipping a duplicate that carries completions leaves it in place, 

956 # which is the intended trade: a leftover task is recoverable, a 

957 # deleted completion history is not. 

958 if duplicate is survivor or (duplicate.get("completions") or []): 

959 continue 

960 await self.delete_task(duplicate["id"], force=True) 

961 removed += 1 

962 

963 if removed: 

964 await self._save() 

965 return removed

978 async def delete_asset(self, asset_id: str) -> dict[str, Any] | None: 

979 """Remove an asset; returns the removed asset (for device cleanup) or None. 

980 

981 Drops the tasks **derived** from this asset's wear parts, but only **unlinks** 

982 a user-owned task that was *manually* linked to one of its consumables (clearing 

983 the dangling part source, keeping the task and its history). Also detaches any 

984 child asset that named it as a parent (so the child becomes standalone). 

985 """ 

986 asset = self._assets.pop(asset_id, None) 

987 if asset is None: 

988 return None 

989 dropped_ids = set() 

990 unlinked = [] 

991 for tid, t in self._tasks.items(): 

992 # An auto-created buy task for this appliance's part goes with it (it 

993 # carries a ``buy`` source, not ``part``, so the loop below would skip it). 

994 buy = _buy_source(t) 

995 if buy is not None and buy.get("asset_id") == asset_id: 

996 dropped_ids.add(tid) 

997 continue 

998 src = _part_source(t) 

999 if src is None or src.get("asset_id") != asset_id: 

1000 continue 

1001 if _is_manual_part_link(t): 

1002 # A user-owned manual link: keep the task, just clear the now-dangling 

1003 # link (its consumable is gone with the appliance). 

1004 t["source"] = None 

1005 unlinked.append(t) 

1006 else: 

1007 dropped_ids.add(tid) 

1008 dropped = [self._tasks[tid] for tid in dropped_ids] 

1009 self._tasks = { 

1010 tid: t for tid, t in self._tasks.items() if tid not in dropped_ids 

1011 } 

1012 for child in self._assets.values(): 

1013 if child.get("parent_asset_id") == asset_id: 

1014 child["parent_asset_id"] = None 

1015 await self._save() 

1016 # Drop any uploaded-document blobs the appliance owned (best-effort cleanup). 

1017 from . import manuals # lazy import: avoids a load-time import cycle 

1018 

1019 await manuals.async_delete_asset_documents(self._hass, asset_id) 

1020 # The appliance and the wear-part tasks it owned are both gone — announce each. 

1021 for task in dropped: 

1022 self._hass.bus.async_fire(EVENT_TASK_DELETED, events.task_event_data(task)) 

1023 # A surviving manual-link task lost its consumable — announce the source change. 

1024 for task in unlinked: 

1025 self._hass.bus.async_fire( 

1026 EVENT_TASK_UPDATED, 

1027 events.task_event_data(task, extra={"changed_fields": ["source"]}), 

1028 ) 

1029 self._hass.bus.async_fire(EVENT_ASSET_DELETED, events.asset_event_data(asset)) 

1030 return asset

1308 self, task_id: str, ts: str, metadata: dict[str, Any] 

1309 ) -> dict[str, Any]: 

1310 """Amend a recorded completion's fields (note/cost/photo/who, and ``reading``). 

1311

1313 **schedule** — amending the maintenance log never moves ``ts``,

1315

1317 **latest** completion of a **usage** sensor task re-anchors ``sensor.baseline``

1320 last completion (see :meth:`_reset_usage_baseline`), so leaving it behind would

1325 for the same reason in reverse: undoing the anchoring completion restores the 

1326 baseline that completion had replaced (recorded on it as ``meter_start``), so 

1327 the meter reverts to the progress the user had. ``move_completion`` leaves the 

1328 baseline alone — the meter follows the *reading*, and re-timestamping a 

1329 completion changes none.

1332 ``home_keeper_task_completion_updated`` (carrying ``meter_baseline`` when the

1341 _reject_synced_problem(existing, None)

1343 metadata, allow_reading=models.task_records_reading(existing) 

1344 ) 

1345 try: 

1346 updated, _replaced_photo = recurrence.update_completion( 

1347 dict(existing), 

1348 ts, 

1349 clean_metadata, 

1350 fields=tuple(COMPLETION_ENTRY_FIELDS), 

1351 ) 

1352 except ValueError as err: 

1353 raise models.TaskValidationError(str(err)) from err

1356 await self._save()

1358 # HA boundary once the panel actually uploads images (frontend phase).

1370

1372 pure bookkeeping and leaves it alone. Returns the new baseline when it moved, 

1373 else ``None`` — the caller puts that in the event so a listener can see the

1393 )

1572 def _stamp_part_replacement(self, task: dict[str, Any], when: Any) -> None: 

1573 src = _part_source(task) 

1574 if not src: 

1575 return 

1576 asset = self._assets.get(src["asset_id"]) 

1577 if not asset: 

1578 return 

1579 # ``as_local`` first (#250): *when* carries the offset the caller supplied — UTC 

1580 # for a completion back-dated through the panel's date picker — and a bare 

1581 # ``.date()`` would take the calendar date in that offset. ``last_replaced`` is 

1582 # a date-only string that ``reconcile`` re-anchors the part's recurrence to, so 

1583 # a one-day shift here shifts the whole wear cycle. The ``hasattr`` guard keeps 

1584 # the true branch a datetime: a plain ``date`` has no ``.date()`` method. 

1585 when_date = ( 

1586 dt_util.as_local(when).date().isoformat() 

1587 if hasattr(when, "date") 

1588 else str(when)[:10] 

1589 ) 

1590 for part in asset.get("parts", []): 

1591 if part.get("id") == src.get("part_id"): 

1592 part["last_replaced"] = when_date 

1593 # Completing a wear-part replacement consumes the part's per-use 

1594 # amount (one whole spare unless it says otherwise); signal a 

1595 # low/out-of-stock crossing so users can automate a reorder.

1599 def _stamp_buy_restock(self, task: dict[str, Any]) -> None: 

1600 """On completing an auto-created buy task, restock its part. 

1601 

1602 Adds the part's ``restock_quantity`` (default 1, and decimal like every other 

1603 spare quantity — a bottle's worth can be 0.5 of a crate) to ``stock`` and 

1604 emits the resulting stock transition — normally a ``restocked`` crossing, which 

1605 the next buy-task reconcile turns into removal of this now-satisfied reminder. 

1606 Buy tasks never carry a ``part`` source, so ``_stamp_part_replacement`` leaves 

1607 them untouched (no double-mutation). 

1608 """ 

1609 src = _buy_source(task) 

1610 if not src: 

1611 return 

1612 asset = self._assets.get(src["asset_id"]) 

1613 if not asset: 

1614 return

1616 if part.get("id") == src.get("part_id"): 

1617 qty = assets.part_restock_quantity(part) 

1618 self._emit_stock_event(assets.adjust_part_stock(part, qty), asset, part)

1624 """Fire the bus event for a stock *transition* (low / out / restocked), if any. 

1625 

1626 Edge-triggered: ``assets.stock_transition`` returns the crossing this change 

1627 caused (or ``none``), so users can automate a reorder / shopping-list add /

1629 

1630 These stay the general-purpose hook even now that auto-buy reminders can be 

1631 mirrored onto a configured to-do list (``shopping_sync.py``): that mirror is

1634 integration of its own.

1640 async def adjust_part_stock( 

1641 self, asset_id: str, part_id: str, delta: float 

1642 ) -> dict[str, Any]: 

1643 """Change a part's on-hand spare quantity by ``delta`` (clamped at zero). 

1644 

1645 Persists and fires the matching edge-triggered stock event once when the 

1646 adjustment crosses a threshold — low-stock, out-of-stock, or restocked (a 

1647 decrease while already low never nags). Returns the updated asset. Raises 

1648 ``KeyError`` for an unknown asset or part. 

1649 """

1654 if part.get("id") == part_id: 

1655 transition = assets.adjust_part_stock(part, delta)
