1"""Pure helpers for actionable **notifications** — the delivery layer (no HA imports). 

2 

3A *notification* is a delivery binding: it references a **profile** (the saved filter 

4in ``profiles.py`` that decides *which* tasks) by ``profile_id`` and adds *how* to 

5deliver them — mobile targets, the button set, snooze duration, style (walk/digest), 

6and automatic triggers. This module owns only that delivery concern: notification 

7normalization, the mobile-app **payload builders**, and the **action-string** 

8encode/decode that routes a notification tap back to the right task and notification 

9(and tells a fresh tap from a stale one — see :func:`is_current_action`). 

10The filter/queue live in ``profiles.py``; HA-aware sending in ``notifier.py``. 

11 

12See ``docs/PROFILES_REFACTOR_PLAN.md`` / ``docs/ACTIONABLE_NOTIFICATIONS_PLAN.md``. 

13"""

31# The notify services Home Keeper delivers to. Actionable payloads go to the 

32# companion app's legacy per-device ``notify.mobile_app_*`` services (the newer 

33# notify entity API doesn't carry ``data.actions``, which every button rides on); 

34# ``persistent_notification`` is allowed alongside them as the one sink that stays 

35# *inside* Home Assistant — it drops the buttons but shows the text, and it cannot 

36# carry anything off the instance. 

37TARGET_PREFIX = "mobile_app_" 

38TARGET_PERSISTENT = "persistent_notification"

40# Notification action verbs (the button behaviours). ``open`` is a client-side URI 

41# deep-link (no backend callback). 

42ACTION_COMPLETE = "complete" 

43ACTION_SNOOZE = "snooze" 

44ACTION_SKIP = "skip" 

45ACTION_OPEN = "open" 

46ACTIONS = (ACTION_COMPLETE, ACTION_SNOOZE, ACTION_SKIP, ACTION_OPEN) 

47DEFAULT_ACTIONS = [ACTION_COMPLETE, ACTION_SNOOZE, ACTION_OPEN]

54DEFAULT_SNOOZE_HOURS = 24

56# Action-string scheme: 

57# ``home_keeper::<verb>::<task_id>::<notification_id>::<due_token>``. The action string 

58# is the only field reliably echoed back in ``mobile_app_notification_action``, so the 

59# verb, task, notification and a freshness token are all encoded into it. The prefix 

60# scopes ours on a global event bus that also carries other integrations' actions. 

61# 

62# The trailing ``due_token`` is the task's ``next_due`` at send time (see 

63# :func:`due_token`), which makes the button a snapshot that can be checked against the 

64# live task on tap — see :func:`is_current_action`. Notifications delivered *before* 

65# tokens existed carry the older four-field form, which still decodes (with no token) 

66# so buttons already sitting on someone's phone keep working. 

67_ACTION_PREFIX = "home_keeper" 

68_ACTION_SEP = "::"

71def due_token(task: dict[str, Any]) -> str: 

72 """The freshness token for *task* — its ``next_due``, verbatim. 

73 

74 Raw rather than hashed on purpose: ISO-8601 never contains the ``::`` separator, 

75 the values compare as exact strings with no parsing or timezone normalization, and 

76 it reads plainly in a debug log or a hand-written test action string. 

77 """ 

78 return str(task.get("next_due") or "")

81def is_current_action( 

82 task: dict[str, Any], token: str | None, *, tokenless_ok: bool 

83) -> bool: 

84 """Whether a tapped action still matches the task state its button was built from. 

85 

86 *token* is the ``next_due`` snapshot the button carried. A mismatch means the task 

87 moved between the send and the tap — completed, snoozed, skipped or rescheduled, 

88 whether from this notification, a second card for the same task, or another surface 

89 entirely — so the tap is stale and must not act on the task again. 

90 

91 Comparing against the task's *live* ``next_due`` (rather than anything scoped to the 

92 notification the tap came from) is what makes two independent cards for one task 

93 invalidate each other: they encode the same token, so whichever is tapped first 

94 advances the task and strands the other. 

95 

96 A notification delivered before tokens existed carries ``token is None``. There is 

97 nothing to compare, so the caller's own due-state check (*tokenless_ok*) decides. 

98 """ 

99 if token is None: 

100 return tokenless_ok 

101 return token == due_token(task)

104def encode_action(verb: str, task_id: str, notification_id: str, token: str) -> str: 

105 """Build the action identifier carried on a notification button. 

106 

107 *token* is the task's freshness token — see :func:`due_token`. 

108 """ 

109 return _ACTION_SEP.join((_ACTION_PREFIX, verb, task_id, notification_id, token))

112def decode_action(action: str | None) -> tuple[str, str, str, str | None] | None: 

113 """Parse an action string into ``(verb, task_id, notification_id, due_token)``. 

114 

115 Accepts both the current five-field form and the legacy four-field one, whose 

116 ``due_token`` comes back as ``None``. Returns ``None`` for anything that isn't a 

117 well-formed Home Keeper action (a foreign integration's action, or a 

118 malformed/empty value) so the listener can ignore it. 

119 """ 

120 if not action: 

121 return None 

122 parts = action.split(_ACTION_SEP) 

123 if len(parts) not in (4, 5) or parts[0] != _ACTION_PREFIX: 

124 return None 

125 verb, task_id, notification_id = parts[1], parts[2], parts[3] 

126 token = parts[4] if len(parts) == 5 else None 

127 if verb not in ACTIONS or not task_id or not notification_id: 

128 return None 

129 if token is not None and not token: 

130 # Our builder always encodes a real ``next_due`` (a task only reaches a 

131 # notification with one), so an empty token is malformed, not legacy. 

132 return None 

133 return verb, task_id, notification_id, token

150def split_targets(value: Any) -> tuple[list[str], list[str]]: 

151 """Partition raw notify targets into ``(accepted, rejected)``. 

152 

153 The picker in the panel only ever offers ``mobile_app_*`` services, but that is a 

154 UI courtesy — nothing stopped a stored notification (or a ``target:`` override on 

155 ``home_keeper.notify``) from naming *any* notify service the admin has set up: 

156 SMTP, Telegram, a webhook. That turns Home Keeper into a confused deputy, relaying 

157 text an unprivileged caller controls out through a channel they could not reach 

158 themselves. So the allowlist is enforced where the value is *stored*, not only 

159 where it is displayed. 

160 

161 ``persistent_notification`` is allowed because it is the one destination that 

162 never leaves the instance: what it delivers, an HA user could already read. 

163 """ 

164 accepted: list[str] = [] 

165 rejected: list[str] = [] 

166 for target in _str_list(value): 

167 (accepted if is_allowed_target(target) else rejected).append(target) 

168 return accepted, rejected

171def normalize_notification(raw: Any) -> dict[str, Any]: 

172 """Coerce one raw notification to its stored, fully-defaulted shape. 

173 

174 A notification references a profile (``profile_id``) and carries delivery: an id 

175 (stable, referenced by action strings), a name, mobile ``targets``, the ordered 

176 ``actions`` button set (clamped to known verbs, de-duplicated), ``snooze_hours``, 

177 ``style`` (walk/digest), and ``auto`` triggers. 

178 """ 

179 raw = raw if isinstance(raw, dict) else {} 

180 actions: list[str] = [] 

181 for a in _str_list(raw.get("actions")): 

182 if a in ACTIONS and a not in actions: 

183 actions.append(a) 

184 auto = raw.get("auto") if isinstance(raw.get("auto"), dict) else {} 

185 try: 

186 snooze_hours = int(raw.get("snooze_hours", DEFAULT_SNOOZE_HOURS)) 

187 except (TypeError, ValueError): 

188 snooze_hours = DEFAULT_SNOOZE_HOURS 

189 if snooze_hours < 1: 

190 snooze_hours = DEFAULT_SNOOZE_HOURS 

191 style = raw.get("style") 

192 targets, rejected = split_targets(raw.get("targets")) 

193 if rejected: 

194 _LOGGER.warning( 

195 "Home Keeper dropped notify target(s) %s: only %s* and %s are supported", 

196 ", ".join(rejected), 

197 TARGET_PREFIX, 

198 TARGET_PERSISTENT, 

199 ) 

200 return { 

201 "id": str(raw.get("id") or uuid.uuid4().hex), 

202 "name": str(raw.get("name") or "Notification"), 

203 "profile_id": str(raw["profile_id"]) if raw.get("profile_id") else None, 

204 "targets": targets, 

205 "actions": actions or list(DEFAULT_ACTIONS), 

206 "snooze_hours": snooze_hours, 

207 "style": style if style in STYLES else STYLE_WALK, 

208 "auto": { 

209 "overdue": bool(auto.get("overdue", False)), 

210 "due_soon": bool(auto.get("due_soon", False)), 

211 }, 

212 }

237# ── payload text translation ───────────────────────────────────────────────────── 

238# 

239# Notification payloads go straight to the mobile app, outside HA's own frontend 

240# translation loading, so the strings must be resolved here rather than left for the 

241# frontend to localize. They are NOT part of strings.json/translations/<lang>.json — 

242# hassfest validates that tree against a fixed set of categories (config, services, 

243# entity, ...) and rejects anything else — so they get their own bundled 

244# ``notification_strings/<lang>.json`` files, flat dotted-key tables read directly 

245# (no HA import, keeping this module pure), the same convention 

246# ``frontend/src/locales/*.json`` uses for the panel. Pluralization uses Babel's CLDR 

247# plural rules (one/few/many/other) the same way ``frontend/src/i18n.ts`` uses the 

248# browser's ``Intl.PluralRules``: a pluralizable key is stored as ``<key>.<category>`` 

249# and looked up by the category *n* resolves to, falling back to ``<key>.other``. 

250 

251_DEFAULT_LANG = "en" 

252_STRINGS_DIR = Path(__file__).parent / "notification_strings" 

253_TOKEN_RE = re.compile(r"\{(\w+)\}") 

254

256@functools.cache 

257def _notification_strings(lang: str) -> dict[str, str]: 

258 """Load the flat notification string table for *lang*, caching by language.""" 

259 path = _STRINGS_DIR / f"{lang}.json" 

260 if not path.exists(): 

261 return {} 

262 try: 

263 data = json.loads(path.read_text(encoding="utf-8")) 

264 except (OSError, ValueError): 

265 return {} 

266 return data if isinstance(data, dict) else {}

292def _tn(lang: str, key: str, n: int, **params: Any) -> str: 

293 """Translate a pluralizable string, selecting the CLDR category for *n*.""" 

294 category = _babel_locale(lang).plural_form(n) 

295 strings = _notification_strings(lang) 

296 en_strings = _notification_strings(_DEFAULT_LANG) 

297 template = ( 

298 strings.get(f"{key}.{category}") 

299 or strings.get(f"{key}.other") 

300 or en_strings.get(f"{key}.{category}") 

301 or en_strings.get(f"{key}.other", key) 

302 ) 

303 return _interpolate(template, params)
