1"""Home-Assistant-aware half of the to-do list sync. 

2 

3Reads every configured to-do list over ``todo.get_items``, asks the pure planner in 

4``todo_list.py`` what each sync wants done about it, applies the answer with 

5``todo.add_item`` / ``todo.update_item`` / ``todo.remove_item``, and feeds the 

6household's side of the loop — a chore ticked off on somebody's phone — back into 

7the store as a completion. Same split as ``shopping.py`` / ``shopping_sync.py``, 

8and the same HA-facing driver base: ``TodoSyncDriver`` 

9(``todo_sync_driver.py``) carries the re-entrancy guard and pass budget, the 

10defensive list snapshotting, the capability-checked ``todo`` calls and the 

11warn-once ledger, along with the three properties both syncs owe their callers. 

12What stays here is what is particular to syncing profile-filtered chores: many 

13targets rather than one, task events feeding the pass, due dates and 

14descriptions on the items, and an inbound tick meaning a task is done. 

15 

16Where the two syncs diverge is what a *vanished* item means: the shopping-list sync 

17leaves a deleted line deleted, while a to-do list sync may read it as "done" so that 

18providers which drop completed items (Todoist) can still tick a chore off. That 

19divergence lives entirely in the planner — see ``todo_list.py`` — and this 

20driver only supplies the inputs it needs. 

21"""

91 # ── lifecycle ──────────────────────────────────────────────────────────── 

92 @callback 

93 def async_start_listeners(self) -> None: 

94 """Watch the synced lists, and the task changes that feed them. 

95 

96 Subscriptions go through ``entry.async_on_unload``, so unload/reload tears 

97 them down. There is no resubscribe path on purpose: the only thing that 

98 changes which lists are synced is an options save, and that reloads the 

99 entry. 

100 """ 

101 self._entry.async_on_unload(self._async_stop) 

102 targets = sorted( 

103 { 

104 str(profile["sync"]["entity_id"]) 

105 for profile in profiles.synced_profiles(self._configured_syncs()) 

106 } 

107 ) 

108 if targets: 

109 self._entry.async_on_unload( 

110 async_track_state_change_event( 

111 self._hass, targets, self._handle_state_change 

112 ) 

113 ) 

114 for event_name in _TASK_EVENTS: 

115 self._entry.async_on_unload( 

116 self._hass.bus.async_listen(event_name, self._handle_task_event) 

117 )

119 @callback 

120 def _handle_task_event(self, event: Event[Any]) -> None: 

121 # Unforced: ``needs_pass`` gates the read, so a completion on a task no 

122 # sync wants costs nothing. Our own inbound completions echo back through 

123 # here as well, and settle for free. 

124 self._hass.async_create_task(self.async_sync()) 

125 

126 @callback 

127 def async_schedule_sweep(self) -> None: 

128 """Ask for a full pass off the coordinator's periodic tick. 

129 

130 Two things only this notices. A list whose outstanding count happens to 

131 land back where it started — one item ticked off while another was added — 

132 produces no state change at all, so the listener alone can miss a tick-off. 

133 And a task *falling due* is not a store mutation, so no task event fires 

134 for it: the periodic pass is what puts a newly-overdue chore on the list. 

135 Costs nothing until something is configured or synced. 

136 

137 A third thing rests on this now, and on it being **unconditional**: an add 

138 we could not confirm is held rather than repeated 

139 (``todo_list.UNCONFIRMED_GRACE``), and nothing about that grace expiring 

140 is a store mutation or a list state change either. This pass is the only 

141 thing that comes back to look, so gating it on ``needs_pass`` — which 

142 answers False for a held entry, correctly, since a pending write is not 

143 drift — would turn every such hold into a permanent one. The early-out 

144 below is deliberately about having *nothing tracked at all*, which a held 

145 entry is not. 

146 """ 

147 if self._stopped: 

148 return 

149 configured = profiles.synced_profiles(self._configured_syncs()) 

150 if not configured and not self._coordinator.store.get_todo_list_items(): 

151 return 

152 self._hass.async_create_task(self.async_sync(force=True))

154 # ── the pass ───────────────────────────────────────────────────────────── 

155 async def _sync_once(self, *, force: bool) -> bool: 

156 """One plan-and-apply pass. Returns True when another pass is warranted.""" 

157 if self._stopped: 

158 return False 

159 store = self._coordinator.store 

160 tracked = store.get_todo_list_items() 

161 synced = self._resolve_syncs() 

162 # Tasks are enriched with their **effective** labels/area — the ones they 

163 # inherit from a device or an area — by the same helper the notifier uses, 

164 # so a profile picks the same tasks for a synced list as it does for a 

165 # notification, the admin list and the card. 

166 enriched = notifier.effective_filter_tasks( 

167 self._hass, list(store.get_tasks().values()) 

168 ) 

169 # One instant for the whole pass: what a profile surfaces and how long an 

170 # unconfirmed add has been waiting are two readings of the same "now", and 

171 # taking the clock twice would let them disagree across the reads between. 

172 now = dt_util.now() 

173 desired = todo_list.desired_by_sync(synced, enriched, now=now) 

174 if not force and not todo_list.needs_pass( 

175 tracked=tracked, desired=desired, synced=synced 

176 ): 

177 return False 

178 

179 targets = { 

180 str(profile["sync"]["entity_id"]) 

181 for profile in profiles.synced_profiles(synced) 

182 } 

183 items_by_entity = await self._read_lists( 

184 todo_list.lists_to_read(tracked, synced), targets=targets 

185 ) 

186 if self._stopped: 186 ↛ 187line 186 didn't jump to line 187 because the condition on line 186 was never true

187 return False 

188 capabilities = { 

189 entity_id: self._capabilities(entity_id) for entity_id in items_by_entity 

190 } 

191 plan = todo_list.plan_sync( 

192 synced=synced, 

193 tracked=tracked, 

194 desired=desired, 

195 items_by_entity=items_by_entity, 

196 capabilities=capabilities, 

197 now=now, 

198 ) 

199 settled = await self._apply(plan, before=tracked) 

200 if self._stopped: 200 ↛ 201line 200 didn't jump to line 201 because the condition on line 200 was never true

201 return False 

202 completed = await self._complete_tasks(plan) 

203 # Persist before settling: the settle below reconciles and refreshes, and 

204 # everything that wakes off it — the task events it fires, the sweep on the 

205 # refresh — re-enters this class. What they find should be what this pass 

206 # concluded, not what the one before it did. 

207 await store.async_set_todo_list_items(settled) 

208 if not completed: 

209 return False 

210 # An inbound completion has to behave like every other completion surface — 

211 # consumables spent, buy reminders reconciled, entities refreshed — so it 

212 # goes through the same single decision point rather than a bare refresh. 

213 await self._coordinator.async_settle_buy_tasks() 

214 return True

216 async def _complete_tasks(self, plan: todo_list.TodoListPlan) -> bool: 

217 """Complete the tasks whose items the household ticked off. 

218 

219 Nothing is restored to the bookkeeping here, and both failure paths mean 

220 it. A task deleted between plan and apply is gone, and the item somebody 

221 ticked off is their record of it now. A task that *refuses* remote 

222 completion — ``require_tag_scan``, by design — must not be quietly re-bound 

223 to a line already ticked off either: leaving the entry dropped puts a fresh 

224 open item back on the list next pass, which is honest feedback that the 

225 tick did not take. 

226 """ 

227 store = self._coordinator.store 

228 completed = False 

229 for op in plan.complete: 

230 try: 

231 await store.complete_task(op.task_id, origin=ORIGIN_TODO_SYNC) 

232 except KeyError: 

233 # Deleted while the pass was in flight; its item is somebody else's 

234 # record now and there is nothing left to sync. 

235 _LOGGER.debug( 

236 "Task %s was gone before the to-do list tick reached it", 

237 op.task_id, 

238 ) 

239 except TaskValidationError as err: 

240 self._warn_once( 

241 f"complete:{op.task_id}", 

242 "Home Keeper could not complete task %s from a to-do list (%s), " 

243 "so the tick did not take and the task will reappear on the list", 

244 op.task_id, 

245 err, 

246 ) 

247 else: 

248 completed = True 

249 return completed

251 # ── reading ────────────────────────────────────────────────────────────── 

252 def _configured_syncs(self) -> list[dict[str, Any]]: 

253 """The stored profiles, normalized — a sync is a profile's ``sync`` block. 

254 

255 A profile whose ``sync.entity_id`` is ``""`` has its sync switched off, and 

256 stays in the list: the planner has to see it to take back off whatever it 

257 put on a list before. 

258 """ 

259 return profiles.normalize_profiles( 

260 current_options(self._entry).get(OPTION_PROFILES, []) 

261 ) 

262 

263 def _resolve_syncs(self) -> list[dict[str, Any]]: 

264 """The configured profiles, with any pointed at our own list switched off. 

265 

266 Syncing Home Keeper's tasks onto Home Keeper's own to-do list is a 

267 feedback loop, and that list declares only ``UPDATE_TODO_ITEM`` so it could 

268 not accept the items anyway. Blanking the target rather than dropping the 

269 profile keeps the planner's own rule intact: a sync the user turned off 

270 clears what it wrote, so anything already on that list still comes back off 

271 it. 

272 """ 

273 ours = set(own_todo_entity_ids(self._hass)) 

274 resolved: list[dict[str, Any]] = [] 

275 for profile in self._configured_syncs(): 

276 entity_id = str(profile["sync"]["entity_id"]) 

277 if entity_id and entity_id in ours: 

278 self._warn_once( 

279 f"own:{entity_id}", 

280 "Home Keeper cannot sync its tasks onto its own to-do list " 

281 "(%s); pick a different list in Settings", 

282 entity_id, 

283 ) 

284 profile = {**profile, "sync": {**profile["sync"], "entity_id": ""}} 

285 resolved.append(profile) 

286 return resolved

288 def _capabilities(self, entity_id: str) -> frozenset[str]: 

289 """Which optional item fields *entity_id* can actually hold. 

290 

291 The planner neither writes nor diffs a field outside this set, so a list 

292 without due dates is never told one — otherwise every pass would "fix" the 

293 same item forever, because the value it wrote was dropped on arrival. 

294 """ 

295 caps: set[str] = set() 

296 if self._supports(entity_id, TodoListEntityFeature.SET_DUE_DATE_ON_ITEM): 

297 caps.add(todo_list.CAP_DUE_DATE) 

298 if self._supports(entity_id, TodoListEntityFeature.SET_DESCRIPTION_ON_ITEM): 

299 caps.add(todo_list.CAP_DESCRIPTION) 

300 return frozenset(caps)

302 # ── writing ────────────────────────────────────────────────────────────── 

303 async def _apply( 

304 self, plan: todo_list.TodoListPlan, *, before: dict[str, dict[str, Any]] 

305 ) -> dict[str, dict[str, Any]]: 

306 """Run the plan and return the bookkeeping that actually holds. 

307 

308 *before* is what was tracked going in: an operation that fails restores its 

309 entry from there, so the next pass retries rather than forgetting. 

310 """ 

311 settled = {key: dict(entry) for key, entry in plan.tracked.items()} 

312 for remove in plan.remove: 

313 ok = await self._call( 

314 remove.entity_id, 

315 TodoListEntityFeature.DELETE_TODO_ITEM, 

316 "remove_item", 

317 {"entity_id": remove.entity_id, "item": [remove.item]}, 

318 ) 

319 if not ok and remove.key in before: 

320 settled[remove.key] = dict(before[remove.key]) 

321 for update in plan.update: 

322 data: dict[str, Any] = { 

323 "entity_id": update.entity_id, 

324 "item": update.item, 

325 } 

326 # Each field is present only when the planner asked for it, and the 

327 # optional two are already capability-gated there — so there is nothing 

328 # left to strip here. 

329 if update.status is not None: 

330 data["status"] = update.status 

331 if update.rename is not None: 

332 data["rename"] = update.rename 

333 if update.due is not None: 

334 data["due_date"] = update.due 

335 if update.description is not None: 

336 data["description"] = update.description 

337 ok = await self._call( 

338 update.entity_id, 

339 TodoListEntityFeature.UPDATE_TODO_ITEM, 

340 "update_item", 

341 data, 

342 ) 

343 if not ok and update.key in before: 

344 settled[update.key] = dict(before[update.key]) 

345 for add in plan.add: 

346 payload: dict[str, Any] = {"entity_id": add.entity_id, "item": add.summary} 

347 if add.due is not None: 

348 payload["due_date"] = add.due 

349 if add.description is not None: 

350 payload["description"] = add.description 

351 ok = await self._call( 

352 add.entity_id, 

353 TodoListEntityFeature.CREATE_TODO_ITEM, 

354 "add_item", 

355 payload, 

356 ) 

357 if not ok: 

358 settled.pop(add.key, None) 

359 return settled
