1"""Pure aggregation for the task **count sensors** (no HA imports). 

2 

3A count sensor answers "how many tasks does this profile surface right now?" in one 

4number a dashboard badge can show, with the neighbouring counts and the next task up 

5as attributes. :func:`count_tasks` does the whole computation over plain dicts and an 

6injected ``now``, so every branch is unit-testable without a Home Assistant runtime — 

7the ``recurrence.py`` / ``sensor_tasks.py`` contract. ``sensor.py`` is a thin 

8publisher over it. 

9 

10Nothing here re-implements the profile status ladder. The state runs through 

11``profiles.matches_filter`` and the scope through ``profiles.due_queue``, so a count 

12can never disagree with the panel, the card, a notification or a to-do list sync 

13about what the same profile means. 

14 

15Two definitions this module pins, because a user watches the numbers and an automation 

16depends on them: 

17 

18* **``due_soon`` is the strict band** — due inside the window and *not yet* overdue, 

19 exactly ``recurrence.is_due_soon`` and the ``binary_sensor`` attribute of the same 

20 name. So a profile whose status is ``due_soon`` has a state of ``overdue + 

21 due_soon``, not ``due_soon``. 

22* **``due_today`` is a calendar date**, not a 24-hour window: the task falls due on 

23 *today* in ``now``'s own timezone. That is what ``home_keeper.set_due_today`` 

24 means, and a badge reading "3 due today" that counted 01:00 tomorrow would be wrong 

25 to the person reading it. 

26 

27``now`` must therefore be timezone-aware **local** time — ``dt_util.now()``, never 

28``dt_util.utcnow()``. A UTC ``now`` shifts ``due_today`` by the offset with no error 

29to signal it. ``tests/unit/test_task_counts.py`` pins that with a non-UTC zone. 

30""" 

31 

32from __future__ import annotations 

33 

34from datetime import date, datetime, timedelta 

35from typing import Any 

36 

37from . import profiles, recurrence 

38from .transitions import DUE_SOON_WINDOW 

39 

40__all__ = ["COUNT_KEYS", "count_tasks"] 

41 

42#: Every key :func:`count_tasks` returns. ``state`` becomes the sensor's state and the 

43#: rest become its attributes verbatim, so this tuple is the published contract. 

44COUNT_KEYS = ( 

45 "state", 

46 "total", 

47 "overdue", 

48 "due_soon", 

49 "due_today", 

50 "next_due", 

51 "next_task_name", 

52 "next_task_id", 

53 "most_overdue_days", 

54) 

55 

56#: Seconds in a day, for the ``most_overdue_days`` conversion. 

57_DAY_SECONDS = 86400.0 

58 

59 

60def _due_date(task: dict[str, Any], now: datetime) -> date: 

61 """The calendar date *task* falls due on, read in ``now``'s timezone. 

62 

63 ``next_due`` is guaranteed non-None by ``matches_filter``, which every task in 

64 the scope has passed — the same invariant ``profiles._due_key`` reads on. 

65 """ 

66 return datetime.fromisoformat(task["next_due"]).astimezone(now.tzinfo).date() 

67 

68 

69def _days_overdue(task: dict[str, Any], now: datetime) -> float: 

70 """How many days *task* is past its due instant, to one decimal place.""" 

71 due = datetime.fromisoformat(task["next_due"]) 

72 return round((now - due).total_seconds() / _DAY_SECONDS, 1) 

73 

74 

75def count_tasks( 

76 tasks: list[dict[str, Any]], 

77 filt: dict[str, Any], 

78 *, 

79 now: datetime, 

80 window: timedelta = DUE_SOON_WINDOW, 

81) -> dict[str, Any]: 

82 """Count the tasks *filt* scopes to, and name the next one up. 

83 

84 *tasks* must already carry effective label and area ids — the HA-aware caller 

85 resolves those with ``notifier.effective_filter_tasks`` before calling, the same 

86 enrichment the notifier does, or a profile filtering on an inherited label counts 

87 a different set here than the panel shows. 

88 

89 ``state`` honours *filt*'s own status, so it is the number that profile surfaces. 

90 Every other count is taken over the profile's **scope** — the same filter with the 

91 status widened to ``all`` — so ``total`` stays the size of the profile's world 

92 however its status tier is set, and the pointers name a task even when the state 

93 is 0. 

94 """ 

95 scope_filt = profiles.with_status(filt, profiles.STATUS_ALL) 

96 # due_queue sorts earliest next_due first, so scope[0] is the next task up and the 

97 # first overdue entry is the most overdue. No second sort. The window is left at 

98 # its default deliberately: the scope status is "all", which admits every enabled 

99 # scheduled task whatever the window says, so passing one here would read as if it 

100 # mattered. 

101 scope = profiles.due_queue(tasks, scope_filt, now=now) 

102 overdue = [task for task in scope if recurrence.is_overdue(task, now=now)] 

103 today = now.date() 

104 upcoming = scope[0] if scope else None 

105 return { 

106 "state": sum( 

107 1 

108 for task in scope 

109 if profiles.matches_filter(task, filt, now=now, window=window) 

110 ), 

111 "total": len(scope), 

112 "overdue": len(overdue), 

113 "due_soon": sum( 

114 1 for task in scope if recurrence.is_due_soon(task, window, now=now) 

115 ), 

116 "due_today": sum(1 for task in scope if _due_date(task, now) == today), 

117 "next_due": upcoming.get("next_due") if upcoming else None, 

118 "next_task_name": upcoming.get("name") if upcoming else None, 

119 "next_task_id": upcoming.get("id") if upcoming else None, 

120 "most_overdue_days": _days_overdue(overdue[0], now) if overdue else None, 

121 }
