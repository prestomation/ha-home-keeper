1"""Item matching shared by the two to-do list syncs. 

2 

3``shopping.py`` and ``todo_list.py`` plan against the same thing — somebody 

4else's ``todo.*`` list, read back over ``todo.get_items`` as plain dicts — and 

5both have to answer the same questions about it before either state machine gets 

6a look in: how is this item addressed in a service call, has it been ticked off, 

7which live item is the one a bookkeeping entry points at, and is there already an 

8open item reading like this? Those are facts about a to-do list rather than about 

9either sync, so they live here once and the planners keep only what genuinely 

10differs between them — what a *vanished* line means, what a key is, when a line 

11is wanted at all. 

12 

13Pure like the planners it serves: nothing here imports Home Assistant, so every 

14branch is unit-testable without an HA runtime (see 

15``tests/unit/test_todo_items.py``). 

16"""

31# ``TodoItemStatus`` values, as the ``todo.get_items`` response spells them. 

32STATUS_NEEDS_ACTION = "needs_action" 

33STATUS_COMPLETED = "completed"

36def item_identity(item: dict[str, Any]) -> str: 

37 """How a to-do item is addressed in a service call. 

38 

39 ``todo.update_item`` / ``todo.remove_item`` accept either the item's uid or 

40 its summary, so a list that does not hand out uids is still addressable. 

41 """ 

42 uid = item.get("uid") 

43 if isinstance(uid, str) and uid: 

44 return uid 

45 return str(item.get("summary") or "")

48def item_is_open(item: dict[str, Any]) -> bool: 

49 """True unless the item has been ticked off.""" 

50 return item.get("status") != STATUS_COMPLETED

53def resolve_tracked( 

54 items: list[dict[str, Any]], 

55 *, 

56 entity_id: str, 

57 uid: Any, 

58 summary: str, 

59 claimed: set[tuple[str, str]], 

60) -> dict[str, Any] | None: 

61 """Find the live item a tracked entry points at. 

62 

63 The uid is authoritative when we captured one. Otherwise we fall back to the 

64 summary — that is how a freshly added item is picked up on the next pass 

65 (``todo.add_item`` returns nothing, so there is no uid to record at the 

66 time), and how a sync re-attaches to its own items if the bookkeeping is 

67 ever lost. An open item wins over a ticked-off one with the same text. 

68 """ 

69 if isinstance(uid, str) and uid: 

70 for item in items: 

71 claim = (entity_id, item_identity(item)) 

72 if item.get("uid") == uid and claim not in claimed: 

73 return item 

74 by_summary = [ 

75 item 

76 for item in items 

77 if item.get("summary") == summary 

78 and (entity_id, item_identity(item)) not in claimed 

79 ] 

80 for item in by_summary: 

81 if item_is_open(item): 

82 return item 

83 return by_summary[0] if by_summary else None

86def find_open( 

87 items: list[dict[str, Any]], 

88 *, 

89 entity_id: str, 

90 summary: str, 

91 claimed: set[tuple[str, str]], 

92) -> dict[str, Any] | None: 

93 """An un-ticked item already reading *summary*, if the list has one.""" 

94 for item in items: 

95 if ( 

96 item_is_open(item) 

97 and item.get("summary") == summary 

98 and (entity_id, item_identity(item)) not in claimed 

99 ): 

100 return item 

101 return None
