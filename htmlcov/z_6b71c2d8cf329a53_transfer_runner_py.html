1"""Running the portable document against a live Home Assistant. 

2 

3``transfer.py`` decides *what* an import does, with no Home Assistant in sight so the 

4whole decision is unit-testable. This module is the half that needs the running 

5system: the registries a document's names resolve against, the device provisioning a 

6new appliance triggers, and the entry reload that rebuilds the entity set afterwards. 

7 

8It lives apart from ``__init__.py`` for one reason: the service and its websocket twin 

9must run the *same* code, and ``websocket_api`` cannot import ``__init__`` — the same 

10split, and the same reason, as ``notifications.py`` (pure) and ``notifier.py``. 

11""" 

12 

13from __future__ import annotations 

14 

15from typing import Any 

16 

17from homeassistant.core import HomeAssistant 

18from homeassistant.helpers import area_registry as ar 

19from homeassistant.helpers import device_registry as dr 

20from homeassistant.util import dt as dt_util 

21 

22from . import devices, transfer 

23from .const import PANEL_VERSION 

24from .coordinator import HomeKeeperCoordinator 

25 

26 

27async def async_export_document( 

28 hass: HomeAssistant, coord: HomeKeeperCoordinator, data: dict[str, Any] 

29) -> dict[str, Any]: 

30 """Build the portable document. Shared by the service and its websocket twin.""" 

31 document = transfer.build_document( 

32 coord.store.list_tasks(), 

33 coord.store.list_assets(), 

34 area_names=devices.area_names(hass), 

35 version=PANEL_VERSION, 

36 now=dt_util.now(), 

37 include=data.get("include"), 

38 ) 

39 return {"document": document, "yaml": transfer.document_to_yaml(document)}

42async def async_import_document( 

43 hass: HomeAssistant, coord: HomeKeeperCoordinator, data: dict[str, Any] 

44) -> dict[str, Any]: 

45 """Plan an import, then apply it unless this is a dry run. 

46 

47 Two passes, and the split is the whole safety story. ``transfer.plan_import`` is 

48 pure and touches nothing: it decides every record's fate against the live store 

49 and reports *every* problem at once, each with a path. Only a plan with no error 

50 is applied — so a document that is wrong anywhere writes nothing, which is what 

51 makes a plain (non-dry-run) import safe to point at a generated file and gives 

52 whoever generated it a clean fix-and-retry loop. 

53 

54 The applying pass is ordered by a dependency the panel never has to think about: 

55 a task attaches to an appliance's *device*, and a virtual appliance has no device 

56 until it is provisioned. So appliances are written first, ``async_reconcile_assets`` 

57 runs once for the whole batch (it writes each new device id back onto its asset), 

58 and only then are the tasks written with their placeholder device references 

59 resolved. One reconcile and one reload for the batch, not one per record. 

60 """ 

61 registry = dr.async_get(hass) 

62 plan = transfer.plan_import( 

63 data["document"], 

64 tasks=coord.store.get_tasks(), 

65 assets=coord.store.get_assets(), 

66 area_ids={area.name: area.id for area in ar.async_get(hass).async_list_areas()}, 

67 # The registry's ids, not its entries: a stated device_id is only kept 

68 # when the device really is on this install. ``devices`` iterates entries, 

69 # so take each one's id rather than reaching for a mapping view that its 

70 # ``Collection`` type does not promise. 

71 device_ids=frozenset(device.id for device in registry.devices), 

72 match=data.get("match", "auto"), 

73 now=dt_util.now(), 

74 ) 

75 dry_run = bool(data.get("dry_run")) 

76 if dry_run or not plan.ok: 

77 return plan.as_report(dry_run=dry_run) 

78 

79 planned_assets = plan.for_section("appliances") 

80 if planned_assets: 

81 await coord.store.async_import_records( 

82 assets_to_write=[ 

83 (r.record_id, r.payload, r.action == "create") for r in planned_assets 

84 ], 

85 tasks_to_write=[], 

86 ) 

87 # Provisions a device for every virtual appliance and writes its id back onto 

88 # the asset — the only moment those ids come into existence. 

89 await devices.async_reconcile_assets(hass, coord.entry, coord.store) 

90 

91 planned_tasks = plan.for_section("tasks") 

92 unattached: list[transfer.Problem] = [] 

93 if planned_tasks: 

94 # Build the write list rather than editing the plan. `PlannedRecord` is 

95 # frozen, and a frozen record whose dict gets rewritten underneath it is a 

96 # half-promise: the plan should stay the record of what was *planned*, so 

97 # anything that reads it back — a report, a log, a future retry — sees the 

98 # same thing the preview showed. 

99 tasks_to_write: list[tuple[str, dict[str, Any], bool]] = [] 

100 for record in planned_tasks: 

101 payload = record.payload 

102 asset_id = transfer.planned_asset_id(payload.get("device_id")) 

103 if asset_id is not None: 

104 device_id = (coord.store.get_asset(asset_id) or {}).get("device_id") 

105 payload = {**payload, "device_id": device_id} 

106 if not device_id: 

107 # Provisioning is the one step that can still fail after 

108 # validation passed, and the appliances are already written by 

109 # now, so there is nothing to roll back to. Import the task 

110 # standalone and *say so*: a task that quietly lost its 

111 # appliance is the kind of thing a migration notices months late. 

112 unattached.append( 

113 transfer.Problem( 

114 section="tasks", 

115 index=record.index, 

116 path=f"tasks[{record.index}].appliance", 

117 message=( 

118 f'"{record.name}" was imported without its appliance: ' 

119 "Home Assistant did not give that appliance a device. " 

120 "Open the appliance, then set the task's appliance " 

121 "again." 

122 ), 

123 severity="warning", 

124 ) 

125 ) 

126 tasks_to_write.append( 

127 (record.record_id, payload, record.action == "create") 

128 ) 

129 await coord.store.async_import_records( 

130 assets_to_write=[], tasks_to_write=tasks_to_write 

131 ) 

132 

133 if plan.records: 

134 # Imported parts may imply derived tasks, and the entity set has to be rebuilt 

135 # for everything that arrived — the same finish an asset mutation does. 

136 await coord.store.reconcile_part_tasks() 

137 await coord.store.reconcile_buy_tasks() 

138 await hass.config_entries.async_reload(coord.entry.entry_id) 

139 report = plan.as_report(dry_run=False) 

140 report["problems"].extend(problem.as_dict() for problem in unattached) 

141 return report
