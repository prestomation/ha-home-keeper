1"""Home-Assistant-aware driver for sensor-based tasks. 

2 

3Subscribes to the entities that sensor-based tasks are bound to, reads their live 

4values — numeric for ``usage``/``threshold``, the state string for ``state`` — and 

5feeds the pure evaluators in ``sensor_tasks.py`` to arm a task (via 

6``store.trigger_task``) or stamp/reset a usage baseline (via 

7``store.set_sensor_baseline``). Unlike ``problem_sync.py`` this is **evaluation 

8only**: sensor tasks are user-created, so there is no registry enumeration, 

9auto-creation/deletion, exclusion options, or entry reload — the watcher never 

10changes which tasks exist, only their armed/dormant state and meter baseline. 

11 

12Edge state for the threshold/state modes (was-the-condition-true, when-it-crossed) 

13lives in this object's memory and is baselined on startup (``async_baseline``) so a 

14restart never replays a spurious arm — mirroring how the coordinator baselines the 

15overdue/due-soon transitions. A binding with a ``for_seconds`` hold also gets a timer. 

16The hold ends in its own time, so the watcher books one re-evaluation per task for the 

17moment the pure evaluator says the hold is due, and releases every pending timer on 

18unload. Without that timer the hold ends nowhere: the only things that re-evaluate the 

19task are the next state change of the bound entity — which an entity that has gone 

20unavailable never sends — and the coordinator's five-minute tick. 

21 

22Completion normally flows through the user surfaces; the 

23one exception is a binding with ``clear_on_recover``, where the watcher completes the 

24task itself (tagged ``ORIGIN_SENSOR_RECOVER``) once the condition goes away. 

25"""

103def _raw_reading(hass: HomeAssistant, cfg: dict[str, Any] | None) -> Any | None: 

104 """The raw value a sensor binding points at, or ``None`` if there isn't one. 

105 

106 Resolves the binding to a live value once — honouring the optional ``attribute`` 

107 (read that attribute instead of the state) and rejecting a missing / unavailable / 

108 unknown entity — so the numeric and state readers can't disagree about what 

109 "there is no reading" means. 

110 """ 

111 if not cfg: 

112 return None 

113 entity_id = cfg.get("entity_id") 

114 if not entity_id: 

115 return None 

116 state = hass.states.get(entity_id) 

117 if state is None or state.state in (STATE_UNKNOWN, STATE_UNAVAILABLE, "", None): 

118 return None 

119 attribute = cfg.get("attribute") 

120 return state.attributes.get(attribute) if attribute else state.state

132def read_sensor_state(hass: HomeAssistant, cfg: dict[str, Any] | None) -> str | None: 

133 """Read the live **state string** a sensor binding points at, or ``None``.

135 The ``state`` mode's counterpart to :func:`read_sensor_value`: a binary sensor

144def read_availability_status(hass: HomeAssistant, cfg: dict[str, Any] | None) -> str: 

145 """Classify a binding's *availability*: available / unavailable / missing. 

146 

147 The ``availability`` mode's counterpart to :func:`read_sensor_value` and 

148 :func:`read_sensor_state`, and **inverts** the "no reading = do nothing" policy 

149 those two share: this mode arms *because* the entity is unavailable/unknown, 

150 so the caller needs to distinguish "entity is unreachable" (arm signal) from 

151 "entity is not yet loaded" (indeterminate; hold edge state so a boot-time gap 

152 can never fabricate a spurious arm or clear). 

153 

154 * ``"missing"`` — the binding has no entity_id, or the entity isn't in the 

155 state machine yet (restored on boot / never seen). Indeterminate. 

156 * ``"unavailable"`` — the entity is present but reporting ``unavailable`` / 

157 ``unknown``; or an attribute binding whose target attribute is missing or 

158 ``None``. The arm signal. 

159 * ``"available"`` — the entity has a real state and (if an attribute is 

160 bound) the attribute has a real value. 

161 

162 Mirrors ``problem_sync._is_problem``'s three-way discipline. 

163 """ 

164 if not cfg: 

165 return sensor_tasks.AVAILABILITY_MISSING 

166 entity_id = cfg.get("entity_id") 

167 if not entity_id: 

168 return sensor_tasks.AVAILABILITY_MISSING 

169 state = hass.states.get(entity_id) 

170 if state is None: 

171 return sensor_tasks.AVAILABILITY_MISSING 

172 if state.state in (STATE_UNAVAILABLE, STATE_UNKNOWN, "", None): 

173 return sensor_tasks.AVAILABILITY_UNAVAILABLE 

174 attribute = cfg.get("attribute") 

175 if attribute: 

176 value = state.attributes.get(attribute) 

177 if value is None or value == "": 

178 return sensor_tasks.AVAILABILITY_UNAVAILABLE 

179 return sensor_tasks.AVAILABILITY_AVAILABLE

182class SensorTaskWatcher: 

183 """Evaluates sensor-based tasks against their bound entities.""" 

184 

185 def __init__( 

186 self, 

187 hass: HomeAssistant, 

188 entry: ConfigEntry, 

189 coordinator: HomeKeeperCoordinator, 

190 ) -> None: 

191 self._hass = hass 

192 self._entry = entry 

193 self._coordinator = coordinator 

194 self._unsub_state: CALLBACK_TYPE | None = None 

195 self._tracked: tuple[str, ...] = () 

196 # In-memory threshold edge state, keyed by task id: 

197 # {"condition_met": bool, "crossed_at": datetime | None} 

198 self._edge: dict[str, dict[str, Any]] = {} 

199 # One pending "the hold is up" timer per task id, keyed the same way. A 

200 # ``for_seconds`` hold ends in its own time, and the bound entity sends no 

201 # state change to mark the moment. An entity that has gone unavailable sends 

202 # nothing ever again, so the watcher books the re-evaluation itself. Each 

203 # entry is the timer's unsubscribe callback, released the same way as 

204 # ``_unsub_state``. 

205 self._hold_timers: dict[str, CALLBACK_TYPE] = {} 

206 # In-memory usage-meter reset-candidate state, keyed by task id: the reading 

207 # from a prior below-baseline tick, awaiting a second consecutive one before 

208 # we treat it as a genuine meter reset (debounces a transient dip/blip to 0). 

209 # Held in memory only — never persisted — so a restart safely re-evaluates 

210 # from the current reading rather than acting on a half-seen reset. 

211 self._usage_reset: dict[str, float | None] = {}

216 return { 

217 tid: task 

218 for tid, task in self._coordinator.store.get_tasks().items() 

219 if task.get("recurrence_type") == REC_SENSOR and task.get("enabled", True) 

220 }

222 def _bound_entities(self) -> tuple[str, ...]: 

223 ids = { 

224 eid 

225 for task in self._sensor_tasks().values() 

226 if (eid := sensor_tasks.bound_entity_id(task)) 

227 } 

228 return tuple(sorted(ids))

231 async def async_baseline(self) -> None: 

232 """Baseline edge state and usage baselines without arming anything. 

233 

234 Called once during setup (before the watcher is attached to the coordinator) 

235 so the first evaluation only reacts to genuine transitions: a threshold sensor 

236 already above its limit at boot is recorded as already-met (no rising edge), and 

237 a usage task with no baseline yet is anchored to the current reading. 

238 

239 Tasks made after the last baseline pass are the one exception (see 

240 :func:`async_mark_tasks_new`). Materializing a declarative companion's tasks 

241 reloads the config entry, which runs this method again; baselining a task made 

242 a second ago would eat the very edge the user made the recipe for. Their edge 

243 is left unset, so the first evaluation reads a standing condition as a fresh 

244 crossing and arms them. The set is consumed here, so it can neither grow 

245 without bound nor reach an unrelated later reload. 

246 """ 

247 fresh = _new_tasks_store(self._hass).pop(self._entry.entry_id, set()) 

248 for tid, task in self._sensor_tasks().items(): 

249 cfg = sensor_tasks.sensor_config(task) 

250 if cfg is None: 

251 continue 

252 mode = cfg.get("mode") 

253 if tid in fresh and sensor_tasks.holds_edge_state(mode): 

254 # Made after the last baseline: leave the edge unset so the first 

255 # evaluation arms on a condition that is already true. A usage meter 

256 # has no edge, so it is still anchored below. 

257 continue 

258 if mode == SENSOR_MODE_THRESHOLD: 

259 reading = read_sensor_value(self._hass, cfg) 

260 met = reading is not None and sensor_tasks.compare( 

261 reading, cfg["comparison"], float(cfg["value"]) 

262 ) 

263 self._edge[tid] = {"condition_met": met, "crossed_at": None} 

264 elif mode == SENSOR_MODE_STATE: 

265 # Record an already-matching sensor as met-without-a-crossing, so a 

266 # vacuum still reporting "water tank low" across a restart doesn't 

267 # re-arm a task the user already dealt with. 

268 self._edge[tid] = { 

269 "condition_met": read_sensor_state(self._hass, cfg) 

270 == cfg.get("state"), 

271 "crossed_at": None, 

272 } 

273 elif mode == SENSOR_MODE_AVAILABILITY: 

274 # Record an already-unavailable entity as met-without-a-crossing: 

275 # a device that was offline before HA restarted must not fabricate a 

276 # fresh "gone offline" task the first time we look at it. A ``missing`` 

277 # (not-yet-loaded) entity is indeterminate — treat as not-met so a 

278 # genuine transition later drives the arm. 

279 status = read_availability_status(self._hass, cfg) 

280 self._edge[tid] = { 

281 "condition_met": status == sensor_tasks.AVAILABILITY_UNAVAILABLE, 

282 "crossed_at": None, 

283 } 

284 else: 

285 reading = read_sensor_value(self._hass, cfg) 

286 if reading is not None and cfg.get("baseline") is None: 

287 await self._coordinator.store.set_sensor_baseline(tid, reading)

302 @callback 

303 def _cancel_hold_timers(self) -> None: 

304 """Drop every pending hold timer. 

305 

306 Registered with ``async_on_unload`` beside :meth:`_unsubscribe_state`, so a 

307 reload cannot leave a callback behind that would fire into the watcher the 

308 reload replaced. 

309 """ 

310 for unsub in self._hold_timers.values(): 

311 unsub() 

312 self._hold_timers.clear() 

313 

314 @callback 

315 def _cancel_hold_timer(self, tid: str) -> None: 

316 """Drop the pending hold timer of one task, if it has one.""" 

317 unsub = self._hold_timers.pop(tid, None) 

318 if unsub is not None: 

319 unsub()

321 @callback 

322 def _schedule_hold(self, tid: str, due_at: datetime | None) -> None: 

323 """Keep exactly one pending hold timer for *tid*. 

324 

325 *due_at* is what the pure evaluator decided (``None`` = nothing is pending). 

326 The prior timer is always released first, so a task can never hold two: a 

327 state change that re-evaluates the task replaces its timer, and a task that 

328 armed, recovered or lost its hold simply has it cancelled. 

329 """ 

330 self._cancel_hold_timer(tid) 

331 if due_at is None: 

332 return 

333 self._hold_timers[tid] = async_track_point_in_time( 

334 self._hass, partial(self._handle_hold_due, tid), due_at 

335 ) 

336 

337 @callback 

338 def _handle_hold_due(self, tid: str, _now: datetime) -> None: 

339 """A hold is up: run the same evaluation a state change would run. 

340 

341 The whole pass runs, not just this task, so the arm, the notes refresh and the 

342 events are the ones every other path produces. A task deleted in the meantime 

343 is simply not in the pass. 

344 """ 

345 self._hold_timers.pop(tid, None) # it has fired; there is nothing to cancel 

346 self._hass.async_create_task(self.async_evaluate(refresh=True))

348 @callback 

349 def _resubscribe_state(self) -> None: 

350 """(Re)point the state listener at the currently bound entity set.""" 

351 tracked = self._bound_entities() 

352 if tracked == self._tracked: 

353 return 

354 self._unsubscribe_state() 

355 self._tracked = tracked 

356 if tracked: 

357 self._unsub_state = async_track_state_change_event( 

358 self._hass, list(tracked), self._handle_state_change

367 # ── evaluation ───────────────────────────────────────────────────────────── 

368 async def async_evaluate(self, *, refresh: bool) -> None: 

369 """Evaluate every sensor task once, applying arm / re-baseline decisions. 

370 

371 ``refresh`` requests a coordinator refresh when something armed — set from the 

372 state-change path (which runs outside the coordinator cycle). The periodic 

373 coordinator tick passes ``refresh=False`` because it runs the transition 

374 detection itself, immediately after, in the same cycle. 

375 """ 

376 # Keep the subscription aligned with the live task set (a task may have been 

377 # added/edited/removed since we last subscribed). 

378 self._resubscribe_state() 

379 now = dt_util.now() 

380 changed_any = False 

381 for tid, task in self._sensor_tasks().items(): 

382 cfg = sensor_tasks.sensor_config(task) 

383 if cfg is None: 

384 continue 

385 mode = cfg.get("mode") 

386 if mode == SENSOR_MODE_STATE: 

387 # A missing state is handled inside the evaluator (it holds the edge 

388 # state rather than reading a dropout as a recovery), so unlike the 

389 # numeric modes there's nothing to skip here. 

390 if await self._evaluate_state( 

391 tid, task, state=read_sensor_state(self._hass, cfg), now=now 

392 ): 

393 changed_any = True 

394 continue 

395 if mode == SENSOR_MODE_AVAILABILITY: 

396 # Availability inverts the "no reading" policy: an ``unavailable`` 

397 # entity is the arm signal here, not something to skip. The evaluator 

398 # holds edge state on ``missing`` (not-yet-loaded), so boot doesn't 

399 # fabricate transitions. 

400 if await self._evaluate_availability( 

401 tid, 

402 task, 

403 status=read_availability_status(self._hass, cfg), 

404 now=now, 

405 ): 

406 changed_any = True 

407 continue 

408 reading = read_sensor_value(self._hass, cfg) 

409 if mode == SENSOR_MODE_USAGE: 

410 # A usage task with a time backstop must still be evaluable with no 

411 # reading — an appliance that's been offline for a year still owes its 

412 # annual service. Without one there's nothing a missing reading can 

413 # decide, so skip rather than churn. 

414 if reading is None and not cfg.get("also_every"): 

415 continue 

416 if await self._evaluate_usage(tid, task, reading=reading, now=now): 

417 changed_any = True 

418 else: 

419 if reading is None: 

420 continue # unavailable / non-numeric — never arm on bad data 

421 if await self._evaluate_threshold(tid, task, reading=reading, now=now): 

422 changed_any = True 

423 # Drop edge state for tasks that no longer exist so it can't leak. A task that 

424 # went away (deleted, disabled, or no longer a sensor task) takes its pending 

425 # hold timer with it. 

426 live = set(self._sensor_tasks()) 

427 for stale in [tid for tid in self._edge if tid not in live]: 

428 del self._edge[stale] 

429 for stale in [tid for tid in self._usage_reset if tid not in live]: 

430 del self._usage_reset[stale] 

431 for stale in [tid for tid in self._hold_timers if tid not in live]: 

432 self._cancel_hold_timer(stale) 

433 if changed_any and refresh: 

434 await self._coordinator.async_request_refresh()

497 async def _async_arm(self, tid: str) -> None: 

498 """Arm a task, refreshing a declarative companion's notes first. 

499 

500 The notes of a task a declarative companion manages are rendered from the 

501 entity's live state, but only when the reconciler runs — at creation, and on 

502 registry changes after that. A template that quotes the reading ("{{ state }} 

503 hours left") therefore describes the value from that moment, not the value 

504 that armed the task. Rendering again on the arm makes the note describe the 

505 reading the person is about to read it for. 

506 

507 The arm transition is the only place this happens: rendering on every 

508 evaluation would write to the store on each tick, for no gain on a task 

509 nobody is looking at. The refresh runs *before* the arm so the 

510 ``home_keeper_task_triggered`` event already carries the fresh note. 

511 

512 A refresh must never stop a task from arming, so a failure is logged and 

513 swallowed. 

514 """ 

515 sync = self._coordinator.declarative_sync 

516 if sync is not None: 

517 try: 

518 await sync.async_refresh_task_notes(tid) 

519 except Exception: # pragma: no cover - defensive 

520 _LOGGER.exception("Could not refresh the notes of task %s", tid) 

521 await self._coordinator.store.trigger_task(tid)

523 async def _apply_edge(self, tid: str, decision: dict[str, Any]) -> bool: 

524 """Carry an edge decision's state forward and apply its action. 

525 

526 Returns whether the task's due-state changed. A ``clear_on_recover`` clear 

527 counts as much as an arming: the task drops off the overdue surfaces, and that 

528 should show up immediately rather than at the next periodic tick. 

529 

530 The decision also says when a pending ``for_seconds`` hold completes, which is 

531 the moment this task must be evaluated again. While the bound entity stays 

532 quiet, only the five-minute coordinator tick would do it, and it would arm the 

533 task as much as five minutes late. 

534 """ 

535 self._edge[tid] = { 

536 "condition_met": decision["condition_met"], 

537 "crossed_at": decision["crossed_at"], 

538 } 

539 self._schedule_hold(tid, decision["hold_due_at"]) 

540 action = decision["action"] 

541 if action == sensor_tasks.ACTION_ARM: 

542 await self._async_arm(tid) 

543 return True 

544 if action == sensor_tasks.ACTION_CLEAR: 

545 # The condition went away on its own, so the work is done: record a real 

546 # completion (history, events, the todo/calendar surfaces all follow) and 

547 # tag it so an automation can tell it apart from someone pressing Done. 

548 await self._coordinator.store.complete_task( 

549 tid, origin=ORIGIN_SENSOR_RECOVER 

550 ) 

551 return True 

552 return False
